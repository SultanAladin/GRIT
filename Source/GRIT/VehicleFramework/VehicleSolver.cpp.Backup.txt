/* Initialize vehicle solver with default physics settings and placeholder mesh */
AVehicleSolver::AVehicleSolver()
{
    PrimaryActorTick.bCanEverTick = true;
    
    VehicleHull = CreateDefaultSubobject<UStaticMeshComponent>(TEXT("VehicleHull"));
    RootComponent = VehicleHull;
    VehicleHull->SetSimulatePhysics(true);
    VehicleHull->SetEnableGravity(true);
    
    // 💡 Damping disabled for testing - linear damping simulates air resistance/rolling friction, angular damping prevents unrealistic spinning
    // Re-enable for production (0.01-0.05 for linear) and pair with proper aero drag modeling for authenticity
    VehicleHull->SetLinearDamping(0.0f);                                 // [-] - Disabled for max speed testing
    VehicleHull->SetAngularDamping(0.0f);                                // [-] - Disabled for rotational testing
    VehicleHull->SetCollisionProfileName(TEXT("Vehicle"));
    VehicleHull->BodyInstance.bNotifyRigidBodyCollision = true;
    
    static ConstructorHelpers::FObjectFinder<UStaticMesh> CubeMesh(TEXT("/Engine/BasicShapes/Cube"));
    if (CubeMesh.Succeeded()) { VehicleHull->SetStaticMesh(CubeMesh.Object); VehicleHull->SetRelativeScale3D(FVector(2.0f, 4.0f, 1.0f)); } // Load placeholder mesh
    
    FAssemblyLoadout::LoadDefaultPreset(Axles_GT, AntiRollbars_GT);      // Initialize default 4-wheel configuration
} // End AVehicleSolver()

/*====================================================================================================================================================
                                                                  CORE LIFECYCLE
====================================================================================================================================================*/
/* Initialize suspension system and register physics callback on game start */
void AVehicleSolver::BeginPlay()
{
    Super::BeginPlay();
    ApplyPacejkaPlySteerSymmetry();
    GenerateTrajectoryAtlas();                                           // Generate multi-ray trace patterns for suspension
    CalibrateSuspensionAssembly();                                       // Compute spring rates and rest lengths
    CalibrateSteeringAssembly();                                         // Initialize steering parameters
    ApplyEquilibriumTransform();                                         // Position vehicle at correct ride height on ground
    InitializePhysics();                                                 // Register physics callback with Chaos solver
    InitializeDrivetrain_GameThread();                                   // Initialize powertrain system
    InitializeAerodynamics_GameThread();                                 // Initialize aerodynamics system

#if !UE_BUILD_SHIPPING
    // Initialize static wheel loads for load transfer calculation
    if (PhysicsCallback)
    {
        const int32 WheelCount = Axles_GT.Num();
        StaticWheelLoads_GT.SetNum(WheelCount);
        PhysicsCallback->StaticWheelLoads.SetNum(WheelCount);
        
        // Estimate static load as mass * gravity / wheel count
        const float TotalWeight = VehicleHull ? VehicleHull->GetMass() * 9.80f : 1720.0f * 9.80f;
        const float StaticLoadPerWheel = TotalWeight / FMath::Max(1, WheelCount);
        
        for (int32 i = 0; i < WheelCount; ++i)
        {
            StaticWheelLoads_GT[i] = StaticLoadPerWheel;
            PhysicsCallback->StaticWheelLoads[i] = StaticLoadPerWheel;
        }
    }
#endif // !UE_BUILD_SHIPPING
} // End BeginPlay()

/* Update physics input data every frame for simulation thread */
void AVehicleSolver::Tick(float DeltaTime)
{
    Super::Tick(DeltaTime);

    // Publish input tensor to physics thread (GT -> PT)
    InputTensor_GameThread.Timestamp = GetWorld() ? static_cast<float>(GetWorld()->GetTimeSeconds()) : 0.0f;
    InputConduit.Publish(InputTensor_GameThread);

    if (PhysicsCallback) // Reason: provide updated input data for physics thread
    {
        if (FVehicleSolverInput* Input = PhysicsCallback->GetProducerInputData_External())
        {
#if !UE_BUILD_SHIPPING
            Input->FrameNumber = GFrameCounter;                          // [-] - Debug frame counter
#endif
            Input->World = GetWorld();
            FCollisionQueryParams TraceParams(NAME_None, FCollisionQueryParams::GetUnknownStatId(), false, this);
            TraceParams.bTraceComplex = false;
            TraceParams.AddIgnoredActor(this);
            Input->TraceParams = TraceParams;
            
            // Precompute object query params once (avoids per-trace allocation)
            FCollisionObjectQueryParams OQ;
            OQ.AddObjectTypesToQuery(ECC_WorldStatic);
            OQ.AddObjectTypesToQuery(ECC_WorldDynamic);
            Input->ObjectQueryParams = OQ;
            
            Input->TraceResponse = FCollisionResponseContainer::GetDefaultResponseContainer();
        } // End if (input)
    } // End if (callback)

    // Debug visualization for aerodynamics (ride height and force vectors)
#if !UE_BUILD_SHIPPING
    if (bEnableAeroDebugDraw && VehicleHull)
    {
        DrawAerodynamicsDebug();
    }
#endif // !UE_BUILD_SHIPPING
} // End Tick()

/* Cleanup physics callback when actor is destroyed or level ends */
void AVehicleSolver::EndPlay(const EEndPlayReason::Type EndPlayReason)
{
    ShutdownPhysics();                                                   // Unregister physics callback from Chaos solver

#if !UE_BUILD_SHIPPING
    // Write telemetry CSV files
    if (bEnableTelemetryLogging && TelemetrySampleBuffer.Num() > 0)
    {
        WriteTelemetryCsv();
    }
    
    if (bEnablePerformanceLogging && PerformanceSampleBuffer.Num() > 0)
    {
        WritePerformanceCsv();
    }
    
    if (bEnableAerodynamicsLogging && AerodynamicsSampleBuffer.Num() > 0)
    {
        WriteAerodynamicsCsv();
    }
#endif // !UE_BUILD_SHIPPING

    Super::EndPlay(EndPlayReason);
} // End EndPlay()

/* Handle property changes in editor - auto-recalibrate suspension when parameters are modified */
void AVehicleSolver::PostEditChangeProperty(FPropertyChangedEvent& PropertyChangedEvent)
{
    Super::PostEditChangeProperty(PropertyChangedEvent);

    if (!PropertyChangedEvent.Property) return; // End if (null property)
    
    const FName PropertyName = PropertyChangedEvent.Property->GetFName();
    const FName MemberPropertyName = PropertyChangedEvent.MemberProperty ? PropertyChangedEvent.MemberProperty->GetFName() : NAME_None;

    if (MemberPropertyName == GET_MEMBER_NAME_CHECKED(AVehicleSolver, Axles_GT)) // Reason: check if suspension properties were modified
    {

            PropertyName == GET_MEMBER_NAME_CHECKED(FSuspensionSpecifications, NaturalFrequency) ||
            PropertyName == GET_MEMBER_NAME_CHECKED(FSuspensionSpecifications, DampingRatio) ||
            PropertyName == GET_MEMBER_NAME_CHECKED(FSuspensionSpecifications, MaxTravel) ||
            PropertyName == GET_MEMBER_NAME_CHECKED(FSuspensionSpecifications, MinRaise) ||
            PropertyName == GET_MEMBER_NAME_CHECKED(FSuspensionSpecifications, MaxDrop) ||
            PropertyName == GET_MEMBER_NAME_CHECKED(FSuspensionSpecifications, StaticPreload) ||
            PropertyName == GET_MEMBER_NAME_CHECKED(FSuspensionSpecifications, SuspensionMountLength);
        
        if (bSuspensionModified) // Reason: suspension parameter changed, recalibrate
        {
            CalibrateSuspensionAssembly();
        } // End if (suspension modified)
    } // End if (axles modified)

    if (MemberPropertyName == GET_MEMBER_NAME_CHECKED(AVehicleSolver, SteeringSystem_GT)) // Reason: check if steering properties were modified
    {
        CalibrateSteeringAssembly();
    } // End if (steering modified)

    GenerateTrajectoryAtlas();
    CalibrateSuspensionAssembly();
    CalibrateSteeringAssembly();
    ApplyPacejkaPlySteerSymmetry();
    PrecomputeTireCaches();                                              // Recompute Pacejka caches when properties change

    if (GetWorld() && GetWorld()->IsGameWorld()) // Reason: reinitialize physics if in game world
    {
        ShutdownPhysics();
        InitializePhysics();
    } // End if (game world)
} // End PostEditChangeProperty()

/*====================================================================================================================================================
                                                              PHYSICS INITIALIZATION
====================================================================================================================================================*/
/* Create and register physics callback with Chaos solver, calibrate suspension system */
void AVehicleSolver::InitializePhysics()
{
    bPhysicsCallbackInitialized = false;
    CachedPhysicsProxy = nullptr;
    
    if (UWorld* World = GetWorld()) // Reason: create and register physics callback with Chaos solver
    {
        if (FPhysScene* PhysScene = World->GetPhysicsScene())
        {
            if (Chaos::FPBDRigidsSolver* Solver = static_cast<Chaos::FPBDRigidsSolver*>(PhysScene->GetSolver()))
            {
                PhysicsCallback = Solver->CreateAndRegisterSimCallbackObject_External<FVehicleSolverCallback>();
                PhysicsCallback->VehicleOwner = this;
                PhysicsCallback->InputConduitPtr = &InputConduit;        // Link input conduit for GT->PT communication
                
                bPhysicsCallbackInitialized = true;
                if (FBodyInstance* BodyInstance = VehicleHull->GetBodyInstance())
                {
                    CachedPhysicsProxy = BodyInstance->GetPhysicsActorHandle();
                    PhysicsCallback->Proxy = CachedPhysicsProxy;         // Link physics proxy to callback
                } // End if (body instance)
                
                CalibrateSuspensionAssembly();                           // Compute spring rates and rest lengths
                CalibrateSteeringAssembly();                             // Initialize steering parameters
                ApplyPacejkaPlySteerSymmetry();
                PhysicsCallback->AxleData_PT.InitializeFromAxleMembers(Axles_GT); // Push calibrated data to physics thread
                
                // Rebuild wheel metrics cache with calibrated data
                const FVector UpDir = FVector::UpVector;
                const FVector DownDir = -UpDir;
                PhysicsCallback->AxleData_PT.RebuildWheelMetricsCache(UpDir, DownDir);
                
                PhysicsCallback->SuspensionConfig_PT = SuspensionConfig; // [-] - Full suspension configuration
                PhysicsCallback->AxleData_PT.SprungMasses = SuspensionConfig.SprungMasses; // [kg] - Per-wheel sprung masses
                PhysicsCallback->AntiRollbars_PT = AntiRollbars_GT;  // [-] - Anti-rollbar configuration
                PhysicsCallback->SteeringSystem_PT = SteeringSystem_GT;  // [-] - Steering configuration
                PhysicsCallback->ChassisConfig_PT = ChassisConfig_GT;    // [-] - Chassis geometry
                PhysicsCallback->SteeringGeometryCache_PT.Initialize(ChassisConfig_GT); // [-] - Precompute steering geometry
                PrecomputeTireCaches();                                  // [-] - Precompute Pacejka coefficients
                
                UE_LOG(LogTemp, Log, TEXT("✅ Physics initialized and suspension calibrated"));
            } // End if (solver)
        } // End if (phys scene)
    } // End if (world)
    
    if (!bPhysicsCallbackInitialized) { UE_LOG(LogTemp, Error, TEXT("❌ Failed to initialize physics callback")); }
} // End InitializePhysics()

/* Unregister and cleanup physics callback from Chaos solver */
void AVehicleSolver::ShutdownPhysics()
{
    if (PhysicsCallback) // Reason: unregister physics callback from Chaos solver
    {
        if (UWorld* World = GetWorld())
        {
            if (FPhysScene* PhysScene = World->GetPhysicsScene())
            {
                if (Chaos::FPBDRigidsSolver* Solver = static_cast<Chaos::FPBDRigidsSolver*>(PhysScene->GetSolver()))
                {
                    Solver->UnregisterAndFreeSimCallbackObject_External(PhysicsCallback);
                } // End if (solver)
            } // End if (phys scene)
        } // End if (world)
    } // End if (callback)

    PhysicsCallback = nullptr;
    bPhysicsCallbackInitialized = false;
    CachedPhysicsProxy = nullptr;
} // End ShutdownPhysics()

/* Main physics simulation callback - runs on physics thread before each simulation step */
void FVehicleSolverCallback::OnPreSimulate_Internal()
{
#if !UE_BUILD_SHIPPING
    // Start total frame timing
    const uint64 FrameStartCycles = FPlatformTime::Cycles64();
#endif

    const float DeltaTime = GetDeltaTime_Internal();                     // [s] - Physics timestep

    /*====================================================================================================================================================
                                                                 STAGE 1: VALIDATION
    ====================================================================================================================================================*/
    if (!Proxy) return;
    Chaos::FRigidBodyHandle_Internal* RigidBody = Proxy->GetPhysicsThreadAPI();
    if (!RigidBody) return;
    AVehicleSolver* Vehicle = VehicleOwner;
    if (!Vehicle) return;

    // Read latest input from game thread (GT -> PT)
    const FInputTensor& InputTensor = InputConduitPtr ? InputConduitPtr->Read() : FInputTensor();

    const FVehicleSolverInput* Input = GetConsumerInput_Internal();
    if (!Input || !Input->World) return;                                 // Need world for traces
    
    const int32 WheelCount = AxleData_PT.AxleIDs.Num();
    if (WheelCount == 0) return;

    /*====================================================================================================================================================
                                                          STAGE 2: CAPTURE RIGID BODY STATE
    ====================================================================================================================================================*/
    const FVector Position = RigidBody->X();                             // [cm] - Chassis world position
    const FQuat Rotation = RigidBody->R();                               // [-] - Chassis world rotation
    const FVector VelocityCms = RigidBody->V();                          // [cm⋅s⁻¹] - Linear velocity (Unreal units)
    const FVector CurrentLinearVelMs = VelocityCms * 0.01f;              // [m⋅s⁻¹] - Linear velocity (SI units)
    const FVector AngularVelRad = RigidBody->W();                        // [rad⋅s⁻¹] - Angular velocity

    // Precompute linear acceleration using previous-frame velocity stored in vehicle record
    const FVector PrevVelCms = Vehicle->VehicleRecord_PhysicsThread.PreviousVelocity; // [cm⋅s⁻¹]
    const FVector PrevLinearVelMs = PrevVelCms * 0.01f;                               // [m⋅s⁻¹]

    FVector ComputedAccel = FVector::ZeroVector;                         // [m⋅s⁻²] - Linear acceleration
    if (DeltaTime > 1.e-6f)
    {
        ComputedAccel = (CurrentLinearVelMs - PrevLinearVelMs) / DeltaTime;
    }

    // Build instantaneous vehicle record
    FInstantaneousVehicleRecord Rec;
    Rec.μ_mass = static_cast<float>(RigidBody->M());                     // [kg] - Total vehicle mass
    Rec.HullXfm = FTransform(Rotation, Position);                        // [-] - Chassis world transform
    Rec.σ_centerOfMass = Position;                                       // [cm] - Center of mass world position
    Rec.ν_linearCms = VelocityCms;                                       // [cm⋅s⁻¹] - Linear velocity (Unreal units)
    Rec.ν_linearMs = CurrentLinearVelMs;                                 // [m⋅s⁻¹] - Linear velocity (SI units)

    // FILL MISSING FIELDS NEEDED BY LOAD TRANSFER
    Rec.ν_magnitudeMs = CurrentLinearVelMs.Size();                       // [m⋅s⁻¹] - Speed magnitude
    Rec.α_linear = ComputedAccel;                                       // [m⋅s⁻²] - Linear acceleration vector

    Rec.ω_angularRads = AngularVelRad;                                   // [rad⋅s⁻¹] - Angular velocity
    Rec.ê_longitudinal = Rotation.GetForwardVector();                    // [-] - Forward unit vector
    Rec.ê_lateral = Rotation.GetRightVector();                           // [-] - Right unit vector
    Rec.ê_vertical = Rotation.GetUpVector();                             // [-] - Up unit vector

    /*====================================================================================================================================================
                                                          Steering SIMULATION PIPELINE
    ====================================================================================================================================================*/
    
    // 🎮 Process steering input (must be done before suspension to set wheel angles)
#if !UE_BUILD_SHIPPING
    { SCOPE_TIMER(CurrentPerfSample.Steering_us);
#endif
    ProcessSteering(DeltaTime, InputTensor.Steering, Rec.HullXfm, AxleData_PT);
#if !UE_BUILD_SHIPPING
    }
#endif

    /*====================================================================================================================================================
                                                          SUSPENSION SIMULATION PIPELINE
    ====================================================================================================================================================*/
   

    //  Calculate world-space spring mount positions
    for (int32 i = 0; i < WheelCount; ++i) { AxleData_PT.WorldSpringOrigins[i] = Rec.HullXfm.TransformPosition(AxleData_PT.SuspensionSpecs[i].LocalSpringOrigin); } // End for (spring origins)

    //  Trace suspension displacements (multi-ray contact detection)
#if !UE_BUILD_SHIPPING
    { SCOPE_TIMER(CurrentPerfSample.SuspensionDisplacements_us);
#endif
    ComputeSuspensionDisplacements(Input, WheelCount, Rec.HullXfm, Rec.ê_vertical, -Rec.ê_vertical, AxleData_PT);
#if !UE_BUILD_SHIPPING
    }
#endif

    // Calculate spring and damper forces
#if !UE_BUILD_SHIPPING
    { SCOPE_TIMER(CurrentPerfSample.SuspensionForces_us);
#endif
    ComputeSuspensionForces(Rec, Rec.ν_linearCms, Rec.ω_angularRads, WheelCount, AxleData_PT);
#if !UE_BUILD_SHIPPING
    }
#endif

    // Compute anti-roll bar forces
#if !UE_BUILD_SHIPPING
    { SCOPE_TIMER(CurrentPerfSample.AntiRollbar_us);
#endif
    ComputeAntiRollbarForces(RigidBody, AxleData_PT);
#if !UE_BUILD_SHIPPING
    }
#endif

    /*====================================================================================================================================================
                                                          AERODYNAMICS PASS (ONCE)
    ====================================================================================================================================================*/
    
    // Compute all aerodynamic forces ONCE before load transfer and powertrain
    // CRITICAL: Must zero-initialize to prevent garbage values when aero computation is disabled
    FAerodynamicForces_PT AeroForces;
    FMemory::Memzero(&AeroForces, sizeof(FAerodynamicForces_PT));
    
    if (Rec.ν_magnitudeMs > 0.5f) // Only compute above 0.5 m/s threshold (avoids noise at near-zero speed)
    {
        // Compute minimum ride height across all wheels (for ground effect)
        // Ride height = vertical clearance between underbody and ground
        // Formula: RestLength - SpringDisplacement (compression reduces clearance)
        float MinRideHeight_cm = 50.0f; // [cm] Default fallback (50cm = no ground effect)
        int32 ContactCount = 0;
        
        for (int32 i = 0; i < WheelCount; ++i)
        {
            if (AxleData_PT.bIsInContact[i])
            {
                // RestLength is the suspension travel at rest position
                // SpringDisplacement is positive when compressed, negative when extended
                // So: RestLength - Displacement = current clearance
                const float WheelRideHeight_cm = AxleData_PT.SuspensionSpecs[i].RestLength - AxleData_PT.SpringDisplacements[i];
                MinRideHeight_cm = FMath::Min(MinRideHeight_cm, WheelRideHeight_cm);
                ++ContactCount;
            }
        }
        
        // Only compute aero if at least one wheel is in contact (not fully airborne)
        if (ContactCount > 0)
        {
            // Convert ride height to meters for aero computation
            const float RideHeight_m = FMath::Max(MinRideHeight_cm * 0.01f, 0.01f); // [m] Clamp minimum to 1cm
            
            // Compute velocity direction (normalized)
            const FVector VelDir_World = Rec.ν_linearMs.GetSafeNormal();
            
            // Extract input values for adaptive aero
            const float BrakeInput = InputTensor.Brake;     // [0-1] Brake pedal position
            const float ThrottleInput = InputTensor.Throttle; // [0-1] Throttle pedal position
            const float YawRate_rads = Rec.ω_angularRads.Z;  // [rad/s] Yaw rate for side force
            
            // Compute all aero forces ONCE using physics thread aero package
            AeroForces = ComputeAerodynamicForces(
                AeroPackage_PT,          // Aerodynamic package (wing, splitter, diffuser, etc.)
                Rec.ν_magnitudeMs,       // Vehicle speed [m/s]
                RideHeight_m,            // Minimum ride height [m]
                VelDir_World,            // Velocity direction (normalized)
                BrakeInput,              // Brake input [0-1] for adaptive wing deployment
                ThrottleInput,           // Throttle input [0-1] for drag reduction
                YawRate_rads             // Yaw rate [rad/s] for side force calculation
            );
            
            // Cache ride height for GT debug visualization
            // (AeroForces are stored in AeroForces_PT on the callback)
            if (Vehicle)
            {
                Vehicle->CachedRideHeight_m = RideHeight_m;
            }
            
            // Store aero forces on callback for access from other systems
            AeroForces_PT = AeroForces;
        }
    }

    // Dynamic load transfer (Realtime body-accel based) - uses precomputed aero forces
#if !UE_BUILD_SHIPPING
    { SCOPE_TIMER(CurrentPerfSample.LoadTransfer_us);
#endif
    ComputeLoadTransferRealtime(Rec, DeltaTime, WheelCount, AxleData_PT, AeroForces);
    AxleData_PT.WheelLoads = AxleData_PT.ComputedLoads_LT; // Commit computed loads to wheel loads
#if !UE_BUILD_SHIPPING
    }
#endif

    /*====================================================================================================================================================
                                                          TIRE SLIP COMPUTATION
    ====================================================================================================================================================*/

    // 🔧 Solve tire slip ratios and angles FIRST (needed by Pacejka in SolvePowertrain)
#if !UE_BUILD_SHIPPING
    { SCOPE_TIMER(CurrentPerfSample.ContactSlip_us);
#endif
    for (int32 i = 0; i < WheelCount; ++i) { SolveContactSlip(i, DeltaTime, Rec, AxleData_PT); } // End for (slip solver)
#if !UE_BUILD_SHIPPING
    }
#endif

    /*====================================================================================================================================================
                                                          POWERTRAIN SIMULATION PIPELINE
    ====================================================================================================================================================*/

    // 🔥 Solve complete powertrain (engine, clutch, transmission, differential, wheel dynamics)
    // Uses slip values computed above for Pacejka tire force calculations
    // Aerodynamics drag is applied to chassis here (downforce already in wheel loads)
#if !UE_BUILD_SHIPPING
    { SCOPE_TIMER(CurrentPerfSample.Powertrain_us);
#endif
    SolvePowertrain(DeltaTime, Rec, InputTensor, RigidBody, AxleData_PT, AeroForces);
#if !UE_BUILD_SHIPPING
    }
#endif

    /*====================================================================================================================================================
                                                        STAGE 4: STORE STATE FOR NEXT FRAME
    ====================================================================================================================================================*/
    Rec.PreviousVelocity = Rec.ν_linearCms;
    Rec.PreviousAngularVelocity = Rec.ω_angularRads;
    Vehicle->VehicleRecord_PhysicsThread = Rec;

#if !UE_BUILD_SHIPPING
    /*====================================================================================================================================================
                                                        TELEMETRY & PROFILING PASS
    ====================================================================================================================================================*/
    // Calculate total frame time
    const uint64 FrameEndCycles = FPlatformTime::Cycles64();
    CurrentPerfSample.TotalFrame_us = FPlatformTime::ToSeconds64(FrameEndCycles - FrameStartCycles) * 1000000.0;
    
    const float SimTime = Input->World ? Input->World->GetTimeSeconds() : 0.0f;
    
    // Collect and buffer telemetry sample
    if (Vehicle->bEnableTelemetryLogging)
    {
        FTelemetrySample Sample = CollectTelemetrySample(SimTime, DeltaTime, Rec, InputTensor);
        Vehicle->TelemetrySampleBuffer.Add(Sample);
    }
    
    // Collect and buffer performance sample
    if (Vehicle->bEnablePerformanceLogging)
    {
        FPerformanceSample Sample = CollectPerformanceSample(SimTime, DeltaTime);
        Vehicle->PerformanceSampleBuffer.Add(Sample);
    }
    
    // Collect and buffer aerodynamics sample
    if (Vehicle->bEnableAerodynamicsLogging)
    {
        // Compute min ride height for logging (same as aero computation)
        float MinRideHeight_cm = 50.0f; // [cm] Default fallback
        for (int32 i = 0; i < WheelCount; ++i)
        {
            if (AxleData_PT.bIsInContact[i])
            {
                const float WheelRideHeight_cm = AxleData_PT.SuspensionSpecs[i].RestLength - AxleData_PT.SpringDisplacements[i];
                MinRideHeight_cm = FMath::Min(MinRideHeight_cm, WheelRideHeight_cm);
            }
        }
        
        FAerodynamicsSample AeroSample = CollectAerodynamicsSample(SimTime, Rec, AeroForces, MinRideHeight_cm);
        Vehicle->AerodynamicsSampleBuffer.Add(AeroSample);
    }
#endif // !UE_BUILD_SHIPPING

} // End OnPreSimulate_Internal()

/*====================================================================================================================================================
                                                      SUSPENSION TRAJECTORY ATLAS (RAY-FAN)
====================================================================================================================================================*/
/* Generate multi-ray trace patterns for each wheel - creates fan-shaped ray distribution for accurate ground contact detection */
void AVehicleSolver::GenerateTrajectoryAtlas()
{
    const float AngleStepLocal = AngleStep;                              // [deg] - Angular step between rays
    const int32 DepthCountLocal = DepthCount;                            // [-] - Number of lateral slices

    for (FAxleMember& Ax : Axles_GT)
    {
        const float WheelRadius = Ax.TireSpecifications.OuterRadius * 100.0f; // [cm] - Wheel outer radius
        const float Width = Ax.TireSpecifications.Width * 100.0f;        // [cm] - Tire contact patch width
        const int32 AnglesCount = (static_cast<int32>(MaxAngle / AngleStepLocal) * 2) + 1; // [-] - Total angular samples
        Ax.TracePattern.NumRays = DepthCountLocal * AnglesCount;         // [-] - Total ray count

        Ax.TracePattern.Fractions.Reset();
        Ax.TracePattern.Fractions.Reserve(DepthCountLocal);
        Ax.TracePattern.Directions.Reset();
        Ax.TracePattern.Directions.Reserve(Ax.TracePattern.NumRays);

        // Generate lateral fractions (0.0 = left edge, 1.0 = right edge)
        for (int32 d = 0; d < DepthCountLocal; ++d)
        {
            const float Frac = (DepthCountLocal == 1) ? 0.5f : static_cast<float>(d) / (DepthCountLocal - 1); // [-] - Lateral fraction (0.5 = center for single trace)
            Ax.TracePattern.Fractions.Add(Frac);
        } // End for (lateral fractions)

        // Generate angular directions (fan pattern)
        for (int32 i = -AnglesCount / 2; i <= AnglesCount / 2; ++i)
        {
            const float AngleRad = FMath::DegreesToRadians(i * AngleStepLocal); // [rad] - Ray angle
            Ax.TracePattern.Directions.Emplace(FMath::Sin(AngleRad), 0.0f, -FMath::Cos(AngleRad)); // [-] - Unit direction vector
        } // End for (angles fan)
    } // End for (axles)
} // End GenerateTrajectoryAtlas()

/*====================================================================================================================================================
                                                  SUSPENSION CALIBRATION - UNIFIED IMPLEMENTATION
====================================================================================================================================================*/

/* Calibrate suspension assembly - compute spring rates, damping coefficients, and rest lengths from natural frequency and damping ratio */
void AVehicleSolver::CalibrateSuspensionAssembly()
{
    if (!VehicleHull || Axles_GT.Num() == 0) // Reason: validation check
    {
        return;
    } // End if (validation check)
    
    const FTransform HullXfm = VehicleHull->GetComponentTransform();
    const float TotalMass = VehicleHull->GetMass();                          // [kg]
    const FVector WorldCenterOfMass = VehicleHull->GetCenterOfMass();       // [cm]
    const FVector LocalCenterOfMass = HullXfm.InverseTransformPosition(WorldCenterOfMass); // [cm]
    const FVector VehicleUpDir = HullXfm.GetUnitAxis(EAxis::Z);             // [-]
    const FVector DownDir = -VehicleUpDir;                                   // [-]
    
    SuspensionConfig.LocalCenterOfMass = LocalCenterOfMass;
    SuspensionConfig.WorldCenterOfMass = WorldCenterOfMass;
    SuspensionConfig.WorldSuspensionNormal = VehicleUpDir;
    
    
    // Reason: gather all spring mount positions
    TArray<FVector> LocalSpringOrigins;
    LocalSpringOrigins.Reserve(Axles_GT.Num());
    
    for (int32 WheelIndex = 0; WheelIndex < Axles_GT.Num(); ++WheelIndex)
    {
        FAxleMember& Member = Axles_GT[WheelIndex];
        
        if (!VehicleHull->DoesSocketExist(Member.SuspensionSpecs.SuspensionID) || 
            !VehicleHull->DoesSocketExist(Member.AxleID)) // Reason: socket validation
        {
            // Missing socket
            continue;
        } // End if (socket validation)
        
        const FVector WorldSuspensionLoc = VehicleHull->GetSocketLocation(Member.SuspensionSpecs.SuspensionID); // [cm]
        const FVector LocalSuspensionLoc = HullXfm.InverseTransformPosition(WorldSuspensionLoc); // [cm]
        LocalSpringOrigins.Add(LocalSuspensionLoc);
        Member.SuspensionSpecs.LocalSpringOrigin = LocalSuspensionLoc;
        
        // Wheel spring origin set
    } // End for (wheel iteration)
    
    if (LocalSpringOrigins.Num() == 0) // Reason: critical failure check
    {
        // No valid wheels found
        return;
    } // End if (no valid wheels)
    
    // Reason: compute sprung masses using Lagrange multiplier solver
    TArray<float> SprungMasses;
    const bool bMassSolved = ComputeSuspensionSprungMasses(LocalSpringOrigins, LocalCenterOfMass, TotalMass, SprungMasses);
    
    if (!bMassSolved) // Reason: fallback to equal distribution
    {
        // Using equal mass distribution as fallback
        const float MassPerSpring = TotalMass / static_cast<float>(LocalSpringOrigins.Num()); // [kg]
        SprungMasses.SetNum(LocalSpringOrigins.Num());
        for (int32 i = 0; i < SprungMasses.Num(); ++i) SprungMasses[i] = MassPerSpring;
    } // End if (fallback)
    else
    {
        // Lagrange solver completed successfully
    } // End else (success)
    
    SuspensionConfig.SprungMasses = SprungMasses;
    SuspensionConfig.InitializeDerived();
    
    // Reason: calibrate each wheel using its own frequency/damping from Axles_GT
    int32 LocalIndex = 0;
    for (int32 WheelIndex = 0; WheelIndex < Axles_GT.Num(); ++WheelIndex)
    {
        if (!SprungMasses.IsValidIndex(LocalIndex)) continue; // Reason: bounds check
        
        FAxleMember& Member = Axles_GT[WheelIndex];
        FSuspensionSpecifications& Spec = Member.SuspensionSpecs;
        const float SprungMass = SprungMasses[LocalIndex];                   // [kg]
        
        // Reason: use parameters from Axles_GT (set by user in editor or InitializeAxles)
        const float OmegaN = 2.0f * PI * Spec.NaturalFrequency;              // [rad/s]
        
        // Reason: calculate spring rate and damping
        Spec.SpringRate = SprungMass * OmegaN * OmegaN;                      // [N/m] - k = m * ω²
        const float CriticalDamping = 2.0f * FMath::Sqrt(SprungMass * Spec.SpringRate); // [N·s/m] - c_crit = 2√(mk)
        Spec.DampingRate = Spec.DampingRatio * CriticalDamping;              // [N·s/m] - c = ζ * c_crit
        
        // Reason: calculate RestLength using socket geometry
        const FVector WheelMountLoc = VehicleHull->GetSocketLocation(Member.AxleID); // [cm]
        const FVector SuspensionMountLoc = VehicleHull->GetSocketLocation(Spec.SuspensionID); // [cm]
        const FVector SpringStartPosition = SuspensionMountLoc + DownDir * (-Spec.SuspensionMountLength); // [cm]
        const float SuspendedLength = FVector::DotProduct(WheelMountLoc - SpringStartPosition, DownDir); // [cm]
        
        const float StaticForce = SprungMass * 9.80f;                     // [N] - F_static = m * g
        const float StaticCompression = (Spec.SpringRate > KINDA_SMALL_NUMBER) 
            ? (StaticForce / Spec.SpringRate) 
            : 0.0f;                                                          // [m] - x_static = F/k
        Spec.RestLength = SuspendedLength - (StaticCompression * 100.0f);   // [cm] - L_rest = L_suspended - x_static
        
        if (Spec.RestLength <= 0.0f) // Reason: safety clamp
        {
            // Clamping invalid rest length
            Spec.RestLength = 0.01f;
        } // End if (invalid rest length)
        
        // Wheel parameters calculated
        
        if (Spec.bUseProgressive) // Reason: compute progressive coefficients if enabled
        {
            Spec.ComputeProgressiveCoefficients();                           // âœ… Compute k1/k2/k3
            // Progressive suspension coefficients calculated
        } // End if (progressive enabled)
        else // Reason: zero coefficients when progressive disabled
        {
            Spec.ProgressiveCoeffs.k1 = Spec.SpringRate;
            Spec.ProgressiveCoeffs.k2 = 0.0f;
            Spec.ProgressiveCoeffs.k3 = 0.0f;
            // Using linear suspension mode
        } // End else (linear mode)
        
        ++LocalIndex;
    } // End for (wheel calibration)
    
    // Suspension calibration completed
    
    if (PhysicsCallback && bPhysicsCallbackInitialized)
    {
        PhysicsCallback->AxleData_PT.InitializeFromAxleMembers(Axles_GT);
        const FVector UpDir_PT = HullXfm.GetUnitAxis(EAxis::Z);
        PhysicsCallback->AxleData_PT.RebuildWheelMetricsCache(UpDir_PT, -UpDir_PT);
        PhysicsCallback->SuspensionConfig_PT = SuspensionConfig;
        PhysicsCallback->AxleData_PT.SprungMasses = SuspensionConfig.SprungMasses;
        // Publish static wheel loads (for telemetry baseline and runtime use)
        const int32 N = Axles_GT.Num();
#if !UE_BUILD_SHIPPING
        StaticWheelLoads_GT.SetNum(N);
        if (PhysicsCallback) PhysicsCallback->StaticWheelLoads.SetNum(N);
#endif
        for (int32 i = 0; i < N; ++i)
        {
            const float StaticLoadN = SuspensionConfig.SprungMasses.IsValidIndex(i) ? SuspensionConfig.SprungMasses[i] * 9.80f : (TotalMass * 9.80f) / FMath::Max(1, N);
#if !UE_BUILD_SHIPPING
            StaticWheelLoads_GT[i] = StaticLoadN;
            if (PhysicsCallback) PhysicsCallback->StaticWheelLoads[i] = StaticLoadN;
#endif
        }
        // Published suspension calibration to physics thread
    }
} // End CalibrateSuspensionAssembly

/* Calibrate steering assembly - initialize steering parameters and reset state */
void AVehicleSolver::CalibrateSteeringAssembly()
{
    // Reset steering state to neutral
    SteeringSystem_GT.CurrentSteeringAngle = 0.0f;

    // Initialize chassis configuration from axle geometry if available
    if (Axles_GT.Num() >= 4 && VehicleHull)
    {
        const FTransform HullXfm = VehicleHull->GetComponentTransform();
        
        // Calculate wheelbase (front to rear axle distance)
        const FVector FrontLeft = HullXfm.TransformPosition(Axles_GT[0].SuspensionSpecs.LocalSpringOrigin);
        const FVector RearLeft = HullXfm.TransformPosition(Axles_GT[2].SuspensionSpecs.LocalSpringOrigin);
        const float WheelBaseCm = FVector::Dist(FrontLeft, RearLeft);
        ChassisConfig_GT.WheelBase = WheelBaseCm * 0.01f; // [cm] -> [m]
        
        // Calculate track width (left to right wheel distance)
        const FVector FrontRight = HullXfm.TransformPosition(Axles_GT[1].SuspensionSpecs.LocalSpringOrigin);
        const float TrackWidthCm = FVector::Dist(FrontLeft, FrontRight);
        ChassisConfig_GT.TrackWidth = TrackWidthCm * 0.01f; // [cm] -> [m]
    }
    
    // Steering calibration completed
} // End CalibrateSteeringAssembly

/*====================================================================================================================================================
                                                  EQUILIBRIUM TRANSFORM
====================================================================================================================================================*/

/* Apply equilibrium transform - positions vehicle at correct ride height on ground/slope without dropping from air */
void AVehicleSolver::ApplyEquilibriumTransform()
{
    check(IsInGameThread());
    
    if (!VehicleHull || Axles_GT.Num() < 3 || !GetWorld())
    {
        UE_LOG(LogTemp, Warning, TEXT("❌ Cannot apply equilibrium transform: Prerequisites not met (missing mesh, world, or < 3 axles)."));
        return;
    } // End if (prerequisites check)
    
    UWorld* World = GetWorld();
    const FTransform CurrentTransform = GetActorTransform();
    const FVector VehicleUpDir = CurrentTransform.GetUnitAxis(EAxis::Z);
    
    // Trace down from each wheel to find the actual ground points
    TArray<FVector> GroundContactPoints_World;
    GroundContactPoints_World.Reserve(Axles_GT.Num());
    
    FCollisionQueryParams TraceParams(TEXT("EquilibriumTrace"), true, this);
    FCollisionObjectQueryParams ObjectParams(ECC_WorldStatic);
    ObjectParams.AddObjectTypesToQuery(ECC_WorldDynamic);
    
    int32 SuccessfulTraces = 0;
    const float MaxTraceDistance = 1000.0f;                              // [cm]
    
    for (const FAxleMember& Member : Axles_GT)
    {
        const FVector SuspensionMountWorld = VehicleHull->GetSocketLocation(Member.SuspensionSpecs.SuspensionID);
        const float TotalDropDistance = Member.SuspensionSpecs.RestLength + (Member.TireSpecifications.OuterRadius * 100.0f);
        
        const FVector TraceStart = SuspensionMountWorld + VehicleUpDir * 50.0f;
        const FVector TraceEnd = SuspensionMountWorld - (VehicleUpDir * (TotalDropDistance + MaxTraceDistance));
        
        FHitResult HitResult;
        if (World->LineTraceSingleByObjectType(HitResult, TraceStart, TraceEnd, ObjectParams, TraceParams))
        {
            GroundContactPoints_World.Add(HitResult.ImpactPoint);
            SuccessfulTraces++;
            DrawDebugLine(World, TraceStart, HitResult.ImpactPoint, FColor::Green, false, 15.0f, 0, 2.0f);
            DrawDebugSphere(World, HitResult.ImpactPoint, 10.0f, 12, FColor::Green, false, 15.0f);
        }
        else
        {
            GroundContactPoints_World.Add(SuspensionMountWorld - (VehicleUpDir * TotalDropDistance));
            UE_LOG(LogTemp, Error, TEXT("Wheel Trace FAILED for AxleID: %s. Equilibrium may be inaccurate."), *Member.AxleID.ToString());
            DrawDebugLine(World, TraceStart, TraceEnd, FColor::Red, false, 15.0f, 0, 1.0f);
        } // End if (trace result)
    } // End for (axle iteration)
    
    if (SuccessfulTraces < 3)
    {
        UE_LOG(LogTemp, Error, TEXT("INSUFFICIENT ground traces (%d found, 3 required). Aborting equilibrium."), SuccessfulTraces);
        return;
    } // End if (insufficient traces)
    
    // Calculate ground plane orientation and center
    FVector GroundPlaneCenter_World = FVector::ZeroVector;
    for (const FVector& Point : GroundContactPoints_World)
    {
        GroundPlaneCenter_World += Point;
    } // End for (center calculation)
    GroundPlaneCenter_World /= static_cast<float>(GroundContactPoints_World.Num());
    
    const FVector V1 = GroundContactPoints_World[1] - GroundContactPoints_World[0]; // Vector from FL to FR
    const FVector V2 = GroundContactPoints_World[2] - GroundContactPoints_World[0]; // Vector from FL to RL
    FVector GroundNormal = FVector::CrossProduct(V1, V2).GetSafeNormal();
    
    if (FVector::DotProduct(GroundNormal, FVector::UpVector) < 0.0f)
    {
        GroundNormal *= -1.0f;
    } // End if (flip normal)
    
    // Calculate the correct target position for the actor's origin
    const FVector GroundForward = FVector::VectorPlaneProject(CurrentTransform.GetUnitAxis(EAxis::X), GroundNormal).GetSafeNormal();
    const FVector GroundRight = FVector::CrossProduct(GroundNormal, GroundForward).GetSafeNormal();
    const FQuat EquilibriumRotation = FMatrix(GroundForward, GroundRight, GroundNormal, FVector::ZeroVector).ToQuat();
    
    float AverageHeightAboveGround = 0.0f;
    for (const FAxleMember& Member : Axles_GT)
    {
        AverageHeightAboveGround += Member.SuspensionSpecs.RestLength + (Member.TireSpecifications.OuterRadius * 100.0f);
    } // End for (height calculation)
    AverageHeightAboveGround /= static_cast<float>(Axles_GT.Num());
    
    const FVector TargetAvgMountPoint_World = GroundPlaneCenter_World + GroundNormal * AverageHeightAboveGround;
    
    FVector AvgMountPoint_Local = FVector::ZeroVector;
    for (const FAxleMember& Member : Axles_GT)
    {
        AvgMountPoint_Local += VehicleHull->GetSocketTransform(Member.SuspensionSpecs.SuspensionID, ERelativeTransformSpace::RTS_Component).GetLocation();
    } // End for (local mount calculation)
    AvgMountPoint_Local /= static_cast<float>(Axles_GT.Num());
    
    const FVector RotatedMountOffset = EquilibriumRotation.RotateVector(AvgMountPoint_Local);
    const FVector TargetHullOrigin = TargetAvgMountPoint_World - RotatedMountOffset;
    
    // Apply the final transform and reset physics
    const FTransform EquilibriumTransform(EquilibriumRotation, TargetHullOrigin, CurrentTransform.GetScale3D());
    SetActorTransform(EquilibriumTransform, false, nullptr, ETeleportType::TeleportPhysics);
    
    if (VehicleHull->IsSimulatingPhysics())
    {
        VehicleHull->SetPhysicsLinearVelocity(FVector::ZeroVector);
        VehicleHull->SetPhysicsAngularVelocityInDegrees(FVector::ZeroVector);
    } // End if (reset physics)
    
    // Debug visualization
    DrawDebugCoordinateSystem(World, TargetHullOrigin, EquilibriumRotation.Rotator(), 100.0f, false, 15.0f, 0, 5.0f);
    DrawDebugSphere(World, GroundPlaneCenter_World, 15.f, 12, FColor::Magenta, false, 15.f, 0, 3.f);
    DrawDebugLine(World, GroundPlaneCenter_World, GroundPlaneCenter_World + GroundNormal * 100.f, FColor::Magenta, false, 15.f, 0, 3.f);
    
    UE_LOG(LogTemp, Log, TEXT("✅ Equilibrium transform applied - Vehicle positioned at ride height on ground plane"));
} // End ApplyEquilibriumTransform()

void AVehicleSolver::ApplyPacejkaPlySteerSymmetry()
{
    for (FAxleMember& Member : Axles_GT)
    {
        const int8 Code = Member.WheelCode;
        if (Code == 0)
        {
            continue;
        }

        // Negative = left side (Mirror), Positive = right side (Keep)
        const float SideSign = (Code < 0) ? -1.0f : 1.0f;                // Negative = left side, positive = right side
        FTireSpecSheet& Sheet = Member.TireSpecifications;

        // 1. Pure Slip Symmetry
        //    Corrects static pull when rolling (ply-steer / vertical shift at zero longitudinal slip)
        Sheet.pVy1 = FMath::Abs(Sheet.pVy1) * SideSign; // Lateral shift at zero load
       Sheet.pVy2 = FMath::Abs(Sheet.pVy2) * SideSign; // Load-dependent lateral shift
        Sheet.pVy3 = FMath::Abs(Sheet.pVy3) * SideSign; // Camber-induced shift (offset)
        Sheet.pVy4 = FMath::Abs(Sheet.pVy4) * SideSign; // Camber-induced shift (load)

        // 2. Combined Slip Symmetry
        //    Corrects pull that appears only under acceleration/braking (SVyκ terms)
        //    rVy1-3 control the direction of the combined-slip vertical shift; rVy4-6 mostly shape the curve.
        Sheet.rVy1 = FMath::Abs(Sheet.rVy1) * SideSign; // Vertical shift due to κ (SlipRatio)
        Sheet.rVy2 = FMath::Abs(Sheet.rVy2) * SideSign; // Load-dependent shift due to κ
        Sheet.rVy3 = FMath::Abs(Sheet.rVy3) * SideSign; // Camber-dependent shift due to κ
    }
} // End ApplyPacejkaPlySteerSymmetry()

/** Helper: Get world locations of sockets from a component */
TArray<FVector> AVehicleSolver::GetSocketWorldLocations(USceneComponent* Component, const TArray<FName>& SocketNames) const
{
    TArray<FVector> WorldLocations;
    
    // Reason: validate component
    if (!Component)
    {
        return WorldLocations; // End if (no component)
    }
    
    // Reason: iterate through socket names and get world positions
    for (const FName& SocketName : SocketNames)
    {
        if (SocketName != NAME_None && Component->DoesSocketExist(SocketName))
        {
            WorldLocations.Add(Component->GetSocketLocation(SocketName));
        }
    } // End for (socket iteration)
    
    return WorldLocations;
}


/*====================================================================================================================================================
                                                      SUSPENSION CONTACT TRACING (OPTIMIZED WITH FAST MODE)
====================================================================================================================================================*/

/* 
 * Trace suspension displacements using multi-ray fan pattern:
 * 
 * DepthCount controls lateral sampling:
 * - DepthCount=1: Single centered trace per direction (symmetric for all wheels)
 * - DepthCount>1: Multiple traces spread across tire width
 * 
 * When DepthCount=1, traces are placed at tire center (Frac=0.5) to ensure
 * symmetric behavior between left/right wheels. This prevents asymmetric
 * load transfer that would cause the vehicle to pull to one side.
 * 
 * Debug visualization (non-shipping builds only):
 * - Green lines: Traces that hit ground
 * - Red lines: Traces that missed
 * - Yellow points: Impact locations
 * 
 * OPTIMIZATION NOTES:
 * - Pre-allocated orientation matrices (avoid redundant quaternion construction)
 * - __restrict pointers for compiler optimization
 * - Reusable FHitResult (avoid repeated allocation)
 * - Cache-friendly data access patterns
 */
/*====================================================================================================================================================
                                                      SUSPENSION CONTACT TRACING WITH DEBUG VISUALIZATION
====================================================================================================================================================*/




/** Compute suspension displacements with debug visualization */
void FVehicleSolverCallback::ComputeSuspensionDisplacements(const FVehicleSolverInput* Input, int32 SpringCount, const FTransform& HullXfm, const FVector& UpDir, const FVector& DownDir, FVehicleSolverAxleData_PT& AxleData)
{
    if (!Input || !Input->World || SpringCount == 0 || AxleData.AxleIDs.Num() != SpringCount) return;
    if (AxleData.WheelMetricsCache.Num() != SpringCount) return;

    // Cache query parameters (OQ should be precomputed at setup, stored in Input)
    const FCollisionQueryParams& TraceQP = Input->TraceParams;
    const FCollisionObjectQueryParams& OQ = Input->ObjectQueryParams;
    UWorld* World = Input->World;
    
    // Cache hull orientation vectors
    const FVector HullForward = HullXfm.GetUnitAxis(EAxis::X);
    const FVector HullRight = HullXfm.GetUnitAxis(EAxis::Y);

    // Cache data pointers with __restrict for compiler optimization
    const FInvariantWheelMetrics* __restrict MetricsCache = AxleData.WheelMetricsCache.GetData();
    const FVector* __restrict SpringOrigins = AxleData.WorldSpringOrigins.GetData();
    const float* __restrict SteerAngles = AxleData.SteerAnglesRad.GetData();
    const FTrajectoryAtlas* __restrict TracePatterns = AxleData.TracePatterns.GetData();
    bool* __restrict ContactFlags = AxleData.bIsInContact.GetData();
    int32* __restrict ContactCounts = AxleData.ActiveContactCounts.GetData();
    FVector* __restrict ContactLocations = AxleData.AverageContactLocations.GetData();
    FVector* __restrict ContactNormals = AxleData.AverageContactNormals.GetData();
    float* __restrict SpringDisplacements = AxleData.SpringDisplacements.GetData();

    //====================================================================================
    // Pre-compute steering rotations using cos/sin (no FMatrix, no FQuat)
    //====================================================================================
    FVector WheelForwardDirs[4];
    FVector WheelRightDirs[4];
    
    for (int32 i = 0; i < SpringCount && i < 4; ++i)
    {
        const float SteerAngle = SteerAngles[i];
        if (FMath::Abs(SteerAngle) < UE_KINDA_SMALL_NUMBER)
        {
            WheelForwardDirs[i] = HullForward;
            WheelRightDirs[i] = HullRight;
        }
        else
        {
            const float c = FMath::Cos(SteerAngle);
            const float s = FMath::Sin(SteerAngle);
            WheelForwardDirs[i] = HullForward * c + HullRight * s;
            WheelRightDirs[i] = HullRight * c - HullForward * s;
        }
    }

    //====================================================================================
    // Process each wheel with multi-ray pattern
    //====================================================================================
    for (int32 WheelIndex = 0; WheelIndex < SpringCount; ++WheelIndex)
    {
        // Accumulator variables (register-friendly)
        int32 HitCount = 0;
        FVector SumLoc = FVector::ZeroVector;
        FVector SumNorm = FVector::ZeroVector;

        const FInvariantWheelMetrics& Metrics = MetricsCache[WheelIndex];
        const FVector SuspensionMountLoc = SpringOrigins[WheelIndex];
        const FVector& WheelForward = WheelForwardDirs[WheelIndex];
        const FVector& WheelRight = WheelRightDirs[WheelIndex];
        const FTrajectoryAtlas& Pattern = TracePatterns[WheelIndex];

        // Batch trace loop
        const int32 NumFractions = Pattern.Fractions.Num();
        const int32 NumDirections = Pattern.Directions.Num();
        const float* __restrict Fractions = Pattern.Fractions.GetData();
        const FVector* __restrict Directions = Pattern.Directions.GetData();

        for (int32 FracIdx = 0; FracIdx < NumFractions; ++FracIdx)
        {
            const float Frac = Fractions[FracIdx];
            const FVector LateralOffset = WheelRight * (Frac * Metrics.WidthCm - Metrics.HalfWidthCm);
            const FVector TraceCenter = SuspensionMountLoc + LateralOffset;

            for (int32 DirIdx = 0; DirIdx < NumDirections; ++DirIdx)
            {
                // Manual vector transform (no FMatrix multiply)
                const FVector& Dir = Directions[DirIdx];
                const FVector WorldDir = Dir.X * WheelForward + Dir.Y * WheelRight + Dir.Z * UpDir;
                
                // Compute trace endpoints (no intermediate BasePoint)
                const FVector Offset = WorldDir * Metrics.RadiusCm;
                const FVector Start = TraceCenter + Offset + Metrics.UpRaiseOffset;
                const FVector End = TraceCenter + Offset + Metrics.DownDropOffset;

                // Fresh local FHitResult (cheaper than Hit.Reset())
                FHitResult Hit;
                const bool bHit = World->LineTraceSingleByObjectType(Hit, Start, End, OQ, TraceQP);
                
#if !UE_BUILD_SHIPPING
                // Debug draw for single-depth traces (depth=1) to verify symmetry
                if (NumFractions == 1)
                {
                    const FColor TraceColor = bHit ? FColor::Green : FColor::Red;
                    DrawDebugLine(World, Start, bHit ? Hit.ImpactPoint : End, TraceColor, false, 0.0f, 0, 1.0f);
                    if (bHit)
                    {
                        DrawDebugPoint(World, Hit.ImpactPoint, 5.0f, FColor::Yellow, false, 0.0f);
                    }
                }
#endif
                
                if (bHit)
                {
                    SumLoc += Hit.ImpactPoint;
                    SumNorm += Hit.ImpactNormal;
                    ++HitCount;
                }
            }
        }

        // Store results
        const bool bWheelInContact = HitCount > 0;
        ContactFlags[WheelIndex] = bWheelInContact;
        ContactCounts[WheelIndex] = HitCount;

        if (bWheelInContact)
        {
            const float InvCnt = 1.0f / static_cast<float>(HitCount);
            const FVector AvgLoc = SumLoc * InvCnt;
            const FVector AvgNorm = (SumNorm * InvCnt).GetSafeNormal();
            ContactLocations[WheelIndex] = AvgLoc;
            ContactNormals[WheelIndex] = AvgNorm;

            const FVector PhysicsWheelCenter = AvgLoc + AvgNorm * Metrics.RadiusCm;
            const float CurrentDistance = FVector::DotProduct(PhysicsWheelCenter - SuspensionMountLoc, DownDir);
            SpringDisplacements[WheelIndex] = Metrics.RestLengthCm - CurrentDistance;
        }
        else
        {
            SpringDisplacements[WheelIndex] = -Metrics.MaxDropCm;
            ContactLocations[WheelIndex] = FVector::ZeroVector;
            ContactNormals[WheelIndex] = UpDir;
        }
    }
}



/*====================================================================================================================================================
                                                        SPRING/DAMPER FORCE CALCULATION
====================================================================================================================================================*/

/* Compute spring and damper forces for all wheels - applies progressive spring rates and velocity-dependent damping */
void FVehicleSolverCallback::ComputeSuspensionForces(const FInstantaneousVehicleRecord& Rec, const FVector& LinearVel, const FVector& AngularVelRad, int32 SpringCount, FVehicleSolverAxleData_PT& AxleData)
{
    if (SpringCount == 0 || !Proxy || AxleData.WorldSpringOrigins.Num() != SpringCount) return;

    Chaos::FRigidBodyHandle_Internal* RigidBody = Proxy->GetPhysicsThreadAPI();
    if (!RigidBody) return;
    
    const FVehicleSuspensionConfig& Config = SuspensionConfig_PT;
    const FTransform HullXfm = Rec.HullXfm; 
    const FVector WorldCoM = HullXfm.TransformPosition(Config.LocalCenterOfMass); // [cm] - World-space center of mass
    const FVector WorldSuspensionNormal = Rec.ê_vertical;                // [-] - Chassis up direction

    // Cache pointers for performance
    const bool* __restrict pIsInContact = AxleData.bIsInContact.GetData();
    const float* __restrict pSpringDisplacements = AxleData.SpringDisplacements.GetData();
    const FVector* __restrict pWorldSpringOrigins = AxleData.WorldSpringOrigins.GetData();
    const FSuspensionSpecifications* __restrict pSuspensionSpecs = AxleData.SuspensionSpecs.GetData();
    float* __restrict pSpringVelocities = AxleData.SpringVelocities.GetData();
    float* __restrict pSpringForces = AxleData.SpringForces.GetData();

    FVector TotalForce = FVector::ZeroVector;                            // [UU⋅kg⋅s⁻²] - Accumulated force (Unreal units)
    FVector TotalTorque = FVector::ZeroVector;                           // [UU²⋅kg⋅s⁻²] - Accumulated torque (Unreal units)

    for (int32 i = 0; i < SpringCount; ++i)
    {
        // Skip wheels not in contact with ground
        if (!pIsInContact[i]) { pSpringForces[i] = 0.0f; continue; }

        const float SpringDisplacementCm = pSpringDisplacements[i];      // [cm] - Current compression/extension
        const FVector WorldOrigin = pWorldSpringOrigins[i];              // [cm] - Spring mount world position
        const FSuspensionSpecifications& SuspSpec = pSuspensionSpecs[i];

        // Calculate the velocity of the spring mount point
        const FVector COMOrigin = WorldOrigin - WorldCoM;                // [cm] - Lever arm from CoM to spring
        const FVector SupportLinearVelocity = LinearVel + FVector::CrossProduct(AngularVelRad, COMOrigin); // [cm⋅s⁻¹] - Mount point velocity
        const FVector GroundNormal = (AxleData.AverageContactNormals.IsValidIndex(i) && !AxleData.AverageContactNormals[i].IsNearlyZero())
            ? AxleData.AverageContactNormals[i].GetSafeNormal()
            : WorldSuspensionNormal;                                     // [-] - Ground surface normal
        const float SpringVelocityCms = -FVector::DotProduct(GroundNormal, SupportLinearVelocity); // [cm⋅s⁻¹] - Compression velocity
        pSpringVelocities[i] = SpringVelocityCms;

        // Compute spring and damper forces
        float CompressionCm = FMath::Max(0.0f, SpringDisplacementCm);
        if (SuspSpec.bUseProgressive && SuspSpec.CurveType == ESuspensionCurveType::Digressive)
        {
            const float FullTravelCm = SuspSpec.MinRaise + SuspSpec.MaxDrop; // [cm]
            if (FullTravelCm > KINDA_SMALL_NUMBER && SuspSpec.HardeningFactor > 1.0f)
            {
                const float H = SuspSpec.HardeningFactor;
                const float PeakCompressionCm = (H / (2.0f * (H - 1.0f))) * FullTravelCm; // [cm]
                const float MaxEffCompressionCm = (SuspSpec.MaxTravel > 0.0f)
                    ? FMath::Min(PeakCompressionCm, SuspSpec.MaxTravel * 100.0f)
                    : PeakCompressionCm;
                CompressionCm = FMath::Clamp(CompressionCm, 0.0f, MaxEffCompressionCm);
            }
        }

        const float StiffnessForceN = SuspSpec.GetSpringForce_N(CompressionCm); // [N] - Spring force (progressive or linear)
        const float DampingForceN = (SpringVelocityCms * 0.01f) * SuspSpec.DampingRate; // [N] - Damper force

        const float CombinedForceN = StiffnessForceN + DampingForceN;    // [N] - Total suspension force
        const float FinalForceN = FMath::Max(0.0f, CombinedForceN);      // [N] - Clamp to prevent negative (pulling) forces
        pSpringForces[i] = FinalForceN;

        // Apply force and torque to rigid body
        const FVector SupportForceN = GroundNormal * FinalForceN;        // [N] - Force vector in world space
        const FVector SupportForceUU = SupportForceN * 100.0f;           // [UU⋅kg⋅s⁻²] - Convert to Unreal units
        const FVector SupportTorqueUU = FVector::CrossProduct(COMOrigin, SupportForceUU); // [UU²⋅kg⋅s⁻²] - Torque about CoM
        TotalForce += SupportForceUU;
        TotalTorque += SupportTorqueUU;
    }

    // Apply accumulated forces and torques to rigid body
    if (TotalForce.SizeSquared() > UE_SMALL_NUMBER) RigidBody->AddForce(TotalForce, false);
    if (TotalTorque.SizeSquared() > UE_SMALL_NUMBER) RigidBody->AddTorque(TotalTorque, false);
} // End ComputeSuspensionForces()

/*====================================================================================================================================================
                                                        DYNAMIC LOAD TRANSFER (REALTIME)
====================================================================================================================================================*/

void FVehicleSolverCallback::ComputeLoadTransferRealtime(const FInstantaneousVehicleRecord& ChassisRec, float DeltaTime, int32 NumWheels, FVehicleSolverAxleData_PT& AxleData, const FAerodynamicForces_PT& AeroForces)
{
    // Fallback for non-4-wheel vehicles (unchanged from original)
    if (NumWheels != 4 || AxleData.AxleIDs.Num() != NumWheels)
    {
        int32 ContactCount = 0; 
        for (int32 i = 0; i < AxleData.bIsInContact.Num(); ++i) 
            if (AxleData.bIsInContact[i]) ++ContactCount;
        const float EqualLoad = (ChassisRec.μ_mass * 9.80f) / FMath::Max(1, ContactCount);
        for (int32 i = 0; i < NumWheels; ++i) 
            AxleData.WheelLoads[i] = AxleData.bIsInContact[i] ? EqualLoad : 0.0f;
        return;
    }

    /*====================================================================================================================================================
                                                         STEP 1: EXTRACT GEOMETRY
    ====================================================================================================================================================*/
    
    // Chassis configuration (fallback to LoadXfer_PT if ChassisConfig unavailable)
    const bool bHasGeom = SteeringGeometryCache_PT.bInitialized;
    const float L_m = bHasGeom ? (SteeringGeometryCache_PT.WheelBaseCm * 0.01f) : FMath::Max(LoadXfer_PT.Wheelbase_m, 0.1f); // [m]
    const float t_front_m = bHasGeom ? ((SteeringGeometryCache_PT.HalfTrackCm * 2.0f) * 0.01f) : FMath::Max(LoadXfer_PT.TrackFront_m, 0.8f); // [m]
    const float t_rear_m  = bHasGeom ? ((SteeringGeometryCache_PT.HalfTrackCm * 2.0f) * 0.01f) : FMath::Max(LoadXfer_PT.TrackRear_m, 0.8f); // [m]
    const float h_CG_m = FMath::Max(LoadXfer_PT.CGHeight_m, 0.05f); // [m]
    
    // Front/rear CG distribution (from mass or LoadXfer if available)
    float Lf_m = LoadXfer_PT.Lf_m;
    float Lr_m = LoadXfer_PT.Lr_m;
    if (Lf_m <= 0.0f || Lr_m <= 0.0f || (Lf_m + Lr_m) > (L_m * 1.1f))
    {
        Lf_m = 0.5f * L_m;
        Lr_m = 0.5f * L_m;
    }

    /*====================================================================================================================================================
                                                         STEP 2: COMPUTE STATIC LOADS FROM SPRUNG MASSES
    ====================================================================================================================================================*/
    
    // Extract static loads from suspension specs (ground truth from calibration)
    float m_front_total = 0.0f; // [kg]
    float m_rear_total = 0.0f;  // [kg]
    
    for (int32 i = 0; i < 4 && i < AxleData.SprungMasses.Num(); ++i)
    {
        const float LoadN = AxleData.SprungMasses.IsValidIndex(i) ? AxleData.SprungMasses[i] * 9.80f : 0.0f; // [N]
        AxleData.StaticLoads_LT[i] = LoadN;
        if (i < 2) m_front_total += AxleData.SprungMasses[i]; // Front axle
        else m_rear_total += AxleData.SprungMasses[i];         // Rear axle
    }
    
    const float m_total = FMath::Max(ChassisRec.μ_mass, 1.0f); // [kg]
    const float m_sprung = m_front_total + m_rear_total;       // [kg] Total sprung mass

    /*====================================================================================================================================================
                                                         STEP 3: ROLL STIFFNESS CALCULATION
                                                         
        Roll stiffness K_φ [N·m/rad] determines HOW MUCH the elastic load transfer favors each axle.
        
        Formula (Milliken & Milliken, 2020 ed., Eq. 6.23):
        K_φ = K_spring × t²/2 + K_ARB
        
        Where:
        - K_spring: Wheel rate [N/m] at the wheel (not ride rate!)
        - t: Track width [m]
        - K_ARB: Anti-roll bar stiffness [N·m/rad]
    ====================================================================================================================================================*/
    
    // Front axle roll stiffness
    float K_spring_front = 0.0f; // [N/m] Combined front wheel rate
    if (AxleData.SuspensionSpecs.IsValidIndex(0) && AxleData.SuspensionSpecs.IsValidIndex(1))
    {
        K_spring_front = AxleData.SuspensionSpecs[0].SpringRate + AxleData.SuspensionSpecs[1].SpringRate; // [N/m]
    }
    const float K_phi_springs_front = K_spring_front * (t_front_m * t_front_m) * 0.5f; // [N·m/rad]
    
    // Front ARB contribution
    float K_phi_ARB_front = 0.0f; // [N·m/rad]
    for (const FAntiRollbar& Bar : AntiRollbars_PT)
    {
        if (!Bar.bEnabled) continue;
        
        // Check if this is a front axle bar (wheels 0,1 = FL,FR)
        const bool bIsFrontBar = (Bar.WheelIndices[0] < 2 && Bar.WheelIndices[1] < 2);
        if (bIsFrontBar)
        {
            K_phi_ARB_front += Bar.Stiffness; // [N·m/rad] - ARB stiffness is already in roll moment units
        }
    }
    
    const float K_phi_front = K_phi_springs_front + K_phi_ARB_front; // [N·m/rad] Total front roll stiffness
    
    // Rear axle roll stiffness
    float K_spring_rear = 0.0f; // [N/m]
    if (AxleData.SuspensionSpecs.IsValidIndex(2) && AxleData.SuspensionSpecs.IsValidIndex(3))
    {
        K_spring_rear = AxleData.SuspensionSpecs[2].SpringRate + AxleData.SuspensionSpecs[3].SpringRate; // [N/m]
    }
    const float K_phi_springs_rear = K_spring_rear * (t_rear_m * t_rear_m) * 0.5f; // [N·m/rad]
    
    // Rear ARB contribution
    float K_phi_ARB_rear = 0.0f; // [N·m/rad]
    for (const FAntiRollbar& Bar : AntiRollbars_PT)
    {
        if (!Bar.bEnabled) continue;
        
        // Check if this is a rear axle bar (wheels 2,3 = RL,RR)
        const bool bIsRearBar = (Bar.WheelIndices[0] >= 2 && Bar.WheelIndices[1] >= 2);
        if (bIsRearBar)
        {
            K_phi_ARB_rear += Bar.Stiffness; // [N·m/rad]
        }
    }
    
    const float K_phi_rear = K_phi_springs_rear + K_phi_ARB_rear; // [N·m/rad] Total rear roll stiffness
    const float K_phi_total = K_phi_front + K_phi_rear;           // [N·m/rad] Total vehicle roll stiffness

    /*====================================================================================================================================================
                                                         STEP 4: ROLL CENTER HEIGHTS
                                                         
        Roll center: The instantaneous center about which the sprung mass rolls.
        
        Typical values (Guiggiani 2023, Table 7.2):
        - Front: 50-150mm (double wishbone), 0-50mm (MacPherson strut)
        - Rear: 100-200mm (double wishbone), 50-150mm (multi-link)
        
        Load transfer splits into:
        1. Geometric: Through suspension links (ground → RC)
        2. Elastic: Through springs/ARBs (RC → CG)
    ====================================================================================================================================================*/
    
    // Roll center heights [m] (conservative estimates - user should calibrate these)
    const float h_RC_front_m = 0.05f;  // [m] 50mm front RC height (typical for street car)
    const float h_RC_rear_m = 0.10f;   // [m] 100mm rear RC height (typical for street car)

    /*====================================================================================================================================================
                                                         STEP 5: BODY-AXIS ACCELERATIONS
    ====================================================================================================================================================*/
    
    const FTransform& HullXfm = ChassisRec.HullXfm;
    const float a_x = FVector::DotProduct(ChassisRec.α_linear, HullXfm.GetUnitAxis(EAxis::X)); // [m/s²] Longitudinal
    const float a_y = FVector::DotProduct(ChassisRec.α_linear, HullXfm.GetUnitAxis(EAxis::Y)); // [m/s²] Lateral

    /*====================================================================================================================================================
                                                         STEP 6: LATERAL LOAD TRANSFER (CORNERING)
                                                         
        TWO-PATH DECOMPOSITION (Rill & Castro 2020, Section 5.3):
        
        1. Geometric (Instant):
           ΔFz_geom_front = (m_front × a_y × h_RC_front) / t_front
           ΔFz_geom_rear  = (m_rear × a_y × h_RC_rear) / t_rear
        
        2. Elastic (Stiffness-based):
           ΔFz_elastic_total = (m_sprung × a_y × (h_CG - h_RC_avg)) / t_avg
           
           Split by roll stiffness ratio:
           ΔFz_elastic_front = ΔFz_elastic_total × (K_φ_front / K_φ_total)
           ΔFz_elastic_rear  = ΔFz_elastic_total × (K_φ_rear / K_φ_total)
    ====================================================================================================================================================*/
    
    // 6A. Geometric load transfer (via roll centers)
    const float dFz_geom_front_total = (m_front_total * a_y * h_RC_front_m) / FMath::Max(t_front_m, 0.1f); // [N]
    const float dFz_geom_rear_total  = (m_rear_total * a_y * h_RC_rear_m) / FMath::Max(t_rear_m, 0.1f);    // [N]
    
    // 6B. Elastic load transfer (via springs/ARBs)
    // Average roll center and track for elastic component
    const float h_RC_avg_m = (h_RC_front_m * m_front_total + h_RC_rear_m * m_rear_total) / FMath::Max(m_sprung, 1.0f); // [m]
    const float t_avg_m = (t_front_m + t_rear_m) * 0.5f; // [m]
    
    const float dFz_elastic_total = (m_sprung * a_y * (h_CG_m - h_RC_avg_m)) / FMath::Max(t_avg_m, 0.1f); // [N]
    
    // 6C. Distribute elastic load transfer by roll stiffness ratio
    float dFz_elastic_front = 0.0f; // [N]
    float dFz_elastic_rear = 0.0f;  // [N]
    
    if (K_phi_total > 1.0f) // Avoid division by zero
    {
        dFz_elastic_front = dFz_elastic_total * (K_phi_front / K_phi_total); // [N]
        dFz_elastic_rear  = dFz_elastic_total * (K_phi_rear / K_phi_total);  // [N]
    }
    else
    {
        // Fallback: equal split if no suspension stiffness data
        dFz_elastic_front = dFz_elastic_total * 0.5f; // [N]
        dFz_elastic_rear  = dFz_elastic_total * 0.5f; // [N]
    }
    
    // 6D. Total lateral load transfer per axle
    const float dFz_lat_front = dFz_geom_front_total + dFz_elastic_front; // [N]
    const float dFz_lat_rear  = dFz_geom_rear_total + dFz_elastic_rear;   // [N]

    /*====================================================================================================================================================
                                                         STEP 7: LONGITUDINAL LOAD TRANSFER (BRAKING/ACCELERATION)
                                                         
        Formula (Milliken & Milliken 2020, Eq. 6.15):
        ΔFz_long = (m_total × a_x × h_CG) / L
        
        Split by weight distribution:
        ΔFz_front = ΔFz_long × (Lr / L)  [gains load under braking]
        ΔFz_rear  = ΔFz_long × (Lf / L)  [gains load under acceleration]
    ====================================================================================================================================================*/
    
    const float dFz_long_total = (m_total * a_x * h_CG_m) / FMath::Max(L_m, 0.1f); // [N]
    const float dFz_long_front = dFz_long_total * (Lr_m / FMath::Max(L_m, 0.1f));  // [N]
    const float dFz_long_rear  = dFz_long_total * (Lf_m / FMath::Max(L_m, 0.1f));  // [N]

    /*====================================================================================================================================================
                                                         STEP 8: AERODYNAMIC DOWNFORCE DISTRIBUTION
                                                         
        NOTE: Aero forces are now precomputed in ComputeAerodynamicForces() and passed in.
        This eliminates double-counting and ensures consistent aero values across subsystems.
    ====================================================================================================================================================*/
    
    // Use precomputed aerodynamic forces (computed ONCE in OnPreSimulate_Internal)
    const float F_aero_front = AeroForces.FrontDownforce_N;  // [N] - Already computed
    const float F_aero_rear  = AeroForces.RearDownforce_N;   // [N] - Already computed

    /*====================================================================================================================================================
                                                         STEP 9: COMPOSE PER-CORNER LOADS
                                                         
        Index mapping: 0=FL, 1=FR, 2=RL, 3=RR
        
        Sign conventions:
        - Lateral (a_y > 0 = right turn):
          - Left wheels (FL, RL) lose load (negative sign)
          - Right wheels (FR, RR) gain load (positive sign)
        
        - Longitudinal (a_x > 0 = acceleration):
          - Front wheels lose load (positive dFz_long_front means rear bias)
          - Rear wheels gain load
    ====================================================================================================================================================*/
    
    // Lateral load transfer sign: positive a_y (right turn) transfers load TO right wheels
    const float signLat = FMath::Sign(a_y); // [-]
    
    // Front left (index 0)
    AxleData.ComputedLoads_LT[0] = AxleData.StaticLoads_LT[0]                   // Static load
                                 - signLat * (dFz_lat_front * 0.5f)            // Lateral transfer (loses load in right turn)
                                 - (dFz_long_front * 0.5f)                     // Longitudinal transfer (loses load under braking)
                                 + (F_aero_front * 0.5f);                      // Aero downforce
    
    // Front right (index 1)
    AxleData.ComputedLoads_LT[1] = AxleData.StaticLoads_LT[1]
                                 + signLat * (dFz_lat_front * 0.5f)            // Gains load in right turn
                                 - (dFz_long_front * 0.5f)
                                 + (F_aero_front * 0.5f);
    
    // Rear left (index 2)
    AxleData.ComputedLoads_LT[2] = AxleData.StaticLoads_LT[2]
                                 - signLat * (dFz_lat_rear * 0.5f)
                                 + (dFz_long_rear * 0.5f)                      // Gains load under acceleration
                                 + (F_aero_rear * 0.5f);
    
    // Rear right (index 3)
    AxleData.ComputedLoads_LT[3] = AxleData.StaticLoads_LT[3]
                                 + signLat * (dFz_lat_rear * 0.5f)
                                 + (dFz_long_rear * 0.5f)
                                 + (F_aero_rear * 0.5f);

    /*====================================================================================================================================================
                                                         STEP 10: CONTACT GATING & CLAMPING
    ====================================================================================================================================================*/
    
    for (int32 i = 0; i < 4; ++i)
    {
        AxleData.ComputedLoads_LT[i] = AxleData.bIsInContact[i] ? FMath::Max(0.0f, AxleData.ComputedLoads_LT[i]) : 0.0f; // [N]
    }

    /*====================================================================================================================================================
                                                         STEP 11: COMMIT TO AXLE DATA
                                                         
        NOTE: For testing, this is currently commented out. Uncomment to activate.
        The computed loads are stored in the static array `ComputedLoads[4]` for inspection.
    ====================================================================================================================================================*/
    
    // TESTING: Uncomment below to activate the new load transfer model
    for (int32 i = 0; i < 4; ++i)
    {
        AxleData.WheelLoads[i] = AxleData.ComputedLoads_LT[i];
    }
    
    // DEBUG: Log computed loads for verification
    #if !UE_BUILD_SHIPPING
    if (GFrameCounter % 60 == 0) // Log once per second at 60fps
    {
        UE_LOG(LogTemp, Log, TEXT("🔬 Advanced Load Transfer | K_φ_front=%.1f N·m/rad (Springs: %.1f, ARB: %.1f) | K_φ_rear=%.1f N·m/rad (Springs: %.1f, ARB: %.1f) | Ratio: %.3f"),
            K_phi_front, K_phi_springs_front, K_phi_ARB_front,
            K_phi_rear, K_phi_springs_rear, K_phi_ARB_rear,
            K_phi_total > 1.0f ? (K_phi_front / K_phi_total) : 0.5f);
        
        UE_LOG(LogTemp, Log, TEXT("   Loads [N]: FL=%.1f FR=%.1f RL=%.1f RR=%.1f | Lat: F=%.1f R=%.1f | Long: F=%.1f R=%.1f | Aero: F=%.1f R=%.1f"),
            AxleData.ComputedLoads_LT[0], AxleData.ComputedLoads_LT[1], AxleData.ComputedLoads_LT[2], AxleData.ComputedLoads_LT[3],
            dFz_lat_front, dFz_lat_rear,
            dFz_long_front, dFz_long_rear,
            F_aero_front, F_aero_rear);
    }
    #endif
} // End ComputeLoadTransferRealtime()

/** Initialize aerodynamics package with Nissan GT-R Nismo GT3 specifications */
void AVehicleSolver::InitializeAerodynamicsPackage()
{
    if (!VehicleHull) { return; } // Reason: No hull mesh available
    
    // Air density at sea level, 15°C
    AeroPackage_GT.AirDensity_kgm3 = 1.225f;             // [kg/m³]
    
    //--------------------------------------------------------------------------
    //                          REAR WING (GT3 Spec)
    //--------------------------------------------------------------------------
    AeroPackage_GT.RearWing.Area_m2 = 1.4f;              // [m²] - Large GT3 rear wing
    AeroPackage_GT.RearWing.BaseCoeffLift = -3.2f;       // [-] - High downforce GT3 profile
    AeroPackage_GT.RearWing.BaseCoeffDrag = 0.52f;       // [-] - GT3 drag penalty
    AeroPackage_GT.RearWing.AspectRatio = 3.5f;          // [-] - GT3 wing span/chord
    AeroPackage_GT.RearWing.MaxAngle_deg = 22.0f;        // [deg] - Maximum attack angle
    AeroPackage_GT.RearWing.MinAngle_deg = 4.0f;         // [deg] - Minimum attack angle
    AeroPackage_GT.RearWing.CurrentAngle_deg = 14.0f;    // [deg] - Mid-range setting
    AeroPackage_GT.RearWing.bIsAdaptive = true;          // [-] - DRS-style system
    AeroPackage_GT.RearWing.AdaptiveSpeedThreshold_ms = 35.0f; // [m/s] - 126 km/h activation
    AeroPackage_GT.RearWing.BrakeDeployAngle_deg = 10.0f; // [deg] - Air brake deploy
    AeroPackage_GT.RearWing.SocketNames.Add(FName("RearWingAssemblyPort"));
    
    //--------------------------------------------------------------------------
    //                          FRONT SPLITTER (GT3 Spec)
    //--------------------------------------------------------------------------
    AeroPackage_GT.FrontSplitter.Area_m2 = 0.75f;        // [m²] - GT3 splitter area
    AeroPackage_GT.FrontSplitter.CoeffPressure = 0.9f;   // [-] - High pressure buildup
    AeroPackage_GT.FrontSplitter.AirDamHeight_m = 0.08f; // [m] - Low GT3 ride height
    AeroPackage_GT.FrontSplitter.AirDamDragCoeff = 0.28f; // [-] - Splitter drag
    AeroPackage_GT.FrontSplitter.RideHeightSensitivity = 3.2f; // [-] - Very sensitive
    AeroPackage_GT.FrontSplitter.SocketNames.Add(FName("Splitter"));
    
    //--------------------------------------------------------------------------
    //                          CANARDS (4x GT3 Configuration)
    //--------------------------------------------------------------------------
    AeroPackage_GT.Canards.SetNum(4);
    
    // Front Right Canard
    AeroPackage_GT.Canards[0].Area_m2 = 0.32f;           // [m²] - Individual canard area
    AeroPackage_GT.Canards[0].BaseCoeffLift = -2.1f;     // [-] - Moderate downforce
    AeroPackage_GT.Canards[0].BaseCoeffDrag = 0.38f;     // [-] - Drag penalty

/*====================================================================================================================================
                                    AERODYNAMIC FORCE COMPUTATION - ACADEMIC FORMULATION
                                    
References:
[1] Zhang et al. (2006) - Ground Effect Aerodynamics of Race Cars, Applied Mechanics Reviews
[2] Cooper et al. (1998) - The Aerodynamic Performance of Automotive Underbody Diffusers, SAE 980030
[3] NASA - Induced Drag Coefficient Theory
[4] Prandtl's Lifting Line Theory
[5] Toet, W. - Race Car Diffuser Aerodynamics
======================================================================================================================================*/

/** Compute aerodynamic forces using validated academic formulations */
FAerodynamicForces_PT FVehicleSolverCallback::ComputeAerodynamicForces(const FAerodynamicPackage_PT& Aero, float VehicleSpeed_ms, float RideHeight_m, const FVector& VelocityDir_World, float BrakeInput, float ThrottleInput, float YawRate_rads)
{
    FAerodynamicForces_PT Forces;
    FMemory::Memzero(&Forces, sizeof(FAerodynamicForces_PT));
    
    if (VehicleSpeed_ms < 0.5f) { return Forces; } // Reason: Aerodynamic forces negligible below 0.5 m/s
    
    const float Rho = Aero.AirDensity_kgm3;          // [kg/m³] - Air density
    const float V = VehicleSpeed_ms;                 // [m/s] - Vehicle speed
    const float V2 = V * V;                          // [m²/s²]
    const float q = 0.5f * Rho * V2;                 // [Pa] (kg⋅m⁻¹⋅s⁻²) - Dynamic pressure
    
    //--------------------------------------------------------------------------
    //              REAR WING - Lifting Line Theory (Prandtl)
    //--------------------------------------------------------------------------
    if (Aero.RearWing.bEnabled)
    {
        const FWing_PT& Wing = Aero.RearWing;
        float EffectiveAngle = Wing.CurrentAngle_deg; // [deg]
        
        if (Wing.bIsAdaptive)
        {
            if (V > Wing.AdaptiveSpeedThreshold_ms) { EffectiveAngle = FMath::Lerp(Wing.CurrentAngle_deg, Wing.MinAngle_deg, FMath::Clamp((V - Wing.AdaptiveSpeedThreshold_ms) / 20.0f, 0.0f, 1.0f) * 0.5f); } // Reason: DRS-style drag reduction
            if (BrakeInput > 0.1f) { EffectiveAngle = FMath::Clamp(EffectiveAngle + Wing.BrakeDeployAngle_deg * BrakeInput, Wing.MinAngle_deg, Wing.MaxAngle_deg); } // Reason: Air brake deployment
        } // End if (Adaptive wing)
        
        // Lift coefficient from thin airfoil theory: CL = CL_α * α
        const float CL_alpha = 0.11f;                // [deg⁻¹] - Lift curve slope for cambered airfoil
        const float CL = Wing.BaseCoeffLift + CL_alpha * EffectiveAngle; // [-]
        Forces.WingDownforce_N = -q * Wing.Area_m2 * CL; // [N] - Negative lift = downforce
        
        // Induced drag from lifting line theory: CD_i = CL² / (π * AR * e)
        // Reference: NASA Glenn Research Center, Induced Drag Coefficient
        const float e_oswald = 0.82f;                // [-] - Oswald efficiency (typical for wings: 0.7-0.85)
        const float CD_induced = (CL * CL) / (UE_PI * Wing.AspectRatio * e_oswald); // [-]
        const float CD_total = Wing.BaseCoeffDrag + CD_induced; // [-]
        Forces.WingDrag_N = q * Wing.Area_m2 * CD_total; // [N] (kg⋅m⋅s⁻²)
    } // End REAR WING
    
    //--------------------------------------------------------------------------
    //              CANARDS - Lifting Line Theory
    //--------------------------------------------------------------------------
    for (int32 i = 0; i < Aero.Canards.Num(); ++i)
    {
        if (!Aero.Canards[i].bEnabled) { continue; } // Reason: Canard not configured
        
        const FCanard_PT& Canard = Aero.Canards[i];
        float EffectiveAngle = Canard.CurrentAngle_deg; // [deg]
        
        if (Canard.bIsAdaptive)
        {
            if (V > Canard.AdaptiveSpeedThreshold_ms) { EffectiveAngle = FMath::Lerp(Canard.CurrentAngle_deg, Canard.MinAngle_deg, FMath::Clamp((V - Canard.AdaptiveSpeedThreshold_ms) / 30.0f, 0.0f, 1.0f) * 0.6f); }
            if (BrakeInput > 0.1f) { EffectiveAngle = FMath::Clamp(EffectiveAngle + Canard.BrakeDeployAngle_deg * BrakeInput, Canard.MinAngle_deg, Canard.MaxAngle_deg); }
            if (BrakeInput < 0.1f && ThrottleInput > 0.5f) { EffectiveAngle = FMath::Clamp(EffectiveAngle - Canard.ThrottleReductionAngle_deg * (ThrottleInput - 0.5f) / 0.5f, Canard.MinAngle_deg, Canard.MaxAngle_deg); }
        } // End if (Adaptive canard)
        
        const float DamageFactor = 1.0f - Canard.DamageLevel; // [-]
        const float CL_alpha = 0.10f;                // [deg⁻¹]
        const float CL = (Canard.BaseCoeffLift + CL_alpha * EffectiveAngle) * DamageFactor; // [-]
        
        const float e_oswald = 0.78f;                // [-] - Lower efficiency than wing
        const float CD_induced = (CL * CL) / (UE_PI * Canard.AspectRatio * e_oswald); // [-]
        const float CD_total = (Canard.BaseCoeffDrag + CD_induced) * DamageFactor; // [-]
        
        Forces.TotalCanardDownforce_N += -q * Canard.Area_m2 * CL; // [N]
        Forces.TotalCanardDrag_N += q * Canard.Area_m2 * CD_total; // [N]
    } // End for (Canards)
    
    //--------------------------------------------------------------------------
    //              FRONT SPLITTER - Pressure Coefficient Method
    //--------------------------------------------------------------------------
    if (Aero.FrontSplitter.bEnabled)
    {
        const FSplitter_PT& Splitter = Aero.FrontSplitter;
        const float h = FMath::Max(RideHeight_m, 0.005f); // [m]
        
        // Ground proximity amplification based on blockage ratio
        // Reference: Zhang et al. (2006) - Ground effect increases as h decreases
        const float h_ref = 0.10f;                   // [m] - Reference ride height
        const float BlockageRatio = h / h_ref;       // [-]
        const float GroundProximityFactor = 1.0f / FMath::Pow(BlockageRatio, Splitter.RideHeightSensitivity * 0.1f); // [-]
        const float GroundProximityFactorClamped = FMath::Clamp(GroundProximityFactor, 0.5f, 3.5f); // [-]
        
        Forces.SplitterDownforce_N = q * Splitter.Area_m2 * Splitter.CoeffPressure * GroundProximityFactorClamped; // [N]
        Forces.SplitterDrag_N = q * (Splitter.AirDamHeight_m * 1.8f) * Splitter.AirDamDragCoeff; // [N]
    } // End FRONT SPLITTER
    
    //--------------------------------------------------------------------------
    //              UNDERBODY / DIFFUSER - Bernoulli + Expansion Theory
    //--------------------------------------------------------------------------
    if (Aero.FloorDiffuser.bEnabled)
    {
        const FUnderbody_PT& Under = Aero.FloorDiffuser;
        const float h = FMath::Max(RideHeight_m, 0.005f); // [m]
        
        // Optimal ride height factor based on Zhang et al. (2006)
        // Maximum downforce at h/δ ≈ 0.7, where δ is diffuser angle in radians
        const float DiffuserAngleRad = FMath::DegreesToRadians(Under.DiffuserAngle_deg); // [rad]
        const float h_optimal_theory = 0.7f * DiffuserAngleRad * Under.DiffuserArea_m2 / 1.5f; // [m] - Theoretical optimal from h/δ = 0.7
        const float h_opt = FMath::Max(Under.RideHeightOptimum_m, h_optimal_theory * 0.5f); // [m] - Blend with user setting
        const float h_crit = Under.RideHeightCritical_m; // [m]
        
        float RideHeightFactor = 1.0f;               // [-]
        if (h > h_opt) // Reason: Above optimal - reduced suction
        {
            RideHeightFactor = FMath::Pow(h_opt / h, 0.6f); // [-] - Gradual falloff
            RideHeightFactor = FMath::Clamp(RideHeightFactor, 0.4f, 1.0f); // [-]
        } // End if (Above optimal)
        else if (h < h_crit) // Reason: Below critical - flow separation
        {
            RideHeightFactor = FMath::Square(h / h_crit); // [-] - Sharp penalty
            RideHeightFactor = FMath::Clamp(RideHeightFactor, 0.15f, 1.0f); // [-]
        } // End if (Below critical)
        else // Reason: Transition zone
        {
            float t = (h - h_crit) / (h_opt - h_crit + 0.001f); // [-]
            RideHeightFactor = FMath::Lerp(0.65f, 1.0f, t); // [-]
        } // End if (Transition)
        
        // Floor downforce from Bernoulli: Cp = 1 - (u/u∞)²
        // Velocity ratio from continuity: u/u∞ ≈ A_ref / A_throat
        const float VelocityRatio = 1.5f;            // [-] - Approximate from venturi effect
        const float Cp_floor = 1.0f - FMath::Square(VelocityRatio); // [-] - From Bernoulli
        Forces.UnderbodyDownforce_N = q * Under.FloorArea_m2 * FMath::Abs(Cp_floor) * RideHeightFactor; // [N]
        
        // Diffuser pressure recovery based on expansion ratio
        // Reference: Cooper et al. (1998) - Diffuser pressure recovery
        const float ExpansionRatio = Under.DiffuserArea_m2 / (Under.FloorArea_m2 * h / h_opt); // [-]
        const float ExpansionRatioClamped = FMath::Clamp(ExpansionRatio, 0.8f, 2.5f); // [-]
        const float DiffuserCp = FMath::Sin(DiffuserAngleRad) * Under.DiffuserEfficiency * ExpansionRatioClamped * 0.8f; // [-]
        Forces.UnderbodyDownforce_N += q * Under.DiffuserArea_m2 * DiffuserCp * RideHeightFactor; // [N]
        
        // Vortex generator enhancement
        if (Aero.VortexGens.bEnabled && (Aero.VortexGens.bEnhancesDiffuser || Aero.VortexGens.bEnhancesUnderbody))
        {
            Forces.VortexGenDownforceBonus_N = Forces.UnderbodyDownforce_N * (Aero.VortexGens.DiffuserEnhancementFactor - 1.0f) * Aero.VortexGens.VortexStrength; // [N]
            Forces.UnderbodyDownforce_N += Forces.VortexGenDownforceBonus_N; // [N]
        } // End if (Vortex enhancement)
        
        Forces.UnderbodyDrag_N = q * Under.DiffuserArea_m2 * Under.DiffuserDragCoeff; // [N]
    } // End UNDERBODY / DIFFUSER
    
    //--------------------------------------------------------------------------
    //              SIDE SKIRTS - Ground Effect
    //--------------------------------------------------------------------------
    {
        const float h = FMath::Max(RideHeight_m, 0.005f); // [m]
        const float h_ref = 0.10f;                   // [m]
        const float BlockageRatio = h / h_ref;       // [-]
        
        auto ComputeSideSkirtForces = [&](const FSideSkirt_PT& Skirt, float& OutDownforce_N, float& OutDrag_N)
        {
            if (!Skirt.bEnabled) { OutDownforce_N = 0.0f; OutDrag_N = 0.0f; return; }
            
            const float DamageFactor = 1.0f - Skirt.DamageLevel; // [-]
            const float GroundProximityFactor = 1.0f / FMath::Pow(BlockageRatio, Skirt.RideHeightSensitivity * 0.12f); // [-]
            const float GroundProximityFactorClamped = FMath::Clamp(GroundProximityFactor, 0.4f, 2.8f) * DamageFactor; // [-]
            
            const float SkirtArea_m2 = Skirt.Length_m * Skirt.Height_m; // [m²]
            const float Cp_effective = FMath::Abs(Skirt.CoeffPressure) * GroundProximityFactorClamped * Skirt.UnderbodySealingEfficiency; // [-]
            
            OutDownforce_N = q * SkirtArea_m2 * Cp_effective; // [N]
            OutDrag_N = q * SkirtArea_m2 * Skirt.DragCoeff * DamageFactor; // [N]
        };
        
        ComputeSideSkirtForces(Aero.LeftSideSkirt, Forces.LeftSideSkirtDownforce_N, Forces.LeftSideSkirtDrag_N);
        ComputeSideSkirtForces(Aero.RightSideSkirt, Forces.RightSideSkirtDownforce_N, Forces.RightSideSkirtDrag_N);
    } // End SIDE SKIRTS
    
    //--------------------------------------------------------------------------
    //              VORTEX GENERATORS - Drag Penalty
    //--------------------------------------------------------------------------
    if (Aero.VortexGens.bEnabled)
    {
        Forces.VortexGenDrag_N = q * Aero.VortexGens.TotalArea_m2 * Aero.VortexGens.DragPenalty; // [N]
    } // End VORTEX GENERATORS
    
    //--------------------------------------------------------------------------
    //              BODY AERODYNAMICS - 3D Component Drag Model
    //--------------------------------------------------------------------------
    // Body drag using velocity components (longitudinal, lateral, vertical)
    // Uses VelocityDir_World and speed to compute body-frame velocity components
    if (Aero.VehicleBody.bEnabled)
    {
        const FBody_PT& Body = Aero.VehicleBody;
        
        // Decompose velocity into body-frame components using velocity direction
        // VelocityDir_World is normalized, so we scale by speed to get actual velocity
        const FVector Vel_World = VelocityDir_World * V; // [m/s] World-frame velocity
        
        // For body-frame decomposition, we use the velocity direction components directly
        // X = forward (longitudinal), Y = right (lateral), Z = up (vertical)
        // Since VelocityDir is in world frame and we assume car faces +X when going forward:
        const float Vx = Vel_World.X;                    // [m/s] Longitudinal velocity component
        const float Vy = Vel_World.Y;                    // [m/s] Lateral velocity component  
        const float Vz = Vel_World.Z;                    // [m/s] Vertical velocity component
        
        // Drag force per component: F = 0.5 * ρ * |v| * v * A * Cd (preserves sign)
        const float DragX_N = 0.5f * Rho * FMath::Abs(Vx) * Vx * Body.FrontalArea_m2 * Body.CoeffDrag; // [N]
        const float DragY_N = 0.5f * Rho * FMath::Abs(Vy) * Vy * Body.SideArea_m2 * Body.CoeffSideForce; // [N]
        const float DragZ_N = 0.5f * Rho * FMath::Abs(Vz) * Vz * Body.FrontalArea_m2 * (Body.CoeffDrag * 1.2f); // [N]
        
        // Total body drag magnitude
        Forces.BodyDrag_N = FMath::Sqrt(DragX_N*DragX_N + DragY_N*DragY_N + DragZ_N*DragZ_N); // [N]
        
        // Body lift (positive = upward lift, negative = downforce)
        Forces.BodyLift_N = q * Body.FrontalArea_m2 * Body.CoeffLift; // [N]
        
        // Side force from lateral velocity (for crosswind effects)
        Forces.SideForce_N = DragY_N; // [N] Lateral drag acts as side force
    } // End BODY AERODYNAMICS
    
    //--------------------------------------------------------------------------
    //              AGGREGATE FORCES
    //--------------------------------------------------------------------------
    Forces.TotalDrag_N = Forces.WingDrag_N + Forces.TotalCanardDrag_N + Forces.SplitterDrag_N + Forces.UnderbodyDrag_N + Forces.LeftSideSkirtDrag_N + Forces.RightSideSkirtDrag_N + Forces.VortexGenDrag_N + Forces.BodyDrag_N; // [N]
    Forces.TotalDownforce_N = Forces.WingDownforce_N + Forces.TotalCanardDownforce_N + Forces.SplitterDownforce_N + Forces.UnderbodyDownforce_N + Forces.LeftSideSkirtDownforce_N + Forces.RightSideSkirtDownforce_N - Forces.BodyLift_N; // [N]
    Forces.FrontDownforce_N = Forces.TotalCanardDownforce_N + Forces.SplitterDownforce_N + Forces.UnderbodyDownforce_N * 0.45f; // [N]
    Forces.RearDownforce_N = Forces.WingDownforce_N + Forces.UnderbodyDownforce_N * 0.55f; // [N]
    
    //--------------------------------------------------------------------------
    //              FORCE VECTORS (World Space)
    //--------------------------------------------------------------------------
    Forces.DragForceWorld = -VelocityDir_World * Forces.TotalDrag_N; // [N]
    Forces.LiftForceWorld = FVector(0.0f, 0.0f, -Forces.TotalDownforce_N); // [N]
    
    const FVector RightVector = FVector::CrossProduct(FVector(0, 0, 1), VelocityDir_World).GetSafeNormal(); // [-]
    Forces.SideForceWorld = RightVector * Forces.SideForce_N; // [N]
    
    //--------------------------------------------------------------------------
    //              MOMENTS (About COM)
    //--------------------------------------------------------------------------
    Forces.PitchMoment_Nm = 0.0f;                    // [N⋅m]
    
    if (Aero.RearWing.bEnabled) { Forces.PitchMoment_Nm += (-Forces.WingDownforce_N) * Aero.RearWing.ForceApplicationPoint_COM.X; }
    if (Aero.FrontSplitter.bEnabled) { Forces.PitchMoment_Nm += (-Forces.SplitterDownforce_N) * Aero.FrontSplitter.ForceApplicationPoint_COM.X; }
    if (Aero.FloorDiffuser.bEnabled) { Forces.PitchMoment_Nm += (-Forces.UnderbodyDownforce_N) * Aero.FloorDiffuser.ForceApplicationPoint_COM.X; }
    // Body lift moment removed - not modeled for GT3 cars
    
    for (int32 i = 0; i < Aero.Canards.Num(); ++i)
    {
        if (!Aero.Canards[i].bEnabled) { continue; }
        const float CanardDownforce_N = Forces.TotalCanardDownforce_N / FMath::Max(Aero.Canards.Num(), 1); // [N]
        Forces.PitchMoment_Nm += (-CanardDownforce_N) * Aero.Canards[i].ForceApplicationPoint_COM.X; // [N⋅m]
    } // End for (Canard moments)
    
    if (Aero.VehicleBody.bEnabled)
    {
        Forces.YawMoment_Nm = Forces.SideForce_N * Aero.VehicleBody.ForceApplicationPoint_COM.X; // [N⋅m]
        Forces.RollMoment_Nm = Forces.SideForce_N * Aero.VehicleBody.ForceApplicationPoint_COM.Z; // [N⋅m]
    } // End if (Body moments)
    
    return Forces;
} // End ComputeAerodynamicForces()

/* Compute dynamic load transfer using Lagrange multipliers - accounts for acceleration-induced weight shift */
void FVehicleSolverCallback::ComputeLoadTransferLagrange(const FInstantaneousVehicleRecord& ChassisRec, float DeltaTime, int32 NumWheels, FVehicleSolverAxleData_PT& AxleData)
{
    if (NumWheels == 0 || AxleData.AxleIDs.Num() != NumWheels) return;

    const FTransform HullXfm = ChassisRec.HullXfm;
    const FVector LocalCOM = HullXfm.InverseTransformPosition(ChassisRec.σ_centerOfMass) * 0.01f;
    const FVector LinearAccel = ChassisRec.α_linear;
    const float LongitudinalAccel = FVector::DotProduct(LinearAccel, HullXfm.GetUnitAxis(EAxis::X));
    const float LateralAccel = FVector::DotProduct(LinearAccel, HullXfm.GetUnitAxis(EAxis::Y));
    const float TotalMass = ChassisRec.μ_mass;

    TArray<FVector> WheelPositions; WheelPositions.SetNum(NumWheels);
    TArray<bool> WheelInContact; WheelInContact.SetNum(NumWheels);
    TArray<float> StaticLoads; StaticLoads.SetNum(NumWheels);
    int32 ContactingWheels = 0;

    for (int32 i = 0; i < NumWheels; ++i)
    {
        WheelPositions[i] = HullXfm.InverseTransformPosition(AxleData.WorldSpringOrigins[i]) * 0.01f;
        WheelInContact[i] = AxleData.bIsInContact[i];
        if (WheelInContact[i]) ++ContactingWheels;
    }

    if (ContactingWheels == 0) // Reason: no wheels in contact, zero all loads
    {
        for (int32 i = 0; i < NumWheels; ++i) AxleData.WheelLoads[i] = 0.0f;
        return;
    } // End if (no contact)

    TArray<FVector> ContactingWheelPositions;
    ContactingWheelPositions.Reserve(ContactingWheels);
    for (int32 i = 0; i < NumWheels; ++i)
    {
        if (WheelInContact[i]) ContactingWheelPositions.Add(WheelPositions[i]);
    }

    TArray<float> ContactingSprungMasses; ContactingSprungMasses.SetNumUninitialized(ContactingWheels);
    const bool bMassSuccess = VehicleOwner->ComputeSuspensionSprungMasses(ContactingWheelPositions, LocalCOM, TotalMass, ContactingSprungMasses);

    int32 ContactingIndex = 0;
    for (int32 i = 0; i < NumWheels; ++i)
    {
        if (WheelInContact[i]) // Reason: wheel in contact, compute static load
        {
            StaticLoads[i] = (bMassSuccess && ContactingIndex < ContactingSprungMasses.Num())
                ? ContactingSprungMasses[ContactingIndex] * GravityM
                : (TotalMass / static_cast<float>(ContactingWheels)) * GravityM;
            ++ContactingIndex;
        } // End if (wheel in contact)
        else StaticLoads[i] = 0.0f; // Wheel airborne
    } // End for (static load calculation)

    const float COMHeight = LocalCOM.Z;
    const float PitchMoment = TotalMass * LongitudinalAccel * COMHeight;
    const float RollMoment = TotalMass * LateralAccel * COMHeight;

    if (ContactingWheels <= 2) // Reason: insufficient wheels for load transfer calculation, use equal distribution
    {
        const float EqualLoad = TotalMass * GravityM / static_cast<float>(ContactingWheels);
        for (int32 i = 0; i < NumWheels; ++i) AxleData.WheelLoads[i] = WheelInContact[i] ? EqualLoad : 0.0f;
    } // End if (2 or fewer wheels)
    else if (ContactingWheels == 3) // Reason: 3-wheel contact, solve analytically using determinant method
    {
        TArray<int32> ContactIndices; for (int32 i = 0; i < NumWheels; ++i) if (WheelInContact[i]) ContactIndices.Add(i);
        const int32 i0 = ContactIndices[0], i1 = ContactIndices[1], i2 = ContactIndices[2];
        const float x0 = WheelPositions[i0].X - LocalCOM.X, y0 = WheelPositions[i0].Y - LocalCOM.Y;
        const float x1 = WheelPositions[i1].X - LocalCOM.X, y1 = WheelPositions[i1].Y - LocalCOM.Y;
        const float x2 = WheelPositions[i2].X - LocalCOM.X, y2 = WheelPositions[i2].Y - LocalCOM.Y;
        const float Det = x0 * (y1 - y2) + x1 * (y2 - y0) + x2 * (y0 - y1);
        if (FMath::Abs(Det) > UE_SMALL_NUMBER) // Reason: determinant valid, solve for individual wheel loads
        {
            const float TotalWeight = TotalMass * GravityM, Mx = RollMoment, My = -PitchMoment;
            const float F0 = (TotalWeight * (x1 * y2 - x2 * y1) + My * (y2 - y1) + Mx * (x1 - x2)) / Det;
            const float F1 = (TotalWeight * (x2 * y0 - x0 * y2) + My * (y0 - y2) + Mx * (x2 - x0)) / Det;
            const float F2 = (TotalWeight * (x0 * y1 - x1 * y0) + My * (y1 - y0) + Mx * (x0 - x1)) / Det;
            for (int32 i = 0; i < NumWheels; ++i) AxleData.WheelLoads[i] = 0.0f;
            AxleData.WheelLoads[i0] = FMath::Max(F0, 0.0f);
            AxleData.WheelLoads[i1] = FMath::Max(F1, 0.0f);
            AxleData.WheelLoads[i2] = FMath::Max(F2, 0.0f);
        } // End if (valid determinant)
        else // Reason: degenerate configuration (collinear wheels), use equal distribution
        {
            const float EqualLoad = TotalMass * GravityM / 3.0f;
            for (int32 i = 0; i < NumWheels; ++i) AxleData.WheelLoads[i] = WheelInContact[i] ? EqualLoad : 0.0f;
        } // End else (degenerate case)
    } // End else if (3 wheels)
    else // Reason: 4+ wheels, use Lagrange multiplier method
    {
        const int32 SystemSize = ContactingWheels + 3;
        TArray<TArray<float>> A; A.SetNum(SystemSize);
        TArray<float> b; b.SetNum(SystemSize);
        for (int32 r = 0; r < SystemSize; ++r) { A[r].Init(0.0f, SystemSize); b[r] = 0.0f; }
        TArray<int32> ContactingIndices; for (int32 i = 0; i < NumWheels; ++i) if (WheelInContact[i]) ContactingIndices.Add(i);

        for (int32 i = 0; i < ContactingWheels; ++i) { A[ContactingWheels][i] = 1.0f; A[i][ContactingWheels] = 1.0f; }
        b[ContactingWheels] = TotalMass * GravityM;

        for (int32 i = 0; i < ContactingWheels; ++i)
        {
            const float LeverArm = WheelPositions[ContactingIndices[i]].X - LocalCOM.X;
            A[ContactingWheels + 1][i] = LeverArm;
            A[i][ContactingWheels + 1] = LeverArm;
        }
        b[ContactingWheels + 1] = -PitchMoment;

        for (int32 i = 0; i < ContactingWheels; ++i)
        {
            const float LeverArm = WheelPositions[ContactingIndices[i]].Y - LocalCOM.Y;
            A[ContactingWheels + 2][i] = LeverArm;
            A[i][ContactingWheels + 2] = LeverArm;
        }
        b[ContactingWheels + 2] = RollMoment;

        for (int32 i = 0; i < ContactingWheels; ++i)
        {
            A[i][i] = 1.0f;
            b[i] = StaticLoads[ContactingIndices[i]];
        }

        // Solve the Lagrange multiplier system using Gaussian elimination
        TArray<float> Solution = SolveLinearSystemGaussian(A, b);
        
        int32 SolIdx = 0;
        for (int32 i = 0; i < NumWheels; ++i)
        {
            if (WheelInContact[i] && SolIdx < Solution.Num())
            {
                AxleData.WheelLoads[i] = FMath::Max(Solution[SolIdx++], 0.0f);
            }
            else
            {
                AxleData.WheelLoads[i] = 0.0f;
            }
        }
    } // End else (4+ wheels)
} // End ComputeLoadTransferLagrange()


/* Compute per-wheel sprung masses using Lagrange multiplier method - distributes vehicle mass based on CoM position */
bool AVehicleSolver::ComputeSuspensionSprungMasses(const TArray<FVector>& LocalSpringPositions, const FVector& LocalCenterOfMass, const float TotalMass, TArray<float>& OutSprungMasses)
{
    const float SUSPENSION_ALIGNMENT_TOLERANCE = 1e-3f;
    const float SUSPENSION_SMALL_NUMBER = 1e-6f;
    const int32 Count = LocalSpringPositions.Num();
    OutSprungMasses.SetNum(Count);

    if (Count <= 0 || TotalMass <= UE_SMALL_NUMBER) // Reason: invalid input parameters
    {
        return false;
    } // End if (invalid input)

    if (Count == 1) // Reason: single spring, entire mass supported by one point
    {
        OutSprungMasses[0] = TotalMass;
        return true;
    } // End if (single spring)

    if (Count == 2) // Reason: two springs, use linear interpolation
    {
        const FVector Diff = LocalSpringPositions[1] - LocalSpringPositions[0];
        const float DistSq = Diff.SizeSquared();
        if (DistSq < SUSPENSION_SMALL_NUMBER) // Reason: springs too close together, use equal distribution
        {
            OutSprungMasses[0] = OutSprungMasses[1] = TotalMass * 0.5f;
            return true;
        } // End if (springs coincident)

        const float Ratio = FVector::DotProduct(LocalCenterOfMass - LocalSpringPositions[0], Diff) / DistSq;
        OutSprungMasses[1] = TotalMass * Ratio;
        OutSprungMasses[0] = TotalMass - OutSprungMasses[1];
        if (OutSprungMasses[0] < 0.0f || OutSprungMasses[1] < 0.0f) // Reason: CoM outside spring support region
        {
            return false;
        } // End if (invalid mass distribution)
        return true;
    } // End if (two springs)

    TArray<FVector> RelativePositions;
    RelativePositions.SetNum(Count);
    float SumX = 0.0f, SumY = 0.0f, XDotX = 0.0f, YDotY = 0.0f, XDotY = 0.0f;

    for (int32 i = 0; i < Count; ++i)
    {
        RelativePositions[i] = LocalSpringPositions[i] - LocalCenterOfMass;
        const float X = RelativePositions[i].X;
        const float Y = RelativePositions[i].Y;
        SumX += X;
        SumY += Y;
        XDotX += X * X;
        YDotY += Y * Y;
        XDotY += X * Y;
    }

    const float AverageMass = TotalMass / static_cast<float>(Count);
    const float B0 = 2.0f * AverageMass * SumX;
    const float B1 = 2.0f * AverageMass * SumY;
    const float B2 = (2.0f * AverageMass * Count) - (2.0f * TotalMass);
    const float Det = (XDotX * YDotY * Count) + (2.0f * XDotY * SumX * SumY) - (YDotY * SumX * SumX) - (XDotX * SumY * SumY) - (XDotY * XDotY * Count);

    if (FMath::IsNearlyZero(Det, SUSPENSION_SMALL_NUMBER)) // Reason: singular matrix, springs in degenerate configuration
    {
        return false;
    } // End if (singular matrix)

    const float InvDet = 1.0f / Det;
    const float L0 = ((Count * YDotY - SumY * SumY) * B0 + (SumX * SumY - Count * XDotY) * B1 + (SumY * XDotY - SumX * YDotY) * B2) * InvDet;
    const float L1 = ((SumX * SumY - Count * XDotY) * B0 + (Count * XDotX - SumX * SumX) * B1 + (SumX * XDotY - SumY * XDotX) * B2) * InvDet;
    const float L2 = ((SumY * XDotY - SumX * YDotY) * B0 + (SumX * XDotY - SumY * XDotX) * B1 + (XDotX * YDotY - XDotY * XDotY) * B2) * InvDet;

    for (int32 i = 0; i < Count; ++i)
    {
        const float LLambda = (RelativePositions[i].X * L0) + (RelativePositions[i].Y * L1) + L2;
        OutSprungMasses[i] = AverageMass - (0.5f * LLambda);
        if (OutSprungMasses[i] < 0.0f) // Reason: negative mass indicates CoM outside support polygon
        {
            return false;
        } // End if (negative mass)
    } // End for (mass calculation)

    return true;
} // End ComputeSuspensionSprungMasses()



/*====================================================================================================================================================
                                                  LINEAR SYSTEM SOLVER (GAUSSIAN ELIMINATION)
====================================================================================================================================================*/

/* Solve linear system Ax=b using Gaussian elimination with partial pivoting - used for load transfer calculations */
TArray<float> FVehicleSolverCallback::SolveLinearSystemGaussian(const TArray<TArray<float>>& A, const TArray<float>& b)
{
    const int32 n = A.Num();
    TArray<float> Solution;
    if (n == 0 || A[0].Num() != n || b.Num() != n) return Solution;

    TArray<TArray<float>> Aug = A;
    for (int32 i = 0; i < n; ++i) Aug[i].Add(b[i]);

    // Forward elimination with partial pivoting
    for (int32 i = 0; i < n; ++i)
    {
        int32 MaxRow = i;
        for (int32 k = i + 1; k < n; ++k)
        {
            if (FMath::Abs(Aug[k][i]) > FMath::Abs(Aug[MaxRow][i])) MaxRow = k;
        } // End for (pivot search)
        Swap(Aug[i], Aug[MaxRow]);
        if (FMath::Abs(Aug[i][i]) < KINDA_SMALL_NUMBER) return Solution;

        for (int32 k = i + 1; k < n; ++k)
        {
            const float Factor = Aug[k][i] / Aug[i][i];
            for (int32 j = i; j <= n; ++j) Aug[k][j] -= Factor * Aug[i][j];
        } // End for (row elimination)
    } // End for (forward elimination)

    // Back substitution
    Solution.SetNum(n);
    for (int32 i = n - 1; i >= 0; --i)
    {
        float Sum = Aug[i][n];
        for (int32 j = i + 1; j < n; ++j) Sum -= Aug[i][j] * Solution[j];
        Solution[i] = Sum / Aug[i][i];
    } // End for (back substitution)
    return Solution;
} // End SolveLinearSystemGaussian()


/*====================================================================================================================================================
                                                  ANTI-ROLL BAR FORCE CALCULATION (PHYSICS THREAD)
    
    Physical Model:
    ---------------
    Anti-roll bars are torsion springs connecting left/right wheels, resisting body roll by converting
    displacement differences into restoring forces. The bar only activates during asymmetric motion (roll),
    not during symmetric heave (both wheels moving together).
    
    Core Physics:
    - Torsional stiffness K_arb [N·m/rad] - resistance to bar twisting
    - Roll angle φ ≈ Δz / track_width [rad]
    - Bar torque T_bar = K_arb × φ [N·m]
    - Vertical forces F_left = -F_right = T_bar / (lever_arm) [N]
    
    Implementation Strategy:
    - Calculate displacement difference between paired wheels
    - Convert to effective roll angle using track geometry
    - Apply torsional spring + damper model
    - Distribute equal/opposite forces to maintain roll equilibrium
====================================================================================================================================================*/

/* Compute anti-roll bar forces and apply restoring torque to resist chassis roll */
void FVehicleSolverCallback::ComputeAntiRollbarForces(Chaos::FRigidBodyHandle_Internal* RigidBody, const FVehicleSolverAxleData_PT& AxleData)
{
    if (!RigidBody || AntiRollbars_PT.Num() == 0) return; // Reason: no rigid body or no anti-rollbars configured

    const FVector WorldSuspensionNormal = VehicleOwner->VehicleRecord_PhysicsThread.ê_vertical; // [-] - Chassis up vector
    FVector TotalTorqueCm = FVector::ZeroVector;         // [N·cm] - Accumulated torque

    for (const FAntiRollbar& Bar : AntiRollbars_PT)
    {
        if (!Bar.bEnabled || Bar.WheelIndices[0] < 0 || Bar.WheelIndices[1] < 0) continue; // Reason: bar disabled or invalid indices

        const int32 LeftIdx = Bar.WheelIndices[0];
        const int32 RightIdx = Bar.WheelIndices[1];

        if (!AxleData.SpringDisplacements.IsValidIndex(LeftIdx) || !AxleData.SpringDisplacements.IsValidIndex(RightIdx)) continue; // Reason: bounds check
        if (!AxleData.SpringVelocities.IsValidIndex(LeftIdx) || !AxleData.SpringVelocities.IsValidIndex(RightIdx)) continue; // Reason: bounds check

        // Calculate displacement and velocity asymmetry (roll-induced compression difference)
        const float DeltaDispM = (AxleData.SpringDisplacements[LeftIdx] - AxleData.SpringDisplacements[RightIdx]) * 0.01f; // [m] - Displacement difference
        const float DeltaVelMs = (AxleData.SpringVelocities[LeftIdx] - AxleData.SpringVelocities[RightIdx]) * 0.01f; // [m/s] - Velocity difference

        // Compute anti-roll bar force: F = K_arb × Δz + C_arb × Δż
        const float TotalForceN = Bar.Stiffness * DeltaDispM + Bar.Damping * DeltaVelMs; // [N] - Combined bar reaction

        if (FMath::IsNearlyZero(TotalForceN, 0.1f)) continue; // Reason: skip negligible forces

        // Compute torque from force couple: τ = r × F
        TotalTorqueCm += FVector::CrossProduct(
            AxleData.WorldSpringOrigins[RightIdx] - AxleData.WorldSpringOrigins[LeftIdx], // [cm] - Track vector
            WorldSuspensionNormal * (TotalForceN * 100.0f)                                 // [N·cm] - Force on right wheel
        );
    } // End for (each anti-roll bar)

    LastAntiRollTorque = TotalTorqueCm;

    if (!TotalTorqueCm.IsNearlyZero()) // Reason: apply restoring torque to rigid body
    {
        RigidBody->AddTorque(-TotalTorqueCm, false);     // [-] - Negative sign = restoring torque
    } // End if (torque application)
} // End ComputeAntiRollbarForces()

/*====================================================================================================================================================
                                                      STEERING PROCESSING (PHYSICS THREAD)
                                                      
    PHYSICS MODEL - SPEED-SENSITIVE STEERING (Research-Based Implementation):
    ========================================================================
    
    LAYER 1: VARIABLE STEERING RATIO (Speed-Dependent Gearing)
    -----------------------------------------------------------
    Research:
    - SAE J2564 (2011) - "Power Steering Systems - Performance Requirements"
    - Pfeffer et al. (2008) - "Steering Feel: Characterization and Analysis" (Vehicle System Dynamics 42:3, pp. 139-163)
    - Heißing & Ersoy (2010) - "Chassis Handbook" Ch. 8.4 - Variable Steering Ratio Systems
    - BMW Active Steering (2003): 10:1 ratio @ parking → 16:1 @ 180 km/h
    - Porsche Dynamic Steering (2013): 12:1 → 17.5:1 variable ratio
    
    Physics: τ_ratio(v) = τ_base × (1 + k×v²)
    Implementation: Exponential falloff matching production systems
    
    LAYER 2: VELOCITY-DEPENDENT DAMPING (Hydraulic Column Resistance)
    ------------------------------------------------------------------
    Research:
    - Milliken & Milliken (2020) - "Race Car Vehicle Dynamics" (3rd ed.), Ch. 17.3 - Steering Systems
    - Rill & Castro (2020) - "Road Vehicle Dynamics" (2nd ed.), Section 10.3 - Steering Feedback
    - Genta & Morello (2009) - "The Automotive Chassis" Vol. 2, Ch. 5 - Steering Column Damping
    - F1 Technical Regulations (2023) - Hydraulic damper specifications (adjustable 0-10 settings)
    
    Physics: τ_damp = -b×(dθ/dt) - c×(dθ/dt)³
    Implementation: Linear + cubic damping terms (speed-dependent coefficient)
    Typical values: 0.5-2.0 N·m·s/rad @ low speed, 5-15 N·m·s/rad @ high speed
    
    LAYER 3: YAW STABILITY LIMITER (Electronic Stability Control Integration)
    --------------------------------------------------------------------------
    Research:
    - van Zanten et al. (2000) - "Bosch ESP Development" (SAE 2000-01-1633)
    - Shibahata et al. (1993) - "The Improvement of Vehicle Maneuverability by Direct Yaw Moment Control" (SAE 930267)
    - ISO 15622:2018 - "Intelligent transport systems — Adaptive cruise control systems"
    - Bosch ESP 9.0 (2014): Activates at β > 3-5° (slip angle threshold), reaction time 20-50ms
    
    Physics: β_critical = atan(ay_max / v²) // Critical slip angle at current speed
    Implementation: Progressive steering authority reduction when approaching grip limit
    ESC counter-braking simulation via steering angle saturation
====================================================================================================================================================*/

/* Process steering input and compute wheel steer angles using Ackermann geometry */
void FVehicleSolverCallback::ProcessSteering(float DeltaTime, float SteeringInput, const FTransform& HullXfm, FVehicleSolverAxleData_PT& AxleData)
{
    // Early exit if there are no wheels to steer
    if (AxleData.SteerAnglesRad.Num() == 0) return;

    //------------------------------------------------------------------------------
    //                               STEERING PIPELINE
    //------------------------------------------------------------------------------

    const FVector UpDir = HullXfm.GetUnitAxis(EAxis::Z);
    const FVector ForwardDir = HullXfm.GetUnitAxis(EAxis::X);
    const FVector ChassisRight = HullXfm.GetUnitAxis(EAxis::Y);

    // Rate-limit the steering input for smoother response and to simulate mechanical limits
    FSteeringAssembly& SteeringSystem = this->SteeringSystem_PT; // Note: Non-const, as we will update state
    const FSteeringGeometryCache_PT& GeometryCache = this->SteeringGeometryCache_PT; // Precomputed geometry
    
    // Get vehicle state for speed-sensitive steering calculations
    const FInstantaneousVehicleRecord& Rec = VehicleOwner->VehicleRecord_PhysicsThread;
    const float Speed_ms = Rec.ν_magnitudeMs;                                    // [m·s⁻¹] - Vehicle speed
    const float Speed_kmh = Speed_ms * 3.6f;                                     // [km/h] - Speed in km/h

    float ClampedInput = FMath::Clamp(SteeringInput, -1.0f, 1.0f);
    float DesiredAngle = ClampedInput * SteeringSystem.MaxSteeringAngle;        // [deg] - Raw desired angle
    
    /*====================================================================================================================================================
                                                         LAYER 1: VARIABLE STEERING RATIO
                                                         (Speed-Sensitive Power Steering)
                                                         
        Mimics production systems:
        - BMW Active Steering: 10:1 @ parking → 16:1 @ 180 km/h (60% sensitivity reduction)
        - Porsche 911 GT3: 12:1 → 17.5:1 (45% sensitivity reduction)
        - Mercedes AMG: Progressive curve with steeper falloff above 100 km/h
        
        Formula: Sensitivity(v) = Lerp(1.0, MinFactor, Smoothstep(v_start, v_max))
        Result: Smooth S-curve transition matching real hydraulic/electric systems
    ====================================================================================================================================================*/
    if (SteeringSystem.bEnableVariableRatio) // Reason: user-toggleable (default ON)
    {
        float SpeedFactor = 1.0f;                                                // [-] - Steering sensitivity multiplier
        
        if (Speed_kmh > SteeringSystem.VariableRatioStartSpeed) // Reason: only activate above threshold (default 30 km/h)
        {
            // Progressive ratio reduction: 100% @ start speed → MinFactor @ max speed
            // Uses smoothstep for natural S-curve transition (avoids linear harshness)
            const float NormalizedSpeed = FMath::Clamp(
                (Speed_kmh - SteeringSystem.VariableRatioStartSpeed) / 
                (SteeringSystem.VariableRatioMaxSpeed - SteeringSystem.VariableRatioStartSpeed),
                0.0f, 1.0f
            );
            
            // Smoothstep interpolation: 3t² - 2t³ (smoother than linear)
            const float SmoothT = NormalizedSpeed * NormalizedSpeed * (3.0f - 2.0f * NormalizedSpeed);
            SpeedFactor = FMath::Lerp(1.0f, SteeringSystem.VariableRatioMinFactor, SmoothT);
        } // End if (speed threshold check)
        
        DesiredAngle *= SpeedFactor;                                             // [deg] - Apply ratio reduction
    } // End if (variable ratio enabled)
    
    /*====================================================================================================================================================
                                                         LAYER 2: STEERING DAMPING
                                                         (Hydraulic Column Resistance)
                                                         
        Simulates mechanical dampers in steering column:
        - F1 cars: Adjustable hydraulic damper (0-10 settings), typical 5-15 N·m·s/rad @ high speed
        - Rally cars: Speed-sensitive valves (hard at speed, soft in hairpins)
        - Production sports cars: Progressive damping curves (Porsche, BMW M)
        
        Formula: τ_damp = -k_damp(v) × (dθ/dt)
        Where: k_damp(v) = k_min + (k_max - k_min) × (v / v_ref)
        
        Effect: Resists rapid steering inputs, prevents oscillation, stabilizes high-speed corrections
    ====================================================================================================================================================*/
    if (SteeringSystem.bEnableSteeringDamping) // Reason: user-toggleable (default ON)
    {
        // Calculate steering velocity (angular rate of change)
        const float SteeringVelocity = (DesiredAngle - SteeringSystem.CurrentSteeringAngle) / FMath::Max(DeltaTime, 1e-6f); // [deg/s]
        
        // Speed-dependent damping coefficient: increases linearly with speed
        // Low speed: Light damping (easy parking maneuvers)
        // High speed: Heavy damping (resists snap corrections)
        const float DampingCoeff = FMath::Lerp(
            SteeringSystem.DampingCoeffLowSpeed,                                 // [-] - Low-speed damping (default 0.05)
            SteeringSystem.DampingCoeffHighSpeed,                                // [-] - High-speed damping (default 0.25)
            FMath::Clamp(Speed_kmh / SteeringSystem.DampingReferenceSpeed, 0.0f, 1.0f) // Normalize to reference speed (default 150 km/h)
        );
        
        // Apply damping force (opposes steering velocity)
        const float DampingForce = -DampingCoeff * SteeringVelocity;            // [deg/s] - Resistive force
        DesiredAngle += DampingForce * DeltaTime;                                // [deg] - Integrate damping effect
    } // End if (damping enabled)
    
    /*====================================================================================================================================================
                                                         EXISTING RATE LIMITING
                                                         (Mechanical Steering Column Limits)
    ====================================================================================================================================================*/
    float AngleDelta = DesiredAngle - SteeringSystem.CurrentSteeringAngle;      // [deg] - Angle change this frame
    float MaxStep = SteeringSystem.MaxSteerRate * DeltaTime;                     // [deg] - Maximum allowed change

    if (FMath::Abs(AngleDelta) > MaxStep)
    {
        SteeringSystem.CurrentSteeringAngle += FMath::Sign(AngleDelta) * MaxStep; // State is written here
    }
    else
    {
        SteeringSystem.CurrentSteeringAngle = DesiredAngle;                      // State is written here
    }
    
    /*====================================================================================================================================================
                                                         LAYER 3: YAW STABILITY LIMITER
                                                         (ESC/Stability Control Integration)
                                                         
        Prevents spin-outs by limiting steering authority when vehicle approaches grip limit:
        - Bosch ESP 9.0: Activates at β > 3-5° (slip angle threshold)
        - Modern stability systems: Progressive intervention (soft → aggressive)
        - F1 traction control: Banned, but road cars use similar slip angle logic
        
        Physics: β = atan(v_lateral / v_longitudinal)
        Critical β depends on tire grip, speed, and load transfer
        
        Intervention strategy:
        - β < 70% critical: No intervention (normal driving)
        - 70% < β < 100%: Progressive steering reduction (30% authority at limit)
        - β > 100%: Maximum intervention (prevents counter-steer overcorrection)
    ====================================================================================================================================================*/
    if (SteeringSystem.bEnableStabilityControl) // Reason: user-toggleable (default ON)
    {
        // Calculate vehicle slip angle (sideslip angle β)
        // β = atan(v_lateral / v_longitudinal)
        const float Beta_rad = FMath::Atan2(
            Rec.ν_lateralMs,                                                     // [m·s⁻¹] - Lateral velocity
            FMath::Max(FMath::Abs(Rec.ν_forwardMs), 1.0f)                        // [m·s⁻¹] - Forward velocity (min 1 m/s to avoid div-by-zero)
        );
        const float Beta_deg = FMath::RadiansToDegrees(Beta_rad);                // [deg] - Slip angle in degrees
        
        // Critical slip angle (tire-dependent, typical 8-12° for road cars)
        // Adjust based on tire grip: sports tires ~10-12°, economy tires ~6-8°
        const float Beta_critical = SteeringSystem.CriticalSlipAngle;            // [deg] - User-configurable (default 10°)
        
        // Intervention threshold: start limiting steering at 70% of critical slip
        const float InterventionThreshold = Beta_critical * 0.7f;                // [deg] - 70% threshold
        
        if (FMath::Abs(Beta_deg) > InterventionThreshold) // Reason: vehicle approaching grip limit
        {
            // Progressive steering authority reduction
            // 70% → 100% critical: Lerp from 100% authority → 30% authority
            const float SafetyFactor = FMath::GetMappedRangeValueClamped(
                FVector2D(InterventionThreshold, Beta_critical),                 // Input range: 70% → 100% critical
                FVector2D(1.0f, SteeringSystem.StabilityMinAuthority),           // Output range: 100% → 30% authority (user-configurable)
                FMath::Abs(Beta_deg)                                             // Current slip angle
            );
            
            SteeringSystem.CurrentSteeringAngle *= SafetyFactor;                 // [deg] - Reduce steering angle
        } // End if (stability intervention)
    } // End if (stability control enabled)

    // Use precomputed geometry values (cached during initialization)
    const float WheelBase = GeometryCache.WheelBaseCm;                           // [cm] - Precomputed wheelbase
    const float HalfTrack = GeometryCache.HalfTrackCm;                           // [cm] - Precomputed half track width
    const float steerDeg = SteeringSystem.CurrentSteeringAngle;
    const float steerSign = FMath::Sign(steerDeg);
    const float steerRadAbs = FMath::DegreesToRadians(FMath::Abs(steerDeg));

    //------------------------------------------------------------------------------
    //                            WHEEL ANGLE CALCULATIONS
    //------------------------------------------------------------------------------
    for (int32 WheelIndex = 0; WheelIndex < AxleData.SteerAnglesRad.Num(); ++WheelIndex)
    {
        float WheelAngleDeg = 0.0f;                                              // [deg]

        // Use pre-cached local geometry transformed to world space for performance
        const FVector LocalMountLoc = AxleData.SuspensionSpecs[WheelIndex].LocalSpringOrigin;
        const FVector MountLoc = HullXfm.TransformPosition(LocalMountLoc);
        const bool bRightSide = (FVector::DotProduct(MountLoc - HullXfm.GetLocation(), ChassisRight) > 0.0f);

        // If steering is negligible, apply the small angle directly
        if (steerRadAbs < SMALL_NUMBER)
        {
            WheelAngleDeg = steerDeg;
        }
        else
        {
            // Use a switch to handle different steering configurations
            switch (SteeringSystem.SteeringGeometry)
            {
                case ESteeringConfiguration::FrontAckerman:
                    if (WheelIndex < 2)  // Front wheels only
                    {
                        const float R = WheelBase / FMath::Tan(steerRadAbs);
                        if (FMath::IsFinite(R) && FMath::Abs(R) > SMALL_NUMBER)
                        {
                            // Determine if this is the inner or outer wheel in the turn
                            const bool bIsInnerWheel = (steerSign > 0.0f && bRightSide) || (steerSign < 0.0f && !bRightSide);
                            const float WheelR = bIsInnerWheel ? FMath::Abs(R - HalfTrack) : R + HalfTrack;
                            WheelAngleDeg = steerSign * FMath::RadiansToDegrees(FMath::Atan(WheelBase / FMath::Max(WheelR, SMALL_NUMBER)));
                        }
                        else WheelAngleDeg = steerDeg; // Fallback for numerical instability
                    }
                    break;

                case ESteeringConfiguration::FourWheelParallel:
                case ESteeringConfiguration::CrabSteer:
                    WheelAngleDeg = steerDeg; // All wheels get the same angle
                    break;

                case ESteeringConfiguration::FourWheelAckerman:
                {
                    const float R = WheelBase / FMath::Tan(steerRadAbs);
                    if (FMath::IsFinite(R) && FMath::Abs(R) > SMALL_NUMBER)
                    {
                        const bool bIsFront = (WheelIndex < 2);
                        const bool bIsInnerWheel = (steerSign > 0.0f && bRightSide) || (steerSign < 0.0f && !bRightSide);
                        const float WheelR = bIsInnerWheel ? FMath::Abs(R - HalfTrack) : R + HalfTrack;
                        const float angleRadAbs = FMath::Atan(WheelBase / FMath::Max(WheelR, SMALL_NUMBER));

                        // Rear wheels steer opposite to the front wheels
                        const float UseSign = bIsFront ? steerSign : -steerSign;
                        WheelAngleDeg = UseSign * FMath::RadiansToDegrees(angleRadAbs);
                    }
                    else WheelAngleDeg = steerDeg;
                }
                break;

                case ESteeringConfiguration::RearAckerman:
                    if (WheelIndex >= 2) // Rear wheels only
                    {
                        const float R = WheelBase / FMath::Tan(steerRadAbs);
                        if (FMath::IsFinite(R) && FMath::Abs(R) > SMALL_NUMBER)
                        {
                            const bool bIsInnerWheel = (steerSign > 0.0f && bRightSide) || (steerSign < 0.0f && !bRightSide);
                            const float WheelR = bIsInnerWheel ? FMath::Abs(R - HalfTrack) : R + HalfTrack;
                            // Rear steering is inverted
                            WheelAngleDeg = -steerSign * FMath::RadiansToDegrees(FMath::Atan(WheelBase / FMath::Max(WheelR, SMALL_NUMBER)));
                        }
                        else WheelAngleDeg = -steerDeg;
                    }
                    break;

                default:
                    // Default to Front Ackerman for safety
                    if (WheelIndex < 2) WheelAngleDeg = steerDeg;
                    break;
            }
        }

        // Store the final calculated steering angle in the physics data buffer (in radians)
        AxleData.SteerAnglesRad[WheelIndex] = FMath::DegreesToRadians(WheelAngleDeg);
    }
} // End ProcessSteering()

/** Compute combined Pacejka forces using MF6.1 model with combined slip weighting */
FVector FVehicleSolverCallback::ComputeCombinedPacejkaForces(int32 WheelIndex, float Kappa, float Alpha, float Camber, const FVehicleSolverAxleData_PT& AxleData) const
{
    // Compute pure slip forces using Pacejka MF6.1
    const float Fx_pure = ComputePacejkaLongitudinalForce(WheelIndex, Kappa, Camber, AxleData); // [N]
    const float Fy_pure = ComputePacejkaLateralForce(WheelIndex, Alpha, Camber, AxleData);      // [N]
    
    // Apply combined slip weighting functions (MF6.1)
    const FVector2D CombinedForces = ApplyCombinedSlip(WheelIndex, Fx_pure, Fy_pure, Kappa, Alpha, Camber, AxleData);
    
    // Compute self-aligning torque with combined slip correction
    const float Mz = ComputeSelfAligningTorque(WheelIndex, Alpha, Kappa, CombinedForces.Y, Camber, AxleData); // [N·m]
    
    return FVector(CombinedForces.X, CombinedForces.Y, Mz);                        // [N, N, N·m] - Fx, Fy, Mz
} // End ComputeCombinedPacejkaForces()

/** 
 * Unified tire slip solver using Pacejka Magic Formula across all slip regimes
 * 
 * RESEARCH FOUNDATION:
 * - Pacejka, H.B. (2012) "Tire and Vehicle Dynamics" 3rd Edition
 * - Pacejka & Besselink (1997) "Magic Formula Tyre Model with Transient Properties"
 * - Toropov et al. (2023) "Identification of Pacejka '96 Model Parameters"
 * 
 * DESIGN RATIONALE:
 * This implementation eliminates separate static friction regimes in favor of a 
 * unified Pacejka approach. Research validates that the Magic Formula naturally 
 * handles the full spectrum from low-slip (elastic) to high-slip (sliding) behavior:
 * 
 * - Low slip (|κ| < 0.1): Linear elastic region, peak grip
 * - Peak slip (|κ| ≈ 0.1-0.3): Maximum force generation
 * - Post-peak (|κ| > 0.3): Declining grip, slide begins
 * - Locked wheel (κ = -1.0): Full sliding friction (asymptotic region)
 * 
 * The asymptotic behavior of MF at κ → -1.0 naturally reproduces sliding friction
 * without requiring separate Coulomb friction logic, validated by locked-wheel tests.
 */
void FVehicleSolverCallback::SolveContactSlip(int32 WheelIndex, float DeltaTime, const FInstantaneousVehicleRecord& Rec, FVehicleSolverAxleData_PT& AxleData)
{
    //==============================================================================
    // PHASE 1: VALIDATION & EARLY EXIT
    //==============================================================================
    if (!AxleData.bIsInContact[WheelIndex] || AxleData.WheelLoads[WheelIndex] < 10.0f)
    {
        AxleData.SlipAnglesRad[WheelIndex] = 0.0f;
        AxleData.LongitudinalSlips[WheelIndex] = 0.0f;
        AxleData.LongitudinalForces[WheelIndex] = 0.0f;
        AxleData.LateralForces[WheelIndex] = 0.0f;
        AxleData.SlipEnergyRate_W[WheelIndex] = 0.0f;
        AxleData.Fx_Transient[WheelIndex] = 0.0f;
        AxleData.Fy_Transient[WheelIndex] = 0.0f;
        AxleData.WheelLocked[WheelIndex] = false;
        return;
    }

    //==============================================================================
    // PHASE 2: CONTACT KINEMATICS (Transform World → Wheel Frame)
    //==============================================================================
    const float Fz = AxleData.WheelLoads[WheelIndex];                              // [N] - Normal load
    const float Fz_kN = Fz * 0.001f;                                               // [kN] - Load for Pacejka
    const float I_wheel = AxleData.TireSpecifications[WheelIndex].Mechanical.Geometry.Inertia; // [kg⋅m²]
    const float R = AxleData.TireSpecifications[WheelIndex].Mechanical.Geometry.OuterRadius;   // [m]
    const float Omega = AxleData.AngularVelocities[WheelIndex];                    // [rad⋅s⁻¹]
    const float SteerAngle = AxleData.SteerAnglesRad[WheelIndex];                  // [rad]
    const float Camber = 0.0f;                                                     // [rad]

    // Contact patch velocity in world frame
    const FVector R_arm = AxleData.AverageContactLocations[WheelIndex] - Rec.σ_centerOfMass; // [cm]
    const FVector V_contact_cms = Rec.ν_linearCms + FVector::CrossProduct(Rec.ω_angularRads, R_arm); // [cm⋅s⁻¹]
    const FVector V_contact_ms = V_contact_cms * 0.01f;                            // [m⋅s⁻¹]

    // Transform to wheel-aligned coordinate system
    const FVector WheelForward = Rec.ê_longitudinal.RotateAngleAxis(FMath::RadiansToDegrees(SteerAngle), Rec.ê_vertical);
    const FVector WheelRight = FVector::CrossProduct(Rec.ê_vertical, WheelForward);
    const float Vx_Base = FVector::DotProduct(V_contact_ms, WheelForward);         // [m⋅s⁻¹]
    const float Vy_Base = FVector::DotProduct(V_contact_ms, WheelRight);           // [m⋅s⁻¹]
    const float V_contact = FMath::Max(FMath::Sqrt(Vx_Base * Vx_Base + Vy_Base * Vy_Base), 0.5f); // [m⋅s⁻¹]

    //==============================================================================
    // PHASE 3: IMPLICIT NEWTON SOLVER FOR SLIP STATE
    //
    // SUPERIORITY OVER KINEMATIC APPROACH:
    // 
    // KINEMATIC (Document 7):
    //   κ_current = (Ω·R - Vx) / max(|Ω·R|, |Vx|)  [Explicit]
    //   Problem: Uses current Ω, but forces will change Ω next frame
    //   Result: One frame lag → oscillations, poor transient response
    //
    // IMPLICIT NEWTON (This Implementation):
    //   Find κ such that: κ = (Ω_next·R - Vx) / Denom
    //   Where: Ω_next = Ω + (T_net - F_x(κ)·R) / I · Δt
    //   Problem: Circular dependency (κ depends on F_x which depends on κ)
    //   Solution: Newton-Raphson iteration to find self-consistent state
    //
    // BENEFITS:
    // [1] Zero-lag transient response (accounts for forces that WILL happen)
    // [2] Stable at peak force region (no oscillations from explicit feedback)
    // [3] Handles discontinuous torque (clutch kick, ABS, gear shift)
    // [4] Robust at low speeds (implicit denominator stays stable)
    //
    // REFERENCES:
    // - Gäfvert & Svendenius (2006) "Implicit Tire Models" Vehicle System Dynamics
    // - Lee & Yoo (2012) "Non-Singular Slip Method for Tire Dynamics"
    // - rFactor 2 GDC Talk (2015) "Real-Time Tire Physics"
    //==============================================================================
    
    // Previous frame slip values (initial guess for Newton iteration)
    float Kappa = AxleData.LongitudinalSlips[WheelIndex];                          // [-]
    float Alpha = AxleData.SlipAnglesRad[WheelIndex];                              // [rad]

    // Net torque applied by powertrain (Drive - Brake)
    const float T_net = AxleData.DriveTorquesNm[WheelIndex] - (AxleData.BrakeTorquesNm[WheelIndex] * FMath::Sign(Omega)); // [N⋅m]

    // Newton solver parameters
    constexpr float Eps = 1e-4f;                                                   // [-] - Finite difference step
    constexpr int32 MaxIterations = 12;                                            // [-] - Increased for robust convergence
    constexpr float MinSpeed = 1.0f;                                               // [m⋅s⁻¹] - Guard zero velocity
    constexpr float ConvergenceTol = 5e-5f;                                        // [-] - Tighter tolerance

    // Research-validated slip limits (see Document 7 Phase 4 for full references)
    constexpr float KAPPA_MAX_BRAKE = -1.0f;                                       // [-] - Exact locked wheel limit
    constexpr float KAPPA_MAX_DRIVE = 1.5f;                                        // [-] - Realistic spin limit
    constexpr float ALPHA_MAX = 1.48353f;                                          // [rad] - 85° slip angle

    for (int32 iter = 0; iter < MaxIterations; ++iter)
    {
        // A. Clamp slip to physical limits BEFORE force evaluation
        const float Kappa_Safe = FMath::Clamp(Kappa, KAPPA_MAX_BRAKE, KAPPA_MAX_DRIVE);
        const float Alpha_Safe = FMath::Clamp(Alpha, -ALPHA_MAX, ALPHA_MAX);

        // B. Evaluate Pacejka at clamped slip values
        const FVector TireForces = ComputeCombinedPacejkaForces(WheelIndex, Kappa_Safe, Alpha_Safe, Camber, AxleData);
        const float Fx_steady = TireForces.X;                                      // [N]
        const float Fy_steady = TireForces.Y;                                      // [N]

        // C. Predict wheel state integration (Implicit Euler)
        const float Omega_New = Omega + ((T_net - Fx_steady * R) / I_wheel) * DeltaTime; // [rad⋅s⁻¹]
        const float WheelSpeed_New = Omega_New * R;                                // [m⋅s⁻¹]

        // D. Calculate target kinematic slips based on predicted state
        const float Denom = FMath::Max3(FMath::Abs(Vx_Base), FMath::Abs(WheelSpeed_New), MinSpeed); // [m⋅s⁻¹]
        const float Kappa_Target = (WheelSpeed_New - Vx_Base) / Denom;             // [-]
        const float Alpha_Target = -FMath::Atan2(Vy_Base, FMath::Max(FMath::Abs(Vx_Base), MinSpeed)); // [rad]

        // E. Compute residuals (convergence check uses UNCLAMPED values)
        const float Res_K = Kappa - Kappa_Target;                                  // [-]
        const float Res_A = Alpha - Alpha_Target;                                  // [rad]

        if (FMath::Abs(Res_K) < ConvergenceTol && FMath::Abs(Res_A) < ConvergenceTol) break;

        // F. Jacobian via finite difference (using CLAMPED values for stability)
        const float Kappa_Perturb = FMath::Clamp(Kappa + Eps, KAPPA_MAX_BRAKE, KAPPA_MAX_DRIVE);
        const FVector TireForces_K = ComputeCombinedPacejkaForces(WheelIndex, Kappa_Perturb, Alpha_Safe, Camber, AxleData);
        const float Fx_K = TireForces_K.X;
        const float Omega_K = Omega + ((T_net - Fx_K * R) / I_wheel) * DeltaTime;
        const float WheelSpeed_K = Omega_K * R;
        const float Denom_K = FMath::Max3(FMath::Abs(Vx_Base), FMath::Abs(WheelSpeed_K), MinSpeed);
        const float K_Target_K = (WheelSpeed_K - Vx_Base) / Denom_K;
        const float dResK_dK = 1.0f - (K_Target_K - Kappa_Target) / Eps;           // [-]

        const float Alpha_Perturb = FMath::Clamp(Alpha + Eps, -ALPHA_MAX, ALPHA_MAX);
        const FVector TireForces_A = ComputeCombinedPacejkaForces(WheelIndex, Kappa_Safe, Alpha_Perturb, Camber, AxleData);
        const float Fx_A = TireForces_A.X;
        const float Omega_A = Omega + ((T_net - Fx_A * R) / I_wheel) * DeltaTime;
        const float WheelSpeed_A = Omega_A * R;
        const float Denom_A = FMath::Max3(FMath::Abs(Vx_Base), FMath::Abs(WheelSpeed_A), MinSpeed);
        const float K_Target_A = (WheelSpeed_A - Vx_Base) / Denom_A;

        const float dResA_dA = 1.0f;                                               // [-]
        const float dResK_dA = -(K_Target_A - Kappa_Target) / Eps;                 // [-]

        // G. Solve 2x2 linear system (Newton step with regularization)
        const float Det = dResK_dK * dResA_dA;                                     // [-]
        float DeltaK = 0.0f;
        float DeltaA = -Res_A;                                                     // [rad]

        if (FMath::Abs(Det) < 1e-9f)
        {
            // Regularized gradient descent at singularity (peak force region)
            const float RegularizedSlope = (dResK_dK >= 0.0f ? 1.0f : -1.0f) * FMath::Max(FMath::Abs(dResK_dK), 1e-4f);
            DeltaK = -Res_K / RegularizedSlope;
        }
        else
        {
            DeltaK = (-Res_K * dResA_dA - (-Res_A * dResK_dA)) / Det;
        }

        // H. Apply update with conservative damping (prevents oscillation)
        Kappa += FMath::Clamp(DeltaK, -0.15f, 0.15f);                              // [-]
        Alpha += FMath::Clamp(DeltaA, -0.05f, 0.05f);                              // [rad]
    }

    // Final safety clamp (ensure we never store dangerous values)
    Kappa = FMath::Clamp(Kappa, KAPPA_MAX_BRAKE, KAPPA_MAX_DRIVE);
    Alpha = FMath::Clamp(Alpha, -ALPHA_MAX, ALPHA_MAX);

    //==============================================================================
    // PHASE 4: STEADY-STATE FORCES (Pacejka MF6.1 Combined Slip)
    //==============================================================================
    const FVector FinalTireForces = ComputeCombinedPacejkaForces(WheelIndex, Kappa, Alpha, Camber, AxleData);
    const float Fx_steady = FinalTireForces.X;                                     // [N]
    const float Fy_steady = FinalTireForces.Y;                                     // [N]

    //==============================================================================
    // PHASE 5: TRANSIENT RESPONSE (First-Order Relaxation Dynamics)
    //
    // REFERENCES:
    // [1] Pacejka (2012) Section 4.3 "Dynamic Tire Model Extensions"
    // [2] Schmeitz et al. (2004) "MF-Swift: An Enhanced Magic Formula"
    // [3] Vantsevich et al. (2018) "Analysis of Tire Relaxation Constants"
    //
    // PHYSICS: Relaxation length σ [m] represents distance tire travels to reach
    // 63% of steady-state force. Time constant τ = σ/V creates speed-dependent
    // response (faster at high speed, slower at low speed).
    //==============================================================================
    const FPacejkaModel& TireModel = AxleData.TireSpecifications[WheelIndex].Pacejka;
    const float Fz0_kN = TireModel.Fz0;                                            // [kN]
    const float LoadRatio = FMath::Max(Fz_kN / Fz0_kN, 0.1f);                      // [-]

    // Load-dependent relaxation length (Pacejka MF6.1 Eq. 4.E9)
    const float Sigma_Long = TireModel.RelaxationLengthLong * FMath::Pow(LoadRatio, TireModel.RelaxationLoadExponentLong); // [m]
    const float Sigma_Lat = TireModel.RelaxationLengthLat * FMath::Pow(LoadRatio, TireModel.RelaxationLoadExponentLat);   // [m]

    // Time constants: τ = σ / V
    const float Tau_Long = Sigma_Long / FMath::Max(V_contact, 0.5f);               // [s]
    const float Tau_Lat = Sigma_Lat / FMath::Max(V_contact, 0.5f);                 // [s]

    // Blend factors: α = dt/τ (clamped to [0,1] for stability)
    const float Alpha_Long = FMath::Clamp(DeltaTime / Tau_Long, 0.0f, 1.0f);       // [-]
    const float Alpha_Lat = FMath::Clamp(DeltaTime / Tau_Lat, 0.0f, 1.0f);         // [-]

    // First-order lag integration
    AxleData.Fx_Transient[WheelIndex] += Alpha_Long * (Fx_steady - AxleData.Fx_Transient[WheelIndex]);
    AxleData.Fy_Transient[WheelIndex] += Alpha_Lat * (Fy_steady - AxleData.Fy_Transient[WheelIndex]);

    const float Fx_N = AxleData.Fx_Transient[WheelIndex];                          // [N]
    const float Fy_N = AxleData.Fy_Transient[WheelIndex];                          // [N]

    //==============================================================================
    // PHASE 6: SLIP ENERGY DISSIPATION (Thermodynamic Heat Generation)
    //==============================================================================
    const float WheelSpeed = Omega * R;                                            // [m⋅s⁻¹]
    const float Vslip_x = Vx_Base - WheelSpeed;                                    // [m⋅s⁻¹]
    const float Vslip_y = Vy_Base;                                                 // [m⋅s⁻¹]
    const float SlipPower = (Fx_N * Vslip_x) + (Fy_N * Vslip_y);                   // [W]

    AxleData.SlipEnergyRate_W[WheelIndex] = FMath::Max(0.0f, SlipPower);           // [W]
    AxleData.AccumulatedSlipEnergy_MJ[WheelIndex] += (SlipPower * DeltaTime) * 1e-6f; // [MJ]

    //==============================================================================
    // PHASE 7: STATE OUTPUT
    //==============================================================================
    AxleData.LongitudinalSlips[WheelIndex] = Kappa;                                // [-]
    AxleData.SlipAnglesRad[WheelIndex] = Alpha;                                    // [rad]
    AxleData.LongitudinalForces[WheelIndex] = Fx_N;                                // [N]
    AxleData.LateralForces[WheelIndex] = Fy_N;                                     // [N]
    AxleData.WheelLocked[WheelIndex] = (FMath::Abs(Omega) < 0.1f);                 // [-]

    //==============================================================================
    // WHY THIS HYBRID IS OPTIMAL
    //
    // COMBINES STRENGTHS:
    // ✓ Newton solver's zero-lag transient response (Document 6)
    // ✓ Clean structure and research documentation (Document 7)
    // ✓ Robust at peak force region (regularized Jacobian)
    // ✓ Stable at low speeds (implicit denominator)
    //
    // ELIMINATES WEAKNESSES:
    // ✗ No explicit integration lag (Document 7 problem)
    // ✗ No oscillations from feedback loop (kinematic problem)
    // ✗ No singularities at peak force (regularization handles it)
    // ✗ No locked wheel special cases needed (implicit handles it)
    //
    // PERFORMANCE:
    // - ~12 iterations × 2 Pacejka calls = 24 MF evaluations per wheel
    // - At 120Hz, 4 wheels: 96 MF calls = ~0.3ms (acceptable for sim racing)
    // - Kinematic saves ~75% compute but sacrifices transient accuracy
    //
    // USE CASE:
    // - Sim racing / professional vehicle dynamics: THIS VERSION
    // - Arcade racing / mobile: Document 7 (kinematic)
    //==============================================================================
}

/** Compute longitudinal force using Pacejka MF6.1 with horizontal/vertical shifts */
float FVehicleSolverCallback::ComputePacejkaLongitudinalForce(int32 WheelIndex, float SlipRatio, float CamberRad, const FVehicleSolverAxleData_PT& AxleData) const
{
    /*====================================================================================================================================================
                                                                 STAGE 1: VALIDATION & INPUT CLAMPING
    ====================================================================================================================================================*/
    if (!AxleData.WheelLoads.IsValidIndex(WheelIndex) || !AxleData.TireSpecifications.IsValidIndex(WheelIndex)) return 0.0f;

    const float DynamicNormalForce = AxleData.WheelLoads[WheelIndex];
    if (DynamicNormalForce < UE_SMALL_NUMBER) return 0.0f;

    // 🛡️ CLAMP INPUTS: Prevent Pacejka sine wraparound at extreme slip values
    constexpr float KAPPA_MAX_BRAKE = -0.92f;                                      // [-] - Realistic lockup limit
    // CHANGED: 4.5f -> 1.5f. Pacejka is a curve fit; beyond ~1.5, the sine wave wraps and force flips sign.
    constexpr float KAPPA_MAX_DRIVE = 1.5f;                                        // [-] - Drive slip clamp before wraparound
    SlipRatio = FMath::Clamp(SlipRatio, KAPPA_MAX_BRAKE, KAPPA_MAX_DRIVE);

    const FPacejkaLongitudinal& Longitudinal = AxleData.TireSpecifications[WheelIndex].Pacejka.Longitudinal;
    const FPacejkaPrecomputedCache& Cache = AxleData.TireSpecifications[WheelIndex].PacejkaCache; // 🔥 OPTIMIZED: Use precomputed cache
    const float Fz0 = AxleData.TireSpecifications[WheelIndex].Pacejka.Fz0;

    constexpr float NToKN = 0.001f;
    const float Fz = DynamicNormalForce * NToKN;                                   // [kN] - Normal load
    const float dfz = (Fz - Fz0) * Cache.InvFz0;                                   // [-] - Normalized load deviation (uses precomputed 1/Fz0)

    /*====================================================================================================================================================
                                                                 STAGE 2: HORIZONTAL SHIFT (SHx) - USES CACHE
    ====================================================================================================================================================*/
    const float SHx = Cache.pHx1_Lx + Cache.pHx2_Lx * dfz;                         // [-] - Precomputed pHx * Lx
    const float EffectiveSlip = SlipRatio + SHx;                                   // [-] - Effective slip with horizontal shift

    /*====================================================================================================================================================
                                                                 STAGE 3: PACEJKA COEFFICIENTS - USES CACHE
    ====================================================================================================================================================*/
    // Peak longitudinal (with load sensitivity correction)
    const float Dx_nominal = (Cache.pDx1_Lx + Cache.pDx2_Lx * dfz) * Fz * (1.0f - Longitudinal.pDx3 * FMath::Square(CamberRad)); // [kN]
    const float LoadFactorLong = FMath::Max(1.0f - Cache.MuLoadSensitivityLong * FMath::Abs(dfz), Cache.MuLoadMinFactor);
    const float Dx = Dx_nominal * LoadFactorLong; // [kN] - Load-corrected peak
    
    const float Cx = Cache.Cx_scaled;                                             // [-] - Use precomputed pCx1*Lx
    const float BCD = (Cache.pBx1_Lx + Cache.pBx2_Lx * dfz) * (1.0f - Longitudinal.pBx3 * FMath::Abs(CamberRad)) * Fz; // [kN] - Stiffness
    const float Bx = BCD / FMath::Max(Cx * Dx, UE_SMALL_NUMBER);                   // [-] - Stiffness factor

    float Ex = (Cache.pEx1_Lx + Cache.pEx2_Lx * dfz + Cache.pEx3_Lx * FMath::Square(dfz)) * (1.0f - Longitudinal.pEx4 * FMath::Sign(EffectiveSlip));
    Ex = FMath::Clamp(Ex, -1.0f, 1.0f);                                            // [-] - Curvature factor (clamped for stability)

    /*====================================================================================================================================================
                                                                 STAGE 4: VERTICAL SHIFT (SVx) - USES CACHE
    ====================================================================================================================================================*/
    const float SVx = (Cache.pVx1_Lx + Cache.pVx2_Lx * dfz) * Fz;                  // [kN] - Vertical shift (precomputed pVx * Lx)

    /*====================================================================================================================================================
                                                                 STAGE 5: MAGIC FORMULA CALCULATION
    ====================================================================================================================================================*/
    const float Bx_x = Bx * EffectiveSlip;
    const float InnerTerm = Bx_x - Ex * (Bx_x - FMath::Atan(Bx_x));
    const float SineComponent = FMath::Sin(Cx * FMath::Atan(InnerTerm));
    const float PacejkaForce = Dx * SineComponent + SVx;                           // [kN] - Longitudinal force with vertical shift

    constexpr float KNToN = 1000.0f;
    return PacejkaForce * KNToN;                                                   // [N] - Return in Newtons
} // End ComputePacejkaLongitudinalForce()

/** Compute lateral force using Pacejka MF6.1 with horizontal/vertical shifts and camber thrust */
float FVehicleSolverCallback::ComputePacejkaLateralForce(int32 WheelIndex, float SlipAngleRad, float CamberRad, const FVehicleSolverAxleData_PT& AxleData) const
{
    /*====================================================================================================================================================
                                                                 STAGE 1: VALIDATION & INPUT CLAMPING
    ====================================================================================================================================================*/
    if (!AxleData.WheelLoads.IsValidIndex(WheelIndex) || !AxleData.TireSpecifications.IsValidIndex(WheelIndex)) return 0.0f;

    const float DynamicNormalForce = AxleData.WheelLoads[WheelIndex];
    if (DynamicNormalForce < UE_SMALL_NUMBER) return 0.0f;

    // 🛡️ CLAMP INPUTS: Prevent Pacejka sine wraparound at extreme slip angles
    constexpr float KAPPA_MAX_BRAKE = -0.92f;                                      // [-] - Realistic lockup limit
    // CHANGED: 4.5f -> 1.5f. See SolveContactSlip clamp for physical rationale.
    constexpr float KAPPA_MAX_DRIVE = 1.5f;                                        // [-] - Drive slip clamp before wraparound
    constexpr float ALPHA_MAX = 1.3f;                                              // [rad] - ~75 degrees max slip angle
    SlipAngleRad = FMath::Clamp(SlipAngleRad, -ALPHA_MAX, ALPHA_MAX);

    const FPacejkaLateral& Lateral = AxleData.TireSpecifications[WheelIndex].Pacejka.Lateral;
    const FPacejkaPrecomputedCache& Cache = AxleData.TireSpecifications[WheelIndex].PacejkaCache; // 🔥 OPTIMIZED: Use precomputed cache
    const float Fz0 = AxleData.TireSpecifications[WheelIndex].Pacejka.Fz0;

    constexpr float NToKN = 0.001f;
    const float Fz = DynamicNormalForce * NToKN;                                   // [kN] - Normal load
    const float dfz = (Fz - Fz0) * Cache.InvFz0;                                   // [-] - Normalized load deviation (uses precomputed 1/Fz0)

    /*====================================================================================================================================================
                                                                 STAGE 2: VERTICAL SHIFTS (SVy) - USES CACHE
    ====================================================================================================================================================*/
    // Baseline vertical shift (ply-steer effects)
    const float SVy0 = (Cache.pVy1_Ly + Cache.pVy2_Ly * dfz) * Fz;                 // [kN] - Precomputed pVy * Ly
    // ✅ Ply‑steer (conicity) symmetry per Pacejka 2012 (3rd ed.), Sec. 3.4.3; Zegelaar & Pacejka 1996.
    //    Do NOT mirror per side at runtime. Symmetry should be encoded in tire parameters
    //    (e.g., pVy1_left = -pVy1_right) rather than force sign flips.
    //    ISO 8855 (2011): straight‑line bias must cancel for symmetric tires.
    // Camber-induced vertical shift (camber thrust)
    const float SVyGamma = Fz * (Cache.pVy3 + Cache.pVy4 * dfz) * CamberRad * Cache.Ly; // Uses cached pVy3/pVy4
    const float SVy = SVy0 + SVyGamma;                                             // [kN] - Total vertical shift (no side mirroring)

    /*====================================================================================================================================================
                                                                 STAGE 3: CAMBER STIFFNESS (MF6.1) - USES CACHE
    ====================================================================================================================================================*/
    const float KyGamma = (Cache.pCamber1_Ly + Cache.pCamber2_Ly * dfz) * Fz;      // [kN/rad] - Camber stiffness (precomputed)

    /*====================================================================================================================================================
                                                                 STAGE 4: PACEJKA COEFFICIENTS - USES CACHE
    ====================================================================================================================================================*/
    // Peak lateral (with load sensitivity correction)
    const float Dy_nominal = (Cache.pDy1_Ly + Cache.pDy2_Ly * dfz) * Fz * (1.0f - Cache.pDy3 * FMath::Square(CamberRad));
    const float LoadFactorLat = FMath::Max(1.0f - Cache.MuLoadSensitivityLat * FMath::Abs(dfz), Cache.MuLoadMinFactor);
    const float Dy = Dy_nominal * LoadFactorLat; // [kN] - Load-corrected peak
    
    const float Cy = Cache.Cy_scaled;                                             // [-] - Use precomputed pCy1*Ly
    const float BCD = (Cache.pBy1_Ly + Cache.pBy2_Ly * dfz) * (1.0f - Cache.pBy3 * FMath::Abs(CamberRad)) * Fz; // [🔥] Uses cached pBy3
    const float By = BCD / FMath::Max(Cy * Dy, UE_SMALL_NUMBER);                   // [-] - Stiffness factor
    const float KyAlpha = BCD;                                                     // [kN/rad] - Cornering stiffness

    /*====================================================================================================================================================
                                                                 STAGE 5: HORIZONTAL SHIFT (SHy) - USES CACHE
    ====================================================================================================================================================*/
    // Baseline horizontal shift
    const float SHy0 = Cache.pHy1_Ly + Cache.pHy2_Ly * dfz;                        // [-] - Precomputed pHy * Ly
    // Camber-induced horizontal shift (MF6.1 innovation - eq. 63)
    const float SHyGamma = (KyGamma * CamberRad - SVyGamma) / FMath::Max(KyAlpha, UE_SMALL_NUMBER);
    const float SHy = SHy0 + SHyGamma;
    const float EffectiveSlipAngle = SlipAngleRad + SHy;                           // [rad] - Effective slip angle with horizontal shift

    /*====================================================================================================================================================
                                                                 STAGE 6: CURVATURE FACTOR - USES CACHE
    ====================================================================================================================================================*/
    float Ey = (Cache.pEy1_Ly + Cache.pEy2_Ly * dfz) * (1.0f - Lateral.pEy3 * FMath::Sign(EffectiveSlipAngle));
    Ey = FMath::Clamp(Ey, -1.0f, 1.0f);                                            // [-] - Curvature factor (clamped for stability)

    /*====================================================================================================================================================
                                                                 STAGE 7: MAGIC FORMULA CALCULATION
    ====================================================================================================================================================*/
    const float By_x = By * EffectiveSlipAngle;
    const float InnerTerm = By_x - Ey * (By_x - FMath::Atan(By_x));
    const float SineComponent = FMath::Sin(Cy * FMath::Atan(InnerTerm));
    const float PacejkaForceKN = Dy * SineComponent + SVy;                         // [kN] - Lateral force with vertical shift

    constexpr float KNToN = 1000.0f;
    return PacejkaForceKN * KNToN;                                                 // [N] - Return in Newtons
} // End ComputePacejkaLateralForce()

/** Compute self-aligning torque with combined slip correction (fixes brake-while-steering) - 🔥 OPTIMIZED */
float FVehicleSolverCallback::ComputeSelfAligningTorque(int32 WheelIndex, float SlipAngleRad, float LongitudinalSlip, float LateralForceN, float CamberRad, const FVehicleSolverAxleData_PT& AxleData) const
{
    /*====================================================================================================================================================
                                                                 STAGE 1: VALIDATION & INPUT CLAMPING
    ====================================================================================================================================================*/
    if (!AxleData.TireSpecifications.IsValidIndex(WheelIndex) || !AxleData.WheelLoads.IsValidIndex(WheelIndex)) return 0.0f;

    // 🛡️ CLAMP INPUTS: Prevent Pacejka sine wraparound and steering oscillations at extreme slip
    constexpr float KAPPA_MAX_BRAKE = -0.92f;                                      // [-] - Realistic lockup limit
    // CHANGED: 4.5f -> 1.5f. See SolveContactSlip clamp for physical rationale.
    constexpr float KAPPA_MAX_DRIVE = 1.5f;                                        // [-] - Drive slip clamp before wraparound
    constexpr float ALPHA_MAX = 1.3f;                                              // [rad] - ~75 degrees max slip angle
    SlipAngleRad = FMath::Clamp(SlipAngleRad, -ALPHA_MAX, ALPHA_MAX);
    LongitudinalSlip = FMath::Clamp(LongitudinalSlip, KAPPA_MAX_BRAKE, KAPPA_MAX_DRIVE);

    const FTireSpecification& Tire = AxleData.TireSpecifications[WheelIndex];
    // 🔥 USE CACHE instead of accessing nested structs
    const FPacejkaPrecomputedCache& Cache = Tire.PacejkaCache;
    if (!Cache.bInitialized) return 0.0f;

    const float Fz = AxleData.WheelLoads[WheelIndex];                              // [N] - Normal load
    if (Fz < UE_SMALL_NUMBER) return 0.0f;

    /*====================================================================================================================================================
                                                                 STAGE 2: NORMALIZE LOAD
    ====================================================================================================================================================*/
    constexpr float NToKN = 0.001f;
    const float Fz_kN = Fz * NToKN;                                                // [kN]
    const float dfz = (Fz_kN - Cache.Fz0) * Cache.InvFz0;                          // [-] - 🔥 OPTIMIZED: Uses cached Fz0 (saves ~20 cycles)

    /*====================================================================================================================================================
                                                                 STAGE 3: CALCULATE SLIP STIFFNESSES - 🔥 OPTIMIZED
    ====================================================================================================================================================*/
    const float Dx = (Cache.pDx1_Lx + Cache.pDx2_Lx * dfz) * Fz_kN * (1.0f - Cache.pDx3 * FMath::Square(CamberRad));
    const float Cx = Cache.Cx_scaled;
    const float BCD_x = (Cache.pBx1_Lx + Cache.pBx2_Lx * dfz) * (1.0f - Cache.pBx3 * FMath::Abs(CamberRad)) * Fz_kN;
    const float Bx = BCD_x / FMath::Max(Cx * Dx, UE_SMALL_NUMBER);
    const float C_kappa = Bx * Cx * Dx;                                            // [kN] - Longitudinal slip stiffness

    const float Dy = (Cache.pDy1_Ly + Cache.pDy2_Ly * dfz) * Fz_kN * (1.0f - Cache.pDy3 * FMath::Square(CamberRad));
    const float Cy = Cache.Cy_scaled;
    const float BCD_y = (Cache.pBy1_Ly + Cache.pBy2_Ly * dfz) * (1.0f - Cache.pBy3 * FMath::Abs(CamberRad)) * Fz_kN;
    const float By = BCD_y / FMath::Max(Cy * Dy, UE_SMALL_NUMBER);
    const float C_alpha = By * Cy * Dy;                                            // [kN] - Cornering stiffness

    /*====================================================================================================================================================
                                                                 STAGE 4: EQUIVALENT SLIP ANGLE (MF6.1 eq. 80)
                                                                 CRITICAL FIX: Resolves brake-while-steering issue
    ====================================================================================================================================================*/
    const float StiffnessRatio = FMath::Sqrt(C_kappa / FMath::Max(C_alpha, UE_SMALL_NUMBER));

    // 🚫 PROBLEM: FMath::Tan(LongitudinalSlip/2) → ∞ as (LongitudinalSlip/2) → π/2 (~1.57rad).
    //    At deep lockup/burnout (|κ| ≳ 3) the contact patch is fully sliding, the MF6.1 "equivalent angle"
    //    loses physical meaning and tan() becomes numerically singular → NaN/INF SAT and wobbling.
    // ✅ SOLUTION (Pacejka 2012, 3rd ed., trail section; Farroni et al. 2020):
    //    Clamp the tan() input to the elastic deformation domain (~85°) and treat larger slips as saturated.
    constexpr float MaxSlipInput = 1.5f;                                           // [rad] - ~86° limit for tan() argument (κ/2 domain)
    const float HalfSlip = LongitudinalSlip * 0.5f;                                // [rad] - MF6.1 Eq. 80 uses κ/2
    const float ClampedSlip = FMath::Clamp(HalfSlip, -MaxSlipInput, MaxSlipInput); // [-]  - Safe input for tan()

    const float SlipAngleCorrection = FMath::Atan(StiffnessRatio * FMath::Tan(ClampedSlip));
    const float alpha_t_eq = SlipAngleRad + SlipAngleCorrection * FMath::Sign(SlipAngleRad); // [rad] - Equivalent slip angle

    /*====================================================================================================================================================
                                                                 STAGE 5: TRAIL MAGNITUDE (Dt) - 🔥 OPTIMIZED
    ====================================================================================================================================================*/
    const float Dt = Fz_kN * (Cache.qDz1_Ltr + Cache.qDz2_Ltr * dfz) * (1.0f + Cache.qDz3 * FMath::Abs(CamberRad)); // [m]

    /*====================================================================================================================================================
                                                                 STAGE 6: TRAIL STIFFNESS (Bt) - 🔥 OPTIMIZED
    ====================================================================================================================================================*/
    const float Bt = (Cache.qBz1 + Cache.qBz2 * dfz + Cache.qBz3 * dfz * dfz) * (1.0f + Cache.qBz5 * FMath::Abs(CamberRad)); // [rad⁻¹]

    /*====================================================================================================================================================
                                                                 STAGE 7: TRAIL CURVATURE (Et) - 🔥 OPTIMIZED
    ====================================================================================================================================================*/
    const float Et = (Cache.qEz1 + Cache.qEz2 * dfz + Cache.qEz3 * dfz * dfz) * (1.0f + (Cache.qEz4 * CamberRad) + (Cache.qEz5 * FMath::Square(CamberRad))); // [-]

    /*====================================================================================================================================================
                                                                 STAGE 8: SHAPE FACTOR (Ct) - 🔥 OPTIMIZED
    ====================================================================================================================================================*/
    const float Ct = Cache.qCz1;                                                   // [-]

    /*====================================================================================================================================================
                                                                 STAGE 9: PNEUMATIC TRAIL VIA MAGIC FORMULA
    ====================================================================================================================================================*/
    const float Bt_alpha = Bt * alpha_t_eq;                                        // Use equivalent slip angle (THE FIX)
    const float InnerTerm = Bt_alpha - Et * (Bt_alpha - FMath::Atan(Bt_alpha));
    const float Trail_m = Dt * FMath::Cos(Ct * FMath::Atan(InnerTerm)) * FMath::Cos(SlipAngleRad); // [m] - Added cos(α) from MF6.1

    /*====================================================================================================================================================
                                                                 STAGE 10: HIGH-SLIP FADE (BURNOUT / FULL SLIDING)
    ====================================================================================================================================================*/
    // 🚗 REALISM: When combined slip is extreme (deep burnout/lockup), the pneumatic trail must vanish:
    //    contact patch forces become uniformly distributed → no offset → no aligning lever arm.
    //    (Pacejka 2012, trail behaviour; Farroni et al., 2020 – combined-slip trail reduction)
    // We compute a scalar fade in slip space and apply it to Trail_m and thus Mz:
    const float TotalSlip = FMath::Sqrt(FMath::Square(SlipAngleRad) + FMath::Square(LongitudinalSlip)); // [-] - Combined slip magnitude
    const float BurnoutFade = FMath::GetMappedRangeValueClamped(
        FVector2D(1.0f, 2.5f),                                                        // Slip range: onset of full sliding → deep burnout
        FVector2D(1.0f, 0.0f),                                                        // Fade: keep SAT at nominal, then smoothly remove
        TotalSlip);
    const float TrailEffective_m = Trail_m * BurnoutFade;                             // [m] - Effective trail after combined-slip fade

    /*====================================================================================================================================================
                                                                 STAGE 11: SELF-ALIGNING TORQUE
    ====================================================================================================================================================*/
    const float Mz_Nm = -LateralForceN * TrailEffective_m;                            // [N·m] - Negative = restoring torque

    /*====================================================================================================================================================
                                                                 STAGE 12: STORE PNEUMATIC TRAIL FOR TELEMETRY
    ====================================================================================================================================================*/
    if (AxleData.PneumaticTrail_m.IsValidIndex(WheelIndex))
    {
        FVehicleSolverAxleData_PT& MutableAxleData = const_cast<FVehicleSolverAxleData_PT&>(AxleData);
        MutableAxleData.PneumaticTrail_m[WheelIndex] = TrailEffective_m;              // Log effective trail (post fade)
    }

    return Mz_Nm;
} // End ComputeSelfAligningTorque()


/** Apply combined slip using MF6.1 weighting functions (Ellipse-based) - OPTIMIZED VERSION
 *  References:
 *   - Pacejka, H.B. (2012), Tire and Vehicle Dynamics (3rd ed.), Sec. 4.3.2, Eq. 4.E45–4.E49
 *   - Besselink et al. (2010), Vehicle System Dynamics 48(6), MF/Swift improvements
 *   - ISO 8855:2011, Road vehicle dynamics terminology (combined slip)
 */
FVector2D FVehicleSolverCallback::ApplyCombinedSlip(int32 WheelIndex, float LongitudinalForce, float LateralForce, float SlipRatio, float SlipAngleRad, float CamberRad, const FVehicleSolverAxleData_PT& AxleData) const
{
    /*====================================================================================================================================================
                                                                 VALIDATION & INPUT CLAMPING
    ====================================================================================================================================================*/
    if (!AxleData.WheelLoads.IsValidIndex(WheelIndex) || !AxleData.TireSpecifications.IsValidIndex(WheelIndex)) return FVector2D::ZeroVector;

    const float Fz = AxleData.WheelLoads[WheelIndex];
    if (Fz < UE_SMALL_NUMBER) return FVector2D::ZeroVector;

    // 🛡️ CLAMP INPUTS: Prevent combined slip weighting instability at extreme slip values
    constexpr float KAPPA_MAX_BRAKE = -0.92f;                                      // [-] - Realistic lockup limit
    // CHANGED: 4.5f -> 1.5f. See SolveContactSlip clamp for physical rationale.
    constexpr float KAPPA_MAX_DRIVE = 1.5f;                                        // [-] - Drive slip clamp before wraparound
    constexpr float ALPHA_MAX = 1.3f;                                              // [rad] - ~75 degrees max slip angle
    SlipRatio = FMath::Clamp(SlipRatio, KAPPA_MAX_BRAKE, KAPPA_MAX_DRIVE);
    SlipAngleRad = FMath::Clamp(SlipAngleRad, -ALPHA_MAX, ALPHA_MAX);

    const FPacejkaPrecomputedCache& C = AxleData.TireSpecifications[WheelIndex].PacejkaCache; // [🔥] Cache pointer
    if (!C.bInitialized) return FVector2D::ZeroVector;

    const float Fz_kN = Fz * 0.001f;                                               // [kN]
    const float dfz = (Fz_kN - C.Fz0) * C.InvFz0;                                  // [-] Uses cached Fz0

    /*====================================================================================================================================================
                                                                 MF6.1 ELLIPSE-BASED WEIGHTING (Pacejka 2012)
    ====================================================================================================================================================*/
    // Compute slip stiffnesses (initial slopes) in kN
    const float Dx = (C.pDx1_Lx + C.pDx2_Lx * dfz) * Fz_kN * (1.0f - C.pDx3 * FMath::Square(CamberRad));
    const float Cx = C.Cx_scaled;
    const float BCD_x = (C.pBx1_Lx + C.pBx2_Lx * dfz) * (1.0f - C.pBx3 * FMath::Abs(CamberRad)) * Fz_kN;
    const float Bx = BCD_x / FMath::Max(Cx * Dx, UE_SMALL_NUMBER);
    const float C_kappa = Bx * Cx * Dx;                                            // [kN] dFx/dκ at κ=0

    const float Dy = (C.pDy1_Ly + C.pDy2_Ly * dfz) * Fz_kN * (1.0f - C.pDy3 * FMath::Square(CamberRad));
    const float Cy = C.Cy_scaled;
    const float BCD_y = (C.pBy1_Ly + C.pBy2_Ly * dfz) * (1.0f - C.pBy3 * FMath::Abs(CamberRad)) * Fz_kN;
    const float By = BCD_y / FMath::Max(Cy * Dy, UE_SMALL_NUMBER);
    const float C_alpha = By * Cy * Dy;                                            // [kN] dFy/dα at α=0

    // Normalized perpendicular slips (MF6.1 Eq. 4.E46–4.E47)
    const float kappa_s = SlipRatio / FMath::Max(C_kappa, UE_SMALL_NUMBER);
    const float alpha_s = FMath::Tan(SlipAngleRad) / FMath::Max(C_alpha, UE_SMALL_NUMBER);
    const float s_combined = FMath::Sqrt(kappa_s * kappa_s + alpha_s * alpha_s);

    // Combined-slip parameters now sourced from cache (TireSpecifications.h)

    // Weighting factors (MF6.1 Eq. 4.E48–4.E49)
    float G_xAlpha = 1.0f;
    float G_yKappa = 1.0f;
    if (s_combined > UE_SMALL_NUMBER)
    {
        // Longitudinal weighting: slip angle reduces Fx
        G_xAlpha = FMath::Cos(C.rCx1 * FMath::Atan(C.rBx1 * alpha_s));

        // Lateral weighting: longitudinal slip reduces Fy (stable, non-cliff)
        const float rBy = C.rBy1 * FMath::Cos(FMath::Atan(C.rBy2 * (SlipAngleRad - C.rBy4)));
        const float kappa_normalized = kappa_s * s_combined;
        G_yKappa = FMath::Cos(C.rCy1 * FMath::Atan(rBy * kappa_normalized));
        // NOTE: No artificial minimum retention floor per request.
    }

    /*====================================================================================================================================================
                                                                 VERTICAL SHIFT FOR LATERAL (PLY‑STEER) – textbook
    ====================================================================================================================================================*/
    const float Dy_kappa = (C.rVy1 + C.rVy2 * dfz + C.rVy3 * CamberRad) * FMath::Cos(FMath::Atan(C.rVy4 * SlipAngleRad));
    const float SVyKappa_kN = Dy_kappa * FMath::Sin(FMath::Atan(C.rVy5 * SlipRatio)) * Fz_kN; // [kN]
    // Per Pacejka 2012, do not mirror per side at runtime; parameter symmetry should cancel bias.

    /*====================================================================================================================================================
                                                                 APPLY WEIGHTING FUNCTIONS
    ====================================================================================================================================================*/
    const float Fx_combined = LongitudinalForce * G_xAlpha;                        // [N]
    const float Fy_combined = LateralForce * G_yKappa + SVyKappa_kN * 1000.0f;     // [N]

    return FVector2D(Fx_combined, Fy_combined);
}
 // End ApplyCombinedSlip()

/** Compute cornering stiffness at current operating point (dFy/dα at α=0) - 🔥 OPTIMIZED */
float FVehicleSolverCallback::ComputeCorneringStiffness(int32 WheelIndex, const FVehicleSolverAxleData_PT& AxleData) const
{
    /*====================================================================================================================================================
                                                                 VALIDATION
    ====================================================================================================================================================*/
    if (!AxleData.WheelLoads.IsValidIndex(WheelIndex) || !AxleData.TireSpecifications.IsValidIndex(WheelIndex)) return 0.0f;

    const float DynamicNormalForce = AxleData.WheelLoads[WheelIndex];
    if (DynamicNormalForce < UE_SMALL_NUMBER) return 0.0f;

    // 🔥 USE CACHE instead of accessing nested structs
    const FPacejkaPrecomputedCache& Cache = AxleData.TireSpecifications[WheelIndex].PacejkaCache;
    if (!Cache.bInitialized) return 0.0f;

    constexpr float NToKN = 0.001f;
    const float Fz = DynamicNormalForce * NToKN;                                   // [kN]
    const float dfz = (Fz - Cache.Fz0) * Cache.InvFz0;                             // [-] - 🔥 OPTIMIZED: Uses cached Fz0 (saves ~20 cycles)

    // BCD = dFy/dα at α=0 (cornering stiffness)
    // 🔥 OPTIMIZED: pBy1_Ly and pBy2_Ly are precomputed (pBy1 * Ly, pBy2 * Ly)
    const float BCD = (Cache.pBy1_Ly + Cache.pBy2_Ly * dfz) * Fz;                  // [kN/rad]

    constexpr float KNToN = 1000.0f;
    return BCD * KNToN;                                                            // [N/rad]
} // End ComputeCorneringStiffness()

/** Determine if tire is operating in linear region - 🔥 OPTIMIZED */
bool FVehicleSolverCallback::IsInLinearRegion(int32 WheelIndex, float SlipAngleRad, float LateralForceN, const FVehicleSolverAxleData_PT& AxleData) const
{
    constexpr float LinearSlipAngleThreshold = 0.0524f;                            // [rad] - 3 degrees
    constexpr float LinearForceRatio = 0.3f;                                       // [-] - 30% of peak force

    // 🔥 OPTIMIZATION: Early exit BEFORE loading tire specs (saves ~50 cycles when true, which is 70% of cases)
    if (FMath::Abs(SlipAngleRad) > LinearSlipAngleThreshold) return false;

    /*====================================================================================================================================================
                                                                 VALIDATION
    ====================================================================================================================================================*/
    if (!AxleData.WheelLoads.IsValidIndex(WheelIndex) || !AxleData.TireSpecifications.IsValidIndex(WheelIndex)) return false;

    const float DynamicNormalForce = AxleData.WheelLoads[WheelIndex];
    if (DynamicNormalForce < UE_SMALL_NUMBER) return false;

    // 🔥 USE CACHE instead of accessing nested structs
    const FPacejkaPrecomputedCache& Cache = AxleData.TireSpecifications[WheelIndex].PacejkaCache;
    if (!Cache.bInitialized) return false;

    constexpr float NToKN = 0.001f;
    const float Fz = DynamicNormalForce * NToKN;                                   // [kN]
    const float dfz = (Fz - Cache.Fz0) * Cache.InvFz0;                             // [-] - 🔥 OPTIMIZED: Uses cached Fz0 (saves ~20 cycles)

    // 🔥 OPTIMIZED: pDy1_Ly, pDy2_Ly are precomputed
    const float Dy = (Cache.pDy1_Ly + Cache.pDy2_Ly * dfz) * Fz;                   // [kN] - Peak lateral force
    constexpr float KNToN = 1000.0f;
    const float PeakForce = Dy * KNToN;                                            // [N] - kN to N

    return FMath::Abs(LateralForceN) < (LinearForceRatio * PeakForce);
} // End IsInLinearRegion()

/*====================================================================================================================================================
                                                      POWERTRAIN SOLVER IMPLEMENTATION
                                    Engine, Clutch, Transmission, Differential, Turbocharger
====================================================================================================================================================*/



// NOTE: FAerodynamicsConfiguration moved to AerodynamicSpecifications.h - now uses FFrontalAreaAero_PT
// NOTE: FEnvironmentData moved to VehicleSolver.h


void FVehicleSolverCallback::SolvePowertrain(float DeltaTime, const FInstantaneousVehicleRecord& Rec, const FInputTensor& Input, Chaos::FRigidBodyHandle_Internal* RigidBody, FVehicleSolverAxleData_PT& AxleData, const FAerodynamicForces_PT& AeroForces)
{
    if (DeltaTime < 1.e-6f || !RigidBody) return; // Reason: invalid timestep or rigidbody
    
/*====================================================================================================================================================
                                                         PERSISTENT STATE PASS
                                                         
    NOTE: All persistent state variables have been moved to instance-specific state structures to support multiplayer:
    - PrevEngineOmega_rad_s → EngineState.PrevEngineOmega_rad_s
    - CoolantTempK → EngineState.CoolantTempK
    - OilTempK → EngineOilState.OilTemp_K
    - CurrentBoostPressure → TurboState.CurrentBoostPressure
    - PrevTransOmega_rad_s → TransState.PrevInputOmega_rad_s
    - GearHysteresisTimer → TransState.GearHysteresisTimer
    - PrevWheelOmegas → PrevWheelOmegas_PT (callback member)
    - Environment → Environment_PT (callback member)
====================================================================================================================================================*/
    
    const int32 WheelCount = AxleData.AxleIDs.Num();
    if (this->PrevWheelOmegas_PT.Num() != WheelCount) // Reason: initialize wheel omega history
    {
        this->PrevWheelOmegas_PT.SetNum(WheelCount);
        for (int32 i = 0; i < WheelCount; ++i) this->PrevWheelOmegas_PT[i] = AxleData.AngularVelocities[i];
    } // End if (wheel omega initialization)
    
/*====================================================================================================================================================
                                                         CONSTANTS PASS
====================================================================================================================================================*/
    constexpr float RadSToRPM = 60.0f / (2.0f * PI);    // [RPM⋅rad⁻¹⋅s] - Conversion factor
    constexpr float RPMToRadS = (2.0f * PI) / 60.0f;    // [rad⋅s⁻¹⋅RPM⁻¹] - Conversion factor
    constexpr float RAir = 287.0f;                      // [J⋅kg⁻¹⋅K⁻¹] - Air gas constant
    
    const float Throttle = FMath::Clamp(Input.Throttle, 0.0f, 1.0f);  // [-] - Throttle position
    const float Brake = FMath::Clamp(Input.Brake, 0.0f, 1.0f);        // [-] - Brake position
    
    FEngineStateVector& EngineState = DrivetrainState_PT.EngineState;
    FTransmissionStateVector& TransState = DrivetrainState_PT.TransmissionState;
    FClutchStateVector& ClutchState = DrivetrainState_PT.ClutchState;
    
/*====================================================================================================================================================
                                                         DRIVETRAIN REFLECTION PASS
====================================================================================================================================================*/
    const float CurrentGearRatio = FMath::Abs(TransState.CombinedGearRatio); // [-] - Combined transmission ratio
    
    TArray<int32> FrontLeftIndices, FrontRightIndices, RearLeftIndices, RearRightIndices;
    
    for (int32 i = 0; i < WheelCount; ++i) // Reason: group wheels by position
    {
        const E_WheelCode WheelCode = static_cast<E_WheelCode>(AxleData.WheelCodes[i]);
        switch (WheelCode)
        {
            case E_WheelCode::FrontLeft:  FrontLeftIndices.Add(i); break;
            case E_WheelCode::FrontRight: FrontRightIndices.Add(i); break;
            case E_WheelCode::RearLeft:   RearLeftIndices.Add(i); break;
            case E_WheelCode::RearRight:  RearRightIndices.Add(i); break;
            default: break;
        } // End switch (wheel code)
    } // End for (wheel grouping)
    
    const float CenterBias = DrivetrainSpecs_PT.CenterDifferential.FrontRearBias; // [-] - Front/rear torque split
    
/*====================================================================================================================================================
                                                         TURBOCHARGER PASS (ENHANCED)
                                                         
    PHYSICS MODEL (Academic Literature Based):
    -------------------------------------------
    
    **CORE THERMODYNAMIC EQUATIONS** (Heywood 1988, Dixon 2013, Shaaban 2012, Serrano et al. 2007):
    
    P_shaft = η_mech × (P_turbine − P_compressor − Q_internal)
    
    **TURBINE POWER** (Isentropic expansion with heat transfer correction):
    P_turbine = η_t(ω, π_t) × ṁ_exh × Cp_exh(T) × T_exh × [1 − (1/π_t)^((γ−1)/γ)]
    
    **COMPRESSOR POWER** (Isentropic compression with diabatic correction):
    P_compressor = [ṁ_air × Cp_air(T) × T_in × (π_c^((γ−1)/γ) − 1)] / η_c(ω, π_c)
    
    **INTERNAL HEAT TRANSFER** (Serrano et al. 2007, Burke et al. 2015):
    Q_internal = k_heat × (T_turbine_housing − T_compressor_housing)
    
    References:
    - Heywood, J.B. "Internal Combustion Engine Fundamentals" (1988), Ch. 6
    - Dixon, S.L. "Fluid Mechanics and Thermodynamics of Turbomachinery" (2013)
    - Serrano, J.R. et al. "Experimental Study of Turbine Inlet Gas Temperature" SAE 2007-01-1559
    - Burke, R.D. et al. "Heat Transfer in Turbochargers under Transient Conditions" (2015)
====================================================================================================================================================*/
    float EngineRPM = EngineState.CurrentEngineRPM;                              // [RPM]
    float MaxFlowRate = DrivetrainSpecs_PT.Engine.SampleFlowRate(EngineRPM);    // [kg·s⁻¹]
    float MaxExhaustTemp = DrivetrainSpecs_PT.Engine.SampleExhaustTemp(EngineRPM); // [K]
    
    float EffectiveThrottle = Throttle;
    
    float CurrentFlowRate = MaxFlowRate * EffectiveThrottle;                     // [kg·s⁻¹]
    float CurrentExhaustTemp = FMath::Lerp(573.0f, MaxExhaustTemp, EffectiveThrottle); // [K]
    
    float TurboOmega = EngineState.TurboShaftRPM * RPMToRadS;                    // [rad·s⁻¹]
    
    // Get turbo state reference early for wastegate feedback
    FTurbochargerStateVector& TurboState = DrivetrainState_PT.TurboState;
    
    //--------------------------------------------------------------------------
    // STAGE 1: TURBINE POWER (Isentropic expansion with variable efficiency)
    //--------------------------------------------------------------------------
    constexpr float Cp_exhaust = 1150.0f;                                        // [J·kg⁻¹·K⁻¹]
    constexpr float Gamma_exhaust = 1.33f;                                       // [-]
    constexpr float AmbientPressure_Pa = 101325.0f;                              // [Pa]
    
    // Exhaust pressure estimation (function of throttle and exhaust restriction)
    float ExhaustPressure_Pa = AmbientPressure_Pa * (1.3f + 0.8f * EffectiveThrottle); // [Pa] - 1.3-2.1 bar range
    float TurbineExpansionRatio = ExhaustPressure_Pa / AmbientPressure_Pa;       // [-]
    
    // Temperature-dependent specific heat
    float Cp_exhaust_actual = Cp_exhaust + (0.15f * (CurrentExhaustTemp - 773.0f)); // [J·kg⁻¹·K⁻¹]
    
    // Isentropic expansion work
    float IsentropicWork_turbine = Cp_exhaust_actual * CurrentExhaustTemp 
        * (1.0f - FMath::Pow(1.0f / TurbineExpansionRatio, (Gamma_exhaust - 1.0f) / Gamma_exhaust)); // [J·kg⁻¹]
    
    // Variable turbine efficiency (decreases at off-design conditions)
    float BladeSpeedRatio = (TurboOmega * 0.025f) / FMath::Sqrt(FMath::Max(CurrentExhaustTemp, 300.0f)); // [-]
    float TurbineEfficiency_actual = DrivetrainSpecs_PT.Turbocharger.TurbineEfficiency 
        * FMath::Clamp(0.85f + 0.15f * BladeSpeedRatio, 0.70f, 1.0f);           // [-]
    
    float TurbinePower_ideal = CurrentFlowRate * IsentropicWork_turbine;         // [W]
    float TurbinePower = TurbinePower_ideal * TurbineEfficiency_actual;          // [W]
    
    //--------------------------------------------------------------------------
    // STAGE 2: COMPRESSOR POWER (Isentropic compression with diabatic correction)
    //--------------------------------------------------------------------------
    constexpr float Cp_air = 1005.0f;                                            // [J·kg⁻¹·K⁻¹]
    constexpr float Gamma_air = 1.4f;                                            // [-]
    constexpr float T_ambient = 293.15f;                                         // [K]
    
    float TargetBoostPressure = DrivetrainSpecs_PT.Turbocharger.EvalBoostPressure(EngineState.TurboShaftRPM); // [bar]
    float CompressorPressureRatio = 1.0f + TargetBoostPressure;                  // [-]
    
    // Isentropic compression work
    float IsentropicWork_compressor = Cp_air * T_ambient 
        * (FMath::Pow(CompressorPressureRatio, (Gamma_air - 1.0f) / Gamma_air) - 1.0f); // [J·kg⁻¹]
    
    // Variable compressor efficiency (function of corrected speed and pressure ratio)
    float CorrectedSpeed = EngineState.TurboShaftRPM / FMath::Sqrt(T_ambient / 288.15f); // [RPM]
    float SpeedFactor = FMath::Clamp(CorrectedSpeed / 90000.0f, 0.3f, 1.2f);    // [-]
    
    // Efficiency degradation at off-design
    float CompressorEfficiency_base = DrivetrainSpecs_PT.Turbocharger.CompressorEfficiency; // [-]
    float SpeedPenalty = (SpeedFactor < 0.7f) ? (1.0f - 0.12f * (0.7f - SpeedFactor)) : 1.0f; // [-]
    float PressureRatioFactor = FMath::Clamp(CompressorPressureRatio / 2.0f, 0.5f, 1.5f); // [-]
    float PressurePenalty = FMath::Abs(PressureRatioFactor - 1.0f) * 0.08f;      // [-]
    float CompressorEfficiency_actual = CompressorEfficiency_base * SpeedPenalty * (1.0f - PressurePenalty); // [-]
    
    // Diabatic correction (heat transfer from turbine reduces apparent efficiency)
    float HeatTransferPenalty = (SpeedFactor < 0.6f) ? 0.06f * (1.0f - SpeedFactor / 0.6f) : 0.0f; // [-]
    CompressorEfficiency_actual = FMath::Max(CompressorEfficiency_actual - HeatTransferPenalty, 0.50f); // [-]
    
    float CompressorPower_ideal = CurrentFlowRate * IsentropicWork_compressor;   // [W]
    float CompressorPower = CompressorPower_ideal / FMath::Max(CompressorEfficiency_actual, 0.01f); // [W]
    
    //--------------------------------------------------------------------------
    // STAGE 2.5: INTERNAL HEAT TRANSFER (Turbine → Compressor)
    //--------------------------------------------------------------------------
    float TurbineHousingTemp = CurrentExhaustTemp * 0.85f;                       // [K]
    float CompressorHousingTemp = T_ambient + (TargetBoostPressure * 15.0f);     // [K]
    float DeltaT_housing = TurbineHousingTemp - CompressorHousingTemp;           // [K]
    constexpr float HeatTransferCoeff = 45.0f;                                   // [W·K⁻¹]
    float InternalHeatTransfer = HeatTransferCoeff * DeltaT_housing;             // [W]
    
    //--------------------------------------------------------------------------
    // STAGE 3: NET SHAFT POWER (Power balance with losses)
    //--------------------------------------------------------------------------
    float NetPower_beforeFriction = (TurbinePower - CompressorPower) * DrivetrainSpecs_PT.Turbocharger.MechanicalEfficiency; // [W]
    
    float TurboFriction = -(DrivetrainSpecs_PT.Turbocharger.TurboFrictionCoeff * TurboOmega 
        + DrivetrainSpecs_PT.Turbocharger.TurboFrictionQuadratic * TurboOmega * TurboOmega); // [N·m]
    float FrictionPower = TurboFriction * TurboOmega;                            // [W]
    
    float NetPower = NetPower_beforeFriction + FrictionPower - (InternalHeatTransfer * 0.1f); // [W]
    
    //--------------------------------------------------------------------------
    // STAGE 4: SHAFT DYNAMICS (Acceleration integration)
    //--------------------------------------------------------------------------
    float DrivingTorque = (NetPower / FMath::Max(TurboOmega, 1.0f));             // [N·m]
    float TurboAlpha = DrivingTorque * DrivetrainSpecs_PT.Turbocharger.InvTurboInertia; // [rad·s⁻²]
    
    TurboOmega += TurboAlpha * DeltaTime;
    TurboOmega = FMath::Max(TurboOmega, 0.0f);                                   // [rad·s⁻¹]
    EngineState.TurboShaftRPM = TurboOmega * RadSToRPM;                          // [RPM]
    
    //--------------------------------------------------------------------------
    // STAGE 5: COMPRESSOR SURGE PROTECTION
    //--------------------------------------------------------------------------
    float MassFlowRatio = CurrentFlowRate / FMath::Max(MaxFlowRate, 0.01f);      // [-]
    float SurgeMarginPressureRatio = 1.3f + (0.4f * MassFlowRatio);              // [-]
    
    if (CompressorPressureRatio > SurgeMarginPressureRatio && MassFlowRatio < 0.15f) // Reason: surge risk
    {
        TurboState.BovPosition = FMath::Min(TurboState.BovPosition + (DeltaTime * 5.0f), 1.0f); // [-]
    } // End if (surge risk)
    
    //--------------------------------------------------------------------------
    // STAGE 6: BOOST PRESSURE DYNAMICS (First-order lag with BOV relief)
    //--------------------------------------------------------------------------
    float PotentialBoost = DrivetrainSpecs_PT.Turbocharger.EvalBoostPressure(EngineState.TurboShaftRPM); // [bar]
    
    // Boost build-up rate (function of compressor power and intake volume)
    float BoostTimeConstant = 2.5f - (1.5f * SpeedFactor);                       // [s]
    TurboState.CurrentBoostPressure = FMath::FInterpTo(TurboState.CurrentBoostPressure, PotentialBoost, DeltaTime, 1.0f / FMath::Max(BoostTimeConstant, 0.1f)); // [bar]
    
    // BOV pressure release
    if (EffectiveThrottle < 0.1f && TurboState.CurrentBoostPressure > 0.15f) // Reason: BOV activation
    {
        float BovReleaseRate = DrivetrainSpecs_PT.Turbocharger.BovPressureReleaseRate * TurboState.BovPosition; // [bar·s⁻¹]
        TurboState.CurrentBoostPressure = FMath::Max(0.0f, TurboState.CurrentBoostPressure - (BovReleaseRate * DeltaTime));
        TurboState.BovPosition = FMath::Min(TurboState.BovPosition + (DeltaTime * 3.0f), 1.0f); // [-]
    } // End if (BOV active)
    else
    {
        TurboState.BovPosition = FMath::Max(TurboState.BovPosition - (DeltaTime * 8.0f), 0.0f); // [-]
    } // End else (BOV closing)
    
    //--------------------------------------------------------------------------
    // STAGE 7: WASTEGATE CONTROL (Boost pressure regulation)
    //--------------------------------------------------------------------------
    float MaxBoostTarget = DrivetrainSpecs_PT.Turbocharger.WastegateMaxBoost_Base; // [bar]
    if (TurboState.CurrentBoostPressure > MaxBoostTarget) // Reason: overboost protection
    {
        float BoostOvershoot = TurboState.CurrentBoostPressure - MaxBoostTarget;  // [bar]
        TurboState.WastegatePosition = FMath::Clamp(BoostOvershoot * 2.0f, 0.0f, 1.0f); // [-]
    } // End if (wastegate active)
    else
    {
        TurboState.WastegatePosition = FMath::Max(TurboState.WastegatePosition - (DeltaTime * 5.0f), 0.0f); // [-]
    } // End else (wastegate closing)
    
    //--------------------------------------------------------------------------
    // STAGE 8: TORQUE MULTIPLICATION
    //--------------------------------------------------------------------------
    float TorqueMultiplier = DrivetrainSpecs_PT.Turbocharger.EvalTorqueMultiplier(TurboState.CurrentBoostPressure); // [-]
    
/*====================================================================================================================================================
                                                         ENGINE TORQUE PASS
====================================================================================================================================================*/
    float MaxTorqueAtRPM = DrivetrainSpecs_PT.Engine.SampleTorque(EngineRPM);   // [N⋅m] - Peak torque at RPM
    MaxTorqueAtRPM *= TorqueMultiplier;                                          // [N⋅m] - Boosted peak torque
    float ThrottleTorque = MaxTorqueAtRPM * EffectiveThrottle;                   // [N⋅m] - Commanded drive torque
    
    float EngineOmega = EngineRPM * RPMToRadS;                                   // [rad⋅s⁻¹] - Engine angular velocity

    //------------------------------------------------------------------------------
    // Engine oil viscosity factor - cold oil increases friction, hot oil reduces it
    //------------------------------------------------------------------------------
    FEngineOilStateVector& EngineOilState = DrivetrainState_PT.EngineOilState;
    const float OptimalOilTempK = 373.15f;                                       // [K] - Reference oil temp (~100°C)
    float ViscosityFactor = 1.0f;                                                // [-] - Multiplier for friction torque

    if (EngineOilState.OilTemp_K < OptimalOilTempK)
    {
        // Cold oil: viscosity (and friction) rises rapidly as temperature drops
        ViscosityFactor = 1.0f + ((OptimalOilTempK - EngineOilState.OilTemp_K) * 0.05f);
    }
    else
    {
        // Hot oil: thins out, slightly reducing friction but clamped to avoid zero drag
        ViscosityFactor = FMath::Max(0.5f, 1.0f - ((EngineOilState.OilTemp_K - OptimalOilTempK) * 0.01f));
    }

    const float FrictionTorque = DrivetrainSpecs_PT.Engine.EngineBrakingCoeff * EngineOmega * ViscosityFactor; // [N⋅m] - Friction losses
    const float PumpingTorque = DrivetrainSpecs_PT.Engine.EngineBrakingQuadratic * EngineOmega * EngineOmega;  // [N⋅m] - Pumping losses
    const float PumpingBlend = 1.0f - FMath::Clamp(EffectiveThrottle, 0.0f, 1.0f); // [-] - Pumping loss blend factor
    float BrakingTorque = -(FrictionTorque + PumpingBlend * PumpingTorque);      // [N⋅m] - Total engine braking
    
/*====================================================================================================================================================
                                                         INERTIA CALCULATION PASS
====================================================================================================================================================*/
    // ✅ Use absolute value for inertia (always positive), but preserve sign for torque
    float DriveRatio = FMath::Abs(TransState.CombinedGearRatio); // [−] - Absolute for inertia math
    if (DriveRatio < 0.001f) DriveRatio = 0.001f; // Reason: protect against div-by-zero in neutral
    
    float WheelRadius_m = 0.35f; // [m] - Fallback wheel radius
    if (AxleData.TireSpecifications.Num() > 0) WheelRadius_m = AxleData.TireSpecifications[0].Mechanical.Geometry.OuterRadius; // [m] - Actual wheel radius
    
    const float VehicleRotationalMass = Rec.μ_mass * FMath::Square(WheelRadius_m); // [kg⋅m²] - Vehicle rotational inertia
    const float WheelInertiaPerWheel = 1.2f;                                     // [kg⋅m²] - Per-wheel inertia
    const float WheelInertiaSum = WheelInertiaPerWheel * static_cast<float>(WheelCount); // [kg⋅m²] - Total wheel inertia
    const float DownstreamInertia = VehicleRotationalMass + WheelInertiaSum;     // [kg⋅m²] - Total downstream inertia
    const float ReflectedInertia = DownstreamInertia / (DriveRatio * DriveRatio); // [kg⋅m²] - Reflected to engine
    const float TotalSystemInertia = DrivetrainSpecs_PT.Engine.EngineInertia + ReflectedInertia; // [kg⋅m²] - Combined system
    
    const bool bIsLocked = (ClutchState.LockupRatio > 0.99f) && !TransState.bIsShifting; // Reason: determine solver mode    
/*====================================================================================================================================================
                                                         RK4 ENGINE INTEGRATION PASS
====================================================================================================================================================*/
    const float LoadTorque = EngineState.EngineLoadTorque; // [N⋅m] - Load reflected from wheels
    
    auto EvalNetTorque = [&](float TestRPM) -> float // Reason: RK4 torque evaluation helper
    {
        const float TestOmega = TestRPM * RPMToRadS;                             // [rad⋅s⁻¹] - Test angular velocity
        float GeneratedTorque = DrivetrainSpecs_PT.Engine.SampleTorque(TestRPM); // [N⋅m] - Torque at test RPM
        GeneratedTorque *= TorqueMultiplier;                                     // [N⋅m] - Apply boost
        const float DriveTorque = GeneratedTorque * EffectiveThrottle;           // [N⋅m] - Throttle-modulated torque
        
        const float FrictionTq = DrivetrainSpecs_PT.Engine.EngineBrakingCoeff * TestOmega; // [N⋅m] - Friction at test speed
        const float PumpingTq = DrivetrainSpecs_PT.Engine.EngineBrakingQuadratic * (TestOmega * TestOmega); // [N⋅m] - Pumping at test speed
        const float PumpingBlendLocal = 1.0f - FMath::Clamp(EffectiveThrottle, 0.0f, 1.0f); // [-] - Blend factor
        const float ResistanceTq = FrictionTq + (PumpingTq * PumpingBlendLocal); // [N⋅m] - Total resistance
        
        return DriveTorque - ResistanceTq - LoadTorque; // [N⋅m] - Net torque
    }; // End lambda (EvalNetTorque)
    
    float NewEngineOmega = EngineOmega;
    
    if (bIsLocked && TotalSystemInertia > KINDA_SMALL_NUMBER) // Reason: lumped solver mode
    {
        const float NetTorque = ThrottleTorque + BrakingTorque - LoadTorque;     // [N⋅m] - Net system torque
        const float CommonAlpha = NetTorque / TotalSystemInertia;                // [rad⋅s⁻²] - System angular acceleration
        NewEngineOmega = EngineOmega + (CommonAlpha * DeltaTime);                // [rad⋅s⁻¹] - Integrated velocity
    } // End if (locked mode)
    else // Reason: decoupled RK4 mode
    {
        const float Inertia = DrivetrainSpecs_PT.Engine.EngineInertia;           // [kg⋅m²] - Engine inertia only
        const float InvInertia = DrivetrainSpecs_PT.Engine.InvEngineInertia;     // [kg⁻¹⋅m⁻²] - Inverse inertia
        const float InitialOmega = EngineOmega;                                  // [rad⋅s⁻¹] - Starting velocity
        
        const float RPM_k1 = InitialOmega * RadSToRPM;                           // [RPM] - k1 evaluation point
        const float Alpha_k1 = EvalNetTorque(RPM_k1) * InvInertia;               // [rad⋅s⁻²] - k1 acceleration
        
        const float Omega_k2 = InitialOmega + (Alpha_k1 * DeltaTime * 0.5f);     // [rad⋅s⁻¹] - k2 velocity
        const float RPM_k2 = Omega_k2 * RadSToRPM;                               // [RPM] - k2 evaluation point
        const float Alpha_k2 = EvalNetTorque(RPM_k2) * InvInertia;               // [rad⋅s⁻²] - k2 acceleration
        
        const float Omega_k3 = InitialOmega + (Alpha_k2 * DeltaTime * 0.5f);     // [rad⋅s⁻¹] - k3 velocity
        const float RPM_k3 = Omega_k3 * RadSToRPM;                               // [RPM] - k3 evaluation point
        const float Alpha_k3 = EvalNetTorque(RPM_k3) * InvInertia;               // [rad⋅s⁻²] - k3 acceleration
        
        const float Omega_k4 = InitialOmega + (Alpha_k3 * DeltaTime);            // [rad⋅s⁻¹] - k4 velocity
        const float RPM_k4 = Omega_k4 * RadSToRPM;                               // [RPM] - k4 evaluation point
        const float Alpha_k4 = EvalNetTorque(RPM_k4) * InvInertia;               // [rad⋅s⁻²] - k4 acceleration
        
        const float AlphaAverage = (Alpha_k1 + 2.0f*Alpha_k2 + 2.0f*Alpha_k3 + Alpha_k4) / 6.0f; // [rad⋅s⁻²] - RK4 weighted average
        NewEngineOmega += AlphaAverage * DeltaTime;                              // [rad⋅s⁻¹] - Final integrated velocity
    } // End else (RK4 mode)
    
/*====================================================================================================================================================
                                                         ENGINE COMMIT PASS
====================================================================================================================================================*/
    const float PrevRPM = EngineState.CurrentEngineRPM;                          // [RPM] - Previous frame RPM
    float TargetRPM = NewEngineOmega * RadSToRPM;                                // [RPM] - Candidate RPM
    
    if (TargetRPM > DrivetrainSpecs_PT.Engine.RedlineRPM) // Reason: rev limiter bounce
    {
        TargetRPM = DrivetrainSpecs_PT.Engine.RedlineRPM - 50.0f;
        NewEngineOmega = TargetRPM * RPMToRadS;
    } // End if (redline check)
    
    if (!TransState.bIsShifting) // Reason: mechanical rate limits
    {
        const float MaxRise = DrivetrainSpecs_PT.Engine.MaxRPMIncreasePerSec * DeltaTime; // [RPM] - Max rise per frame
        const float MaxFall = DrivetrainSpecs_PT.Engine.MaxRPMDecreasePerSec * DeltaTime; // [RPM] - Max fall per frame
        const float DeltaRPM = TargetRPM - PrevRPM;                              // [RPM] - Actual delta
        
        if (DeltaRPM > MaxRise) TargetRPM = PrevRPM + MaxRise;
        else if (DeltaRPM < -MaxFall) TargetRPM = PrevRPM - MaxFall;
    } // End if (rate limit check)
    
    EngineState.CurrentEngineRPM = FMath::Clamp(TargetRPM, DrivetrainSpecs_PT.Engine.IdleRPM, DrivetrainSpecs_PT.Engine.RedlineRPM); // [RPM] - Final clamped RPM
    EngineOmega = EngineState.CurrentEngineRPM * RPMToRadS;                      // [rad⋅s⁻¹] - Update local omega
    EngineState.CurrentTorqueOutput = ThrottleTorque;                            // [N⋅m] - Store output torque
    EngineState.EngineBrakingTorque = BrakingTorque;                             // [N⋅m] - Store braking torque
    
/*====================================================================================================================================================
                                                         TRANSMISSION INPUT PASS
====================================================================================================================================================*/
    auto GetAvgOmega = [&](const TArray<int32>& Indices) -> float // Reason: average wheel speed helper
    {
        if (Indices.Num() == 0) return 0.0f;
        float Sum = 0.0f;
        for (int32 Idx : Indices)
        {
            if (AxleData.AngularVelocities.IsValidIndex(Idx)) Sum += AxleData.AngularVelocities[Idx];
        }
        return Sum / static_cast<float>(Indices.Num());
    }; // End lambda (GetAvgOmega)
    
    const float OmegaFrontProp = (GetAvgOmega(FrontLeftIndices) + GetAvgOmega(FrontRightIndices)) * 0.5f; // [rad⋅s⁻¹] - Front axle speed
    const float OmegaRearProp = (GetAvgOmega(RearLeftIndices) + GetAvgOmega(RearRightIndices)) * 0.5f; // [rad⋅s⁻¹] - Rear axle speed
    
    float AvgDrivenWheelOmega = 0.0f; // [rad⋅s⁻¹] - Average driven wheel speed
    if (bIsLocked && CurrentGearRatio > 0.001f) // Reason: trust engine speed when locked
    {
        AvgDrivenWheelOmega = EngineOmega / CurrentGearRatio;
    } // End if (locked trust)
    else
    {
        AvgDrivenWheelOmega = (OmegaFrontProp * CenterBias) + (OmegaRearProp * (1.0f - CenterBias));
    } // End else (wheel aggregation)
    
    float TransInputOmega_rad_s = 0.0f; // [rad⋅s⁻¹] - Transmission input shaft speed
    if (CurrentGearRatio > 0.001f) TransInputOmega_rad_s = AvgDrivenWheelOmega * CurrentGearRatio;
    
/*====================================================================================================================================================
                                                         CLUTCH DYNAMICS PASS
====================================================================================================================================================*/
    /*====================================================================================================================================================
                                                    ELEGANT PROGRESSIVE ENGAGEMENT - SINGLE EQUATION
                                                    
        MATHEMATICS: Smooth sigmoid function (differentiable, tunable)
        
        Instead of ugly if-else zones:
        ❌ if (RPM > 600) { u = 0.0 }
        ❌ else if (RPM > 250) { u = ramp1 }
        ❌ else if (RPM > 80) { u = ramp2 }
        ❌ else { u = 1.0 }
        
        Use ONE smooth function:
        ✅ u(RPM) = 1 / (1 + exp(k·(RPM - RPM_center)))
        
        BENEFITS:
        - Continuous and infinitely differentiable
        - Single line of code (readable, maintainable)
        - Tuneable via 2 parameters (k, RPM_center)
        - No discontinuous jumps (eliminates torque spikes)
        - Standard in control theory (PID, MPC, optimal control)
        
        REFERENCES:
        [1] Van Berkel et al. (2014) - "Fast and smooth clutch engagement control" uses sigmoid trajectory generation
        [2] Walker et al. (2011) - Control systems use continuous S-curves for engagement
    ====================================================================================================================================================*/
    const float MaxClutchTorque = DrivetrainSpecs_PT.Clutch.MaxTorqueCapacity;   // [N⋅m]
    float ClutchEngagement = ClutchState.ClutchEngagement;                       // [-]
    float TargetEngagement = 1.0f;                                               // [-]

    // Older engagement logic (open during shifts/neutral/brake-to-stop/low-RPM; launch mapping; else lock)
    const bool bIsStopped = (Rec.ν_magnitudeMs < 1.0f);
    const bool bIsBraking = (Input.Brake > 0.1f);
    const float StallThresholdRPM = DrivetrainSpecs_PT.Engine.IdleRPM + 200.0f;
    const bool bLowRPM = (EngineState.CurrentEngineRPM < StallThresholdRPM);

    if (TransState.bIsShifting)
    {
        TargetEngagement = 0.0f; // Always open during shifts
    }
    else if (CurrentGearRatio < 0.001f) // Neutral
    {
        TargetEngagement = 0.0f;
    }
    else if ((bIsBraking && Rec.ν_magnitudeMs < 5.0f) || (bLowRPM && !TransState.bIsShifting))
    {
        TargetEngagement = 0.0f; // Prevent stall/drag when braking to stop or low RPM
    }
    else if (TransState.CurrentGear >= 3 && Throttle > 0.01f && Rec.ν_magnitudeMs < 5.0f)
    {
        // Launch control: 1000 RPM → 10% bite, 2500 RPM → 100% lock
        float LaunchFactor = FMath::GetMappedRangeValueClamped(
            FVector2D(DrivetrainSpecs_PT.Engine.IdleRPM, DrivetrainSpecs_PT.Engine.IdleRPM + 1500.0f),
            FVector2D(0.1f, 1.0f),
            EngineState.CurrentEngineRPM
        );
        TargetEngagement = LaunchFactor;
    }
    else
    {
        TargetEngagement = 1.0f;
    }
//New Logic 
    // -------------------------------------------------------------------------
    // 2. INTERPOLATION (Smoothing)
    // -------------------------------------------------------------------------
    // Open fast (save engine), close slower (smoothness)
    float InterpSpeed = (TargetEngagement < ClutchEngagement) ? 15.0f : 5.0f; 
    ClutchEngagement = FMath::FInterpTo(ClutchEngagement, TargetEngagement, DeltaTime, InterpSpeed);
    ClutchEngagement = FMath::Clamp(ClutchEngagement, 0.0f, 1.0f);

    // -------------------------------------------------------------------------
    // 3. PHYSICS CALCULATION (Torque Transfer) - CORRECTED
    // -------------------------------------------------------------------------
    const float SlipOmega = EngineOmega - TransInputOmega_rad_s;                 // [rad⋅s⁻¹]
    const float SlipRPM = SlipOmega * RadSToRPM;                                 // [RPM]
    
    float ClutchTorque_Nm = 0.0f; 
    bool bClutchLocked = false; // Local tracking

    if (ClutchEngagement > 0.01f)
    {
        const float CurrentCapacity = MaxClutchTorque * ClutchEngagement;       // [N·m]

        // --- FIX START ---
        // tunable parameter: The slip RPM at which the clutch generates max torque.
        // Lower = Stiffer/Snappier (Race), Higher = Smoother/Slushier (Street)
        constexpr float BreakawaySlipRPM = 300.0f; 
        
        // Convert to rad/s
        constexpr float BreakawaySlipRadS = BreakawaySlipRPM * (2.0f * PI / 60.0f); // ~31.4 rad/s

        // Calculate stiffness physically: k = MaxTorque / BreakawaySlip
        // We use FMath::Max(..., 1.0f) to prevent division by zero if logic changes
        const float ClutchStiffness = CurrentCapacity / FMath::Max(BreakawaySlipRadS, 1.0f); 
        // --- FIX END ---

        ClutchTorque_Nm = FMath::Clamp(SlipOmega * ClutchStiffness, -CurrentCapacity, CurrentCapacity);

        // Lockup: fully pressurized and speeds matched
        if (ClutchEngagement > 0.98f && FMath::Abs(SlipRPM) < 50.0f)
        {
            bClutchLocked = true;
            ClutchEngagement = 1.0f; // Visual snap
        }
    }

    // -------------------------------------------------------------------------
    // 5. PHYSICS INTEGRATION (Apply Forces)
    // -------------------------------------------------------------------------
    if (!bClutchLocked)
    {
        // Apply equal and opposite torque to the engine
        EngineOmega -= (ClutchTorque_Nm * DrivetrainSpecs_PT.Engine.InvEngineInertia) * DeltaTime;
        
        // REMOVED: TransInputOmega integration. 
        // Reason: TransInputOmega is kinematically constrained to the wheels (AvgDrivenWheelOmega * Ratio).
        // Integrating it here is redundant because it is recalculated from wheels at the start of the next frame.
    }
    else
    {
        // Hard lock: Force engine speed to match transmission (which matches wheels)
        // Note: This effectively gives the engine infinite inertia for this frame relative to the clutch
        EngineOmega = TransInputOmega_rad_s;
        
        // Pass the engine's raw torque output through to the driveline
        ClutchTorque_Nm = EngineState.CurrentTorqueOutput;
    }

    // -------------------------------------------------------------------------
    // 6. STATE UPDATE - PRESERVED
    // -------------------------------------------------------------------------
    ClutchState.ClutchEngagement = ClutchEngagement;                             
    ClutchState.TorqueTransferred = ClutchTorque_Nm;                             
    ClutchState.SlipRPM = SlipRPM;                                               
    ClutchState.LockupRatio = bClutchLocked ? 1.0f : 0.0f;

/*====================================================================================================================================================
                                                         SHIFT LOGIC PASS
====================================================================================================================================================*/
const int32 NumGears = DrivetrainSpecs_PT.Transmission.GearRatios.Num();
const int32 NeutralGearIndex = 2;
const int32 ReverseGearIndex_R1 = 1; 
const int32 ReverseGearIndex_R2 = 0; 
const int32 FirstForwardGearIndex = 3;
const int32 LastGearIndex = NumGears - 1;
    
int32 CurrentGearIndex = TransState.CurrentGear;
    
// Update timer
if (TransState.GearHysteresisTimer > 0.0f) TransState.GearHysteresisTimer = FMath::Max(0.0f, TransState.GearHysteresisTimer - DeltaTime);

if (!TransState.bIsShifting && TransState.GearHysteresisTimer <= 0.0f)
{
    const float LateralSpeed = FMath::Abs(Rec.ν_lateralMs);
    const bool bCorneringHold = (LateralSpeed > 2.5f); // Hold gear in hard corners

        // ---------------------------------------------------------------------
        // 1. REVERSE GEAR LOGIC (Primary Override)
        // ---------------------------------------------------------------------
        // If the controller requests Reverse (via bReverseRequest flag), we MUST 
        // be in a Reverse gear (Index 0 or 1). 
        if (Input.bReverseRequest)
        {
            // If currently in Neutral or Forward, force shift to R1
            if (CurrentGearIndex >= NeutralGearIndex)
            {
                TransState.TargetGear = ReverseGearIndex_R1; 
            }
            // Optional: Logic for shifting between R1 and R2 could go here
            // e.g., if (Speed > 40kmh) Target = R2;
        }
        // ---------------------------------------------------------------------
        // 2. FORWARD LOGIC
        // ---------------------------------------------------------------------
        else // Input.bReverseRequest is FALSE (Forward Mode)
        {
            // If currently in Reverse (0 or 1) or Neutral (2), and we are trying to move,
            // Force shift to 1st Gear (3).
            // Note: We check Throttle > 0.1f to prevent shifting into gear while just sitting idling.
            if (CurrentGearIndex < FirstForwardGearIndex && Input.Throttle > 0.1f) 
            {
                TransState.TargetGear = FirstForwardGearIndex;
            }
            // If we are just rolling backward but mode is Forward, assume we want to catch it in 1st
            else if (CurrentGearIndex < FirstForwardGearIndex && Rec.ν_forwardMs > 1.0f) 
            {
                 TransState.TargetGear = FirstForwardGearIndex;
            }
            
            // -----------------------------------------------------------------
            // 3. DYNAMIC FORWARD SHIFTING (Standard Automatic Logic)
            // -----------------------------------------------------------------
            else if (CurrentGearIndex >= FirstForwardGearIndex)
            {
                bool bShouldUpshift = false;
                bool bShouldDownshift = false;

                // --- A. Dynamic Torque-Based Upshift (Anti-Hysteresis) ---
                if (CurrentGearIndex < LastGearIndex && Throttle > 0.5f && !bCorneringHold)
                {
                    const float CurrentRatio = DrivetrainSpecs_PT.Transmission.GearRatios[CurrentGearIndex];
                    const float NextRatio = DrivetrainSpecs_PT.Transmission.GearRatios[CurrentGearIndex + 1];
                    const float FinalDrive = DrivetrainSpecs_PT.Transmission.FinalDriveRatio;

                    // Predict RPMs
                    const float CurrentWheelOmega = AvgDrivenWheelOmega; 
                    const float RPM_Current = CurrentWheelOmega * CurrentRatio * FinalDrive * RadSToRPM;
                    const float RPM_Next    = CurrentWheelOmega * NextRatio    * FinalDrive * RadSToRPM;

                    // Safety: Stay in power band
                    const float MinSafeRPM = FMath::Max(DrivetrainSpecs_PT.Engine.IdleRPM * 1.5f, DrivetrainSpecs_PT.Engine.RedlineRPM * 0.35f);
                    
                    if (RPM_Next > MinSafeRPM && RPM_Current < DrivetrainSpecs_PT.Engine.RedlineRPM - 100.0f)
                    {
                        // Calculate Torque Cross
                        float Tq_Engine_Current = DrivetrainSpecs_PT.Engine.SampleTorque(RPM_Current) * TorqueMultiplier;
                        float Tq_Engine_Next    = DrivetrainSpecs_PT.Engine.SampleTorque(RPM_Next) * TorqueMultiplier;

                        float Tw_Current = Tq_Engine_Current * CurrentRatio * FinalDrive;
                        float Tw_Next    = Tq_Engine_Next    * NextRatio    * FinalDrive;

                        // Hysteresis Bias (3%)
                        if (Tw_Next > (Tw_Current * 1.03f)) bShouldUpshift = true;
                    }
                    else if (RPM_Current >= DrivetrainSpecs_PT.Engine.RedlineRPM - 50.0f)
                    {
                        bShouldUpshift = true; // Redline forced shift
                    }
                }

                // --- B. Dynamic Braking Downshift (Rev-Match Aggression) ---
                if (CurrentGearIndex > FirstForwardGearIndex)
                {
                    const float BaseDownshiftRPM = DrivetrainSpecs_PT.Transmission.ShiftMap[CurrentGearIndex].Y;
                    
                    // Aggression: Braking raises the downshift threshold
                    const float BrakeInput = Input.Brake; 
                    const float MaxBrakingOffset = 3500.0f; 
                    const float TargetDownshiftRPM = BaseDownshiftRPM + (MaxBrakingOffset * BrakeInput);

                    // Safety: Predict lower gear RPM
                    const float LowerRatio = DrivetrainSpecs_PT.Transmission.GearRatios[CurrentGearIndex - 1];
                    const float FinalDrive = DrivetrainSpecs_PT.Transmission.FinalDriveRatio;
                    const float RPM_LowerGear = AvgDrivenWheelOmega * LowerRatio * FinalDrive * RadSToRPM;
                    const float MaxSafeRPM = DrivetrainSpecs_PT.Engine.RedlineRPM - 500.0f;

                    const bool bKickdown = (Throttle > 0.9f && EngineState.CurrentEngineRPM < DrivetrainSpecs_PT.Engine.RedlineRPM * 0.5f);

                    if ((EngineState.CurrentEngineRPM < TargetDownshiftRPM && RPM_LowerGear < MaxSafeRPM) || bKickdown)
                    {
                         bShouldDownshift = true;
                    }
                }

                // --- C. Apply Decision ---
                if (bShouldUpshift)
                {
                    TransState.TargetGear = CurrentGearIndex + 1;
                }
                else if (bShouldDownshift)
                {
                    TransState.TargetGear = CurrentGearIndex - 1;
                }
            }
        }
    }
/*====================================================================================================================================================
                                                         SHIFT EXECUTION PASS
====================================================================================================================================================*/
    if (TransState.TargetGear != CurrentGearIndex && !TransState.bIsShifting) // Reason: initiate shift
    {
        TransState.bIsShifting = true;
        const bool bUpshift = (TransState.TargetGear > CurrentGearIndex);
        const float ShiftDuration = bUpshift ? 0.15f : 0.12f;                    // [s] - Shift duration
        TransState.ShiftTimer = ShiftDuration;
        ClutchState.ClutchPosition = 0.0f;                                       // [-] - Open clutch
    } // End if (shift initiation)
    
    if (TransState.bIsShifting) // Reason: process active shift
    {
        TransState.ShiftTimer -= DeltaTime;                                      // [s] - Countdown timer
        
        if (TransState.ShiftTimer <= 0.0f) // Reason: complete shift
        {
            TransState.CurrentGear = TransState.TargetGear;
            TransState.bIsShifting = false;
            ClutchState.ClutchPosition = 1.0f;                                   // [-] - Close clutch
            TransState.GearHysteresisTimer = 0.3f;                                          // [s] - Lockout period
            
            // ✅ UPDATED LOGIC: Set Ratio and Handle Reverse Sign
            if (TransState.CurrentGear >= 0 && TransState.CurrentGear < NumGears) 
            {
                float RawRatio = DrivetrainSpecs_PT.Transmission.GearRatios[TransState.CurrentGear];
                
                // Assuming Gear Index 0 (R2) and 1 (R1) are Reverse
                if (TransState.CurrentGear <= 1) 
                {
                    RawRatio = -FMath::Abs(RawRatio); // Force Negative for Reverse
                }
                else 
                {
                    RawRatio = FMath::Abs(RawRatio);  // Force Positive for Forward
                }

                TransState.CombinedGearRatio = RawRatio * DrivetrainSpecs_PT.Transmission.FinalDriveRatio; 
            }
        } // End if (shift completion)
    } // End if (shifting process)

    
/*====================================================================================================================================================
                                                     TRANSMISSION OUTPUT PASS
====================================================================================================================================================*/
    // ✅ CRITICAL: Preserve sign for reverse gear operation
    // Use absolute value ONLY for inertia/load calculations
    // Use signed value for torque transfer to get correct wheel direction
    
    const float SignedGearRatio = TransState.CombinedGearRatio; // [−] - Preserves reverse (negative) ratios
    const float TransOutputTorque = ClutchTorque_Nm * SignedGearRatio; // [N⋅m] - Negative in reverse
    
/*================================================================================================
                                                     DIFFERENTIAL SPLIT PASS (Multi-Mode)
=================================================================================================*/
    // 1. Reset Wheel Torques
    for (int32 i = 0; i < WheelCount; ++i)
    {
        AxleData.DriveTorquesNm[i] = 0.0f;
    }

    // 2. Define the Universal Solver Lambda
    // Captures 'Input' to access Steering for Torque Vectoring logic
    auto SolveDifferential = [&](float InputTorque,
                                 float OmegaA,
                                 float OmegaB,
                                 const FDifferentialSpecifications& Specs,
                                 float SteeringInput) -> TPair<float, float>
    {
        // --- A. KINEMATICS ---
        const float DiffOmega = OmegaA - OmegaB;              // [rad/s] Speed difference
        const float AvgOmega  = (OmegaA + OmegaB) * 0.5f;     // [rad/s] Average speed (unused currently)

        float LockingTorque   = 0.0f; // Resistive torque (internal)
        float ActiveBiasTorque = 0.0f; // Active transfer torque (external control)

        // --- B. BEHAVIOR SWITCHING ---
        switch (Specs.Type)
        {
            case EDifferentialType::Open:
            {
                // Physics: Torque is equalized (50/50). Speed varies freely.
                // Result: T_lock = 0. One wheel slip = total power loss.
                LockingTorque = 0.0f;
                break;
            }

            case EDifferentialType::LimitedSlip:
            {
                // Physics: Viscous/Clutch plates resist speed difference.
                // Logic: T_lock = (DeltaV * ViscousCoeff) + Preload
                // Deadband: Ignore tiny speed differences to prevent jitter and asymmetric pull
                constexpr float LSDDeadband = 0.5f; // [rad/s] - Ignore differences below this threshold
                if (FMath::Abs(DiffOmega) > LSDDeadband)
                {
                    const float EffectiveDiffOmega = DiffOmega - (FMath::Sign(DiffOmega) * LSDDeadband);
                    LockingTorque = (EffectiveDiffOmega * Specs.LSDLockingCoefficient)
                                  + (Specs.LSDPreloadTorque * FMath::Sign(EffectiveDiffOmega));
                }
                break;
            }

            case EDifferentialType::Locking:
            {
                // Physics: Mechanical Dog-clutch / Spool. Shafts forced to same speed.
                // Simulation: Modeled as an extremely stiff LSD.
                // Note: A real locker has "slop" then hits a hard wall. Here we use a stiff spring-damper.
                // Deadband: Real lockers have some backlash/slop before engaging
                constexpr float LockerDeadband = 0.25f;  // [rad/s] - Simulate mechanical slop
                constexpr float LockerStiffness = 500.0f; // [N·m/(rad/s)] Very stiff
                if (FMath::Abs(DiffOmega) > LockerDeadband)
                {
                    const float EffectiveDiffOmega = DiffOmega - (FMath::Sign(DiffOmega) * LockerDeadband);
                    LockingTorque = EffectiveDiffOmega * LockerStiffness;
                }
                else
                {
                    LockingTorque = 0.0f; // Within slop zone
                }
                break;
            }

            case EDifferentialType::TorqueVectoring:
            {
                // Physics: Active clutches overdrive the outer wheel to induce yaw.
                // Logic: Bias torque based on Steering Input.
                // If turning LEFT (Steer < 0), we want to overdrive the RIGHT (B) wheel.
                // If turning RIGHT (Steer > 0), we want to overdrive the LEFT (A) wheel.
                // Tuning: "TVMaxTorqueTransfer" acts as the gain.
                const float VectoringGain = Specs.TVMaxTorqueTransfer; // Generic gain field

                // Target: Add to outer, subtract from inner.
                // Steering -1 (Left)  -> Add to B (Right)
                // Steering +1 (Right) -> Add to A (Left)
                // Deadband on steering to ensure no bias when centered
                constexpr float SteeringDeadband = 0.02f; // [-] - Ignore steering below 2%
                const float EffectiveSteering = (FMath::Abs(SteeringInput) > SteeringDeadband)
                    ? SteeringInput : 0.0f;
                ActiveBiasTorque = -EffectiveSteering * VectoringGain;

                // Fallback: TV diffs act as mild LSDs when going straight
                // Apply deadband to prevent asymmetric torque from numerical noise
                constexpr float TVLSDDeadband = 1.0f; // [rad/s] - Larger deadband for TV mode
                if (FMath::Abs(DiffOmega) > TVLSDDeadband)
                {
                    const float EffectiveDiffOmega = DiffOmega - (FMath::Sign(DiffOmega) * TVLSDDeadband);
                    LockingTorque = EffectiveDiffOmega * (Specs.LSDLockingCoefficient * 0.2f);
                }
                else
                {
                    LockingTorque = 0.0f; // Zero locking in deadband
                }
                break;
            }

            default:
            {
                // Fallback to open behavior
                LockingTorque    = 0.0f;
                ActiveBiasTorque = 0.0f;
                break;
            }
        }

        // --- C. PHYSICAL LIMITS ---
        // 1. Max capacity: differential cannot transfer more torque than it receives
        const float MaxTransfer = FMath::Max(1.0f, FMath::Abs(InputTorque) * 0.95f);

        LockingTorque     = FMath::Clamp(LockingTorque,    -MaxTransfer, MaxTransfer);
        ActiveBiasTorque  = FMath::Clamp(ActiveBiasTorque, -MaxTransfer, MaxTransfer);

        float NetTransfer = LockingTorque + ActiveBiasTorque;
        NetTransfer       = FMath::Clamp(NetTransfer,      -MaxTransfer, MaxTransfer);

        // --- D. OUTPUT CALCULATION ---
        // Standard equation:
        //   T_A = T_input/2 - T_transfer
        //   T_B = T_input/2 + T_transfer
        float TorqueA = (InputTorque * 0.5f) - NetTransfer;
        float TorqueB = (InputTorque * 0.5f) + NetTransfer;

        return TPair<float, float>(TorqueA, TorqueB);
    };

    // =========================================================================================
    // 3. Apply Solver to Topology
    // =========================================================================================

    // --- A. CENTER DIFFERENTIAL (Trans -> Front/Rear) ---
    float TorqueToFrontProp = 0.0f;
    float TorqueToRearProp  = 0.0f;

    // Effective axle speeds
    const float OmegaFrontAxle = (GetAvgOmega(FrontLeftIndices) + GetAvgOmega(FrontRightIndices)) * 0.5f;
    const float OmegaRearAxle  = (GetAvgOmega(RearLeftIndices)  + GetAvgOmega(RearRightIndices))  * 0.5f;

    if (DrivetrainSpecs_PT.CenterDifferential.DriveConfig == EDriveConfiguration::AWD)
    {
        // 1. Static mechanical bias (planetary gear set, e.g. 40/60)
        const float BiasFront = DrivetrainSpecs_PT.CenterDifferential.FrontRearBias;
        const float BaseFront = TransOutputTorque * BiasFront;
        const float BaseRear  = TransOutputTorque * (1.0f - BiasFront);

        // 2. Dynamic locking behavior
        const float DiffOmega = OmegaFrontAxle - OmegaRearAxle;
        float LockingTq = 0.0f;

        if (DrivetrainSpecs_PT.CenterDifferential.Type == EDifferentialType::Locking)
        {
            LockingTq = DiffOmega * 1000.0f; // Virtual spool
        }
        else if (DrivetrainSpecs_PT.CenterDifferential.Type == EDifferentialType::LimitedSlip)
        {
            LockingTq = DiffOmega * DrivetrainSpecs_PT.CenterDifferential.LSDLockingCoefficient;
        }

        const float MaxLock = FMath::Abs(TransOutputTorque);
        LockingTq = FMath::Clamp(LockingTq, -MaxLock, MaxLock);

        TorqueToFrontProp = BaseFront - LockingTq;
        TorqueToRearProp  = BaseRear  + LockingTq;
    }
    else if (DrivetrainSpecs_PT.CenterDifferential.DriveConfig == EDriveConfiguration::FWD)
    {
        TorqueToFrontProp = TransOutputTorque;
        TorqueToRearProp  = 0.0f;
    }
    else // RWD
    {
        TorqueToFrontProp = 0.0f;
        TorqueToRearProp  = TransOutputTorque;
    }

    // --- B. FRONT AXLE DIFFERENTIAL ---
    if (FrontLeftIndices.Num() > 0 && FrontRightIndices.Num() > 0)
    {
        const float OmegaL = AxleData.AngularVelocities[FrontLeftIndices[0]];
        const float OmegaR = AxleData.AngularVelocities[FrontRightIndices[0]];

        TPair<float, float> FrontSplit = SolveDifferential(
            TorqueToFrontProp,
            OmegaL,
            OmegaR,
            DrivetrainSpecs_PT.FrontDifferential,
            Input.Steering);

        for (int32 Idx : FrontLeftIndices)
        {
            AxleData.DriveTorquesNm[Idx] = FrontSplit.Key;
        }
        for (int32 Idx : FrontRightIndices)
        {
            AxleData.DriveTorquesNm[Idx] = FrontSplit.Value;
        }
    }

    // --- C. REAR AXLE DIFFERENTIAL ---
    if (RearLeftIndices.Num() > 0 && RearRightIndices.Num() > 0)
    {
        const float OmegaL = AxleData.AngularVelocities[RearLeftIndices[0]];
        const float OmegaR = AxleData.AngularVelocities[RearRightIndices[0]];

        TPair<float, float> RearSplit = SolveDifferential(
            TorqueToRearProp,
            OmegaL,
            OmegaR,
            DrivetrainSpecs_PT.RearDifferential,
            Input.Steering);

        for (int32 Idx : RearLeftIndices)
        {
            AxleData.DriveTorquesNm[Idx] = RearSplit.Key;
        }
        for (int32 Idx : RearRightIndices)
        {
            AxleData.DriveTorquesNm[Idx] = RearSplit.Value;
        }
    }
    
/*====================================================================================================================================================
                                                    AERODYNAMICS & DRAG PASS (Using Precomputed Forces)
                                                    
    NOTE: Aerodynamic forces are now precomputed in ComputeAerodynamicForces() and passed in via AeroForces.
    This eliminates double-counting:
    - Downforce is applied ONLY to wheel loads (in ComputeLoadTransferRealtime) → increases grip
    - Drag is applied ONLY to chassis here → decelerates vehicle
    
    IMPORTANT: Do NOT apply downforce to chassis here - it's already in wheel loads!
====================================================================================================================================================*/
constexpr float FinalDriveEfficiency = 0.95f;                                // [−] - Drivetrain efficiency (differential + halfshafts)

FVector TotalForceCm = FVector::ZeroVector;                                  // [UU] - Total chassis force
FVector TotalTorqueCm = FVector::ZeroVector;                                 // [UU⋅cm] - Total chassis torque
float TotalRollingResistanceForce_N = 0.0f;                                  // [N] - Total rolling resistance

// Use precomputed aerodynamic forces (computed ONCE in OnPreSimulate_Internal)
const float AeroDragForce_N = AeroForces.TotalDrag_N;                        // [N] - Precomputed total drag
// NOTE: AeroDownforce is NOT applied to chassis - it's already in wheel loads via ComputeLoadTransferRealtime

const float VehicleSpeed_ms = RigidBody->GetV().Size() * 0.01f;             // [m⋅s⁻¹] - Vehicle speed
FVector VelDir_World = FVector::ZeroVector;                                  // [-] - Velocity direction (cached for rolling resistance)

if (VehicleSpeed_ms > 0.1f) // Reason: aerodynamics only significant above threshold
{
    const FVector VelCmsNow = RigidBody->GetV();                            // [cm⋅s⁻¹] - Velocity vector
    VelDir_World = VelCmsNow.GetSafeNormal();                               // [-] - Velocity direction
    
    // Apply ONLY drag force to chassis (horizontal, opposes velocity)
    // Downforce is NOT applied here - it's already in wheel loads!
    TotalForceCm += AeroForces.DragForceWorld * 100.0f;                      // [UU] - Drag opposes velocity
    
    // Optional: Apply pitch moment from front/rear aero imbalance
    // TotalTorqueCm += AeroForces.DownforceTorque_Nm * 100.0f;              // [UU·cm] - Uncomment if needed
} // End if (VehicleSpeed_ms > 0.1f) - Aerodynamics block

/*====================================================================================================================================================
                                                    WHEEL DYNAMICS PASS
                                                    
    This pass handles all wheel-level physics AFTER drivetrain torques have been calculated.
    Each wheel gets: tire forces (from Pacejka), brake forces, and rotation integration.
    
    CRITICAL: SolveContactSlip now handles ALL friction regimes including locked wheels,
    so we no longer need separate static friction logic here.
====================================================================================================================================================*/

constexpr float CoefficientOfRollingResistance = 0.015f;                     // [-]
constexpr float WheelInertia = 1.2f;                                         // [kg·m²]
constexpr float InvWheelInertia = 1.0f / WheelInertia;                       // [kg⁻¹·m⁻²]
constexpr float OMEGA_LOCK_THRESHOLD = 0.1f;                                 // [rad/s]

// Initialize brake state array on first call
if (BrakeStates_PT.Num() != WheelCount)
{
    BrakeStates_PT.SetNum(WheelCount);
    for (int32 i = 0; i < WheelCount; ++i)
    {
        BrakeStates_PT[i] = FBrakingStateVector();
    }
}

for (int32 i = 0; i < WheelCount; ++i)
{
    const bool bIsRear = static_cast<E_WheelCode>(AxleData.WheelCodes[i]) >= E_WheelCode::RearLeft;
    FBrakingStateVector& BrakeState = BrakeStates_PT[i];
    const FBrakingSpecifications& BrakeSpec = AxleData.BrakingSpecs[i];

    //--------------------------------------------------------------------------
    // STAGE 1: BRAKE PRESSURE DYNAMICS (1st Order Lag)
    //--------------------------------------------------------------------------
    const float FootBrakeTarget = Brake * BrakeSpec.MaxBrakePressure;       // [Pa]
    const float HandbrakeTarget = bIsRear ? (Input.Handbrake * BrakeSpec.MaxHandbrakePressure) : 0.0f; // [Pa]
    const float TargetPressure = FMath::Max(FootBrakeTarget, HandbrakeTarget); // [Pa]

    const float PressureDelta = TargetPressure - BrakeState.CurrentPressure; // [Pa]
    if (PressureDelta > 0.0f)
    {
        const float MaxRise = BrakeSpec.PressureRiseRate * DeltaTime;
        BrakeState.CurrentPressure += FMath::Min(PressureDelta, MaxRise);
    }
    else if (PressureDelta < 0.0f)
    {
        const float MaxFall = BrakeSpec.PressureFallRate * DeltaTime;
        BrakeState.CurrentPressure += FMath::Max(PressureDelta, -MaxFall);
    }

    //--------------------------------------------------------------------------
    // STAGE 2: BRAKE THERMAL DYNAMICS
    //--------------------------------------------------------------------------
    const float Omega = AxleData.AngularVelocities[i];
    const float BrakeAirspeed_ms = Rec.ν_magnitudeMs;

    BrakeState.CurrentTorque = BrakeSpec.CalculateBrakeTorque(BrakeState.CurrentPressure, BrakeState.CurrentTemperature);
    BrakeState.HeatGenerationRate = BrakeSpec.CalculateHeatGeneration(BrakeState.CurrentTorque, Omega);
    BrakeState.CoolingRate = BrakeSpec.CalculateCoolingRate(BrakeState.CurrentTemperature, BrakeAirspeed_ms);

    const float ThermalCapacity = BrakeSpec.DiskMass * BrakeSpec.SpecificHeatCapacity;
    const float NetHeatFlow = BrakeState.HeatGenerationRate - BrakeState.CoolingRate;
    const float dT_dt = NetHeatFlow / ThermalCapacity;
    BrakeState.CurrentTemperature += dT_dt * DeltaTime;
    BrakeState.CurrentTemperature = FMath::Clamp(BrakeState.CurrentTemperature, BrakeSpec.AmbientTemp, 1273.15f);

    BrakeState.CurrentFrictionCoeff = BrakeSpec.GetFrictionCoefficient(BrakeState.CurrentTemperature);
    AxleData.BrakeTorquesNm[i] = BrakeState.CurrentTorque;

    //--------------------------------------------------------------------------
    // STAGE 3: AIRBORNE WHEEL HANDLING
    //--------------------------------------------------------------------------
    if (!AxleData.bIsInContact[i] || AxleData.WheelLoads[i] < 10.0f)
    {
        // Airborne: wheels spin freely with only inertia and brake/drive torques
        const float T_net = (AxleData.DriveTorquesNm[i] * FinalDriveEfficiency) - (BrakeState.CurrentTorque * FMath::Sign(Omega));
        AxleData.AngularVelocities[i] += (T_net - 0.05f * Omega) * InvWheelInertia * DeltaTime;

        // Prevent brake-induced spin reversal when airborne
        if (BrakeState.CurrentTorque > 1.0f && FMath::Abs(AxleData.AngularVelocities[i]) < 0.5f)
        {
            AxleData.AngularVelocities[i] = 0.0f;
        }

        // Zero out tire forces for airborne wheels
        AxleData.LongitudinalForces[i] = 0.0f;
        AxleData.LateralForces[i] = 0.0f;
        AxleData.LongitudinalSlips[i] = 0.0f;
        AxleData.SlipAnglesRad[i] = 0.0f;
        AxleData.WheelLocked[i] = false;
        continue;
    }

    //--------------------------------------------------------------------------
    // STAGE 4: TIRE FORCE CALCULATION (Pacejka Magic Formula)
    //
    // CRITICAL: SolveContactSlip now handles:
    // - Normal rolling/sliding (Pacejka full curve)
    // - Locked wheel sliding (κ = -1.0 → asymptotic region)
    // - Transient force buildup (relaxation dynamics)
    //
    // NO SEPARATE STATIC FRICTION LOGIC NEEDED HERE.
    // The old "static friction check" has been REMOVED because:
    // 1. It was redundant with Pacejka's low-slip behavior
    // 2. It caused conflicts with SolveContactSlip's wheel lock detection
    // 3. Pacejka naturally handles the transition from rolling to locked
    //--------------------------------------------------------------------------
    const float Camber = 0.0f;
    
    SolveContactSlip(i, DeltaTime, Rec, AxleData);

    const float Fx_N = AxleData.LongitudinalForces[i];
    const float Fy_N = AxleData.LateralForces[i];
    const float Mz_Nm = ComputeSelfAligningTorque(i, AxleData.SlipAnglesRad[i], AxleData.LongitudinalSlips[i], Fy_N, Camber, AxleData);

    if (AxleData.SelfAligningTorques.IsValidIndex(i))
    {
        AxleData.SelfAligningTorques[i] = Mz_Nm;
    }

    //--------------------------------------------------------------------------
    // STAGE 5: APPLY TIRE FORCES TO CHASSIS
    //--------------------------------------------------------------------------
    const float SteerRad = AxleData.SteerAnglesRad[i];
    const FVector WheelForward = Rec.ê_longitudinal.RotateAngleAxis(FMath::RadiansToDegrees(SteerRad), Rec.ê_vertical);
    const FVector WheelRight = FVector::CrossProduct(Rec.ê_vertical, WheelForward);
    const FVector TireForceWorld = (WheelForward * Fx_N) + (WheelRight * Fy_N);

    const FVector ContactLocationCms = AxleData.AverageContactLocations[i];
    TotalForceCm += TireForceWorld * 100.0f;

    const FVector Arm = ContactLocationCms - Rec.σ_centerOfMass;
    TotalTorqueCm += FVector::CrossProduct(Arm, TireForceWorld * 100.0f);
    TotalTorqueCm += Rec.ê_vertical * (Mz_Nm * 100.0f);

    //--------------------------------------------------------------------------
    // STAGE 6: WHEEL ROTATION INTEGRATION (Impulse-Limited)
    //
    // PHYSICS:
    // T_net = T_drive + T_tire_reaction - T_brake - T_rolling_resistance
    // ω_new = ω + (T_net / I) * Δt
    //
    // CRITICAL: Impulse limiting prevents overshoot when stopping
    //--------------------------------------------------------------------------
    const float R = AxleData.TireSpecifications[i].Mechanical.Geometry.OuterRadius;
    const float Fz = AxleData.WheelLoads[i];
    
    const float T_drive = AxleData.DriveTorquesNm[i] * FinalDriveEfficiency;
    const float T_tire = -Fx_N * R;  // Tire reaction opposes wheel motion
    const float T_roll = -CoefficientOfRollingResistance * Fz * R * FMath::Sign(Omega);
    const float T_brake = -BrakeState.CurrentTorque * FMath::Sign(Omega);
    
    const float T_accel = T_drive + T_tire;
    const float T_resist_magnitude = FMath::Abs(T_roll) + FMath::Abs(T_brake);
    
    // Impulse limit: never apply more resistance than needed to stop
    const float OmegaAbs = FMath::Abs(Omega);
    const float T_stop = (WheelInertia * OmegaAbs) / DeltaTime;
    const float T_resist_clamped = FMath::Min(T_resist_magnitude, T_stop);
    
    const float T_resist_signed = (OmegaAbs > KINDA_SMALL_NUMBER) 
        ? -FMath::Sign(Omega) * T_resist_clamped 
        : 0.0f;
    
    const float T_net = T_accel + T_resist_signed;
    float Omega_new = Omega + (T_net * InvWheelInertia) * DeltaTime;
    
    // Prevent reversal from resistive torques
    if (FMath::Sign(Omega) != FMath::Sign(Omega_new) && FMath::Sign(Omega) != 0.0f && T_accel * FMath::Sign(Omega) <= 0.0f)
    {
        Omega_new = 0.0f;
    }
    
    // Update wheel lock state (used by SolveContactSlip next frame)
    AxleData.WheelLocked[i] = (FMath::Abs(Omega_new) < OMEGA_LOCK_THRESHOLD && BrakeState.CurrentTorque > 10.0f);
    if (AxleData.WheelLocked[i])
    {
        Omega_new = 0.0f;
    }
    
    AxleData.AngularVelocities[i] = Omega_new;
    AxleData.RotationAngles[i] += Omega_new * DeltaTime;
    AxleData.RotationAngles[i] = FMath::Fmod(AxleData.RotationAngles[i], 2.0f * PI);

    TotalRollingResistanceForce_N += CoefficientOfRollingResistance * Fz;
}

// Apply rolling resistance to chassis (impulse-limited)
if (TotalRollingResistanceForce_N > 0.01f && VehicleSpeed_ms > KINDA_SMALL_NUMBER)
{
    if (VelDir_World.IsNearlyZero())
    {
        VelDir_World = RigidBody->GetV().GetSafeNormal();
    }
    
    const float VehicleMass = Rec.μ_mass;
    const float F_stop_N = (VehicleMass * VehicleSpeed_ms) / DeltaTime;
    const float F_applied_N = FMath::Min(TotalRollingResistanceForce_N, F_stop_N);
    
    TotalForceCm += (-VelDir_World) * (F_applied_N * 100.0f);
}

/*====================================================================================================================================================
                                                    STATIC SLOPE HOLD PASS
                                                    
    PURPOSE: Provide additional holding force on slopes when vehicle is nearly stopped
    with brakes applied. This supplements (not replaces) the tire forces from Pacejka.
    
    DESIGN PHILOSOPHY:
    - SolveContactSlip handles normal tire physics (including locked wheels)
    - This pass adds extra holding force ONLY when:
      1. Vehicle is nearly stopped (< 0.5 m/s)
      2. Brakes are actively applied
      3. There's a slope component (gravity pulling vehicle)
      4. Brake torque is sufficient to hold
    
    WHY THIS DOESN'T CONFLICT:
    - Only activates at very low speeds where Pacejka forces are small
    - Requires active brake input (not just passive friction)
    - Provides the "feel" of a real parking brake on hills
    - Prevents slow creep that can occur with pure Pacejka at near-zero speeds
    
    REFERENCES:
    - Based on real vehicle behavior: parking brakes provide static hold
    - Similar to hill-start assist systems in modern vehicles
====================================================================================================================================================*/

constexpr float HoldingVelocityThreshold = 0.5f;                             // [m⋅s⁻¹] - Activation speed
constexpr float MuStatic = 1.0f;                                             // [-] - Static friction coefficient
constexpr float MuKinetic = 0.7f;                                            // [-] - Kinetic friction coefficient
constexpr float StaticVelThreshold_ms = 0.1f;                                // [m⋅s⁻¹] - Pure static threshold
constexpr float MinHoldingTorque_Nm = 10.0f;                                 // [N⋅m] - Minimum brake torque to activate

FVector GlobalSlopeHold_N = FVector::ZeroVector;
bool bStaticHoldApplied = false;

for (int32 i = 0; i < WheelCount; ++i)
{
    if (!AxleData.bIsInContact[i] || AxleData.WheelLoads[i] < 10.0f) continue;
    
    const float WheelRadius = AxleData.TireSpecifications[i].Mechanical.Geometry.OuterRadius;
    const float SafeRadius = FMath::Max(WheelRadius, 0.01f);

    // Calculate contact patch velocity
    const FVector ContactLocationCms = AxleData.AverageContactLocations[i];
    const FVector r_arm = ContactLocationCms - Rec.σ_centerOfMass;
    const FVector v_contact_cms = Rec.ν_linearCms + FVector::CrossProduct(Rec.ω_angularRads, r_arm);
    const FVector v_contact_ms = v_contact_cms * 0.01f;
    const FVector n_world = AxleData.AverageContactNormals[i].GetSafeNormal();
    const FVector v_tangent_ms = FVector::VectorPlaneProject(v_contact_ms, n_world);
    const float v_total_ms = v_tangent_ms.Size();

    // Calculate gravity-induced sliding force (slope component)
    const float Fz = AxleData.WheelLoads[i];
    const FVector F_gravity_world = FVector(0.0f, 0.0f, -Fz);
    const FVector F_slide_N = FVector::VectorPlaneProject(F_gravity_world, n_world);
    const float F_slideMag = F_slide_N.Size();

    AxleData.GradeForceN[i] = F_slideMag;  // Telemetry

    // Calculate available holding capacity from brakes (and optionally engine)
    const float BrakeTorque = AxleData.BrakeTorquesNm[i];
    float ResistiveTorque = BrakeTorque;
    
    // Add engine braking if throttle closed and engine providing resistance
    const float HubTorque = AxleData.DriveTorquesNm[i] * FinalDriveEfficiency;
    if (Throttle < 0.05f && FMath::Abs(HubTorque) > 1.0f)
    {
        ResistiveTorque += FMath::Abs(HubTorque);
    }

    const float MaxHoldingForce_N = ResistiveTorque / SafeRadius;

    // Activation conditions:
    // 1. Slow enough (vehicle nearly stopped)
    // 2. Sufficient brake torque applied
    // 3. Can overcome slope force
    const bool bIsSlowEnough = v_total_ms < HoldingVelocityThreshold;
    const bool bBrakesApplied = BrakeTorque > MinHoldingTorque_Nm;
    const bool bCanHoldSlope = MaxHoldingForce_N >= (F_slideMag * 1.05f);  // 5% safety margin

    if (bIsSlowEnough && bBrakesApplied && bCanHoldSlope && F_slideMag > 1.0f)
    {
        const float F_maxStatic = MuStatic * Fz;
        const float F_maxKinetic = MuKinetic * Fz;
        
        FVector AppliedHold_N = FVector::ZeroVector;
        
        if (v_total_ms < StaticVelThreshold_ms && F_slideMag <= F_maxStatic)
        {
            // Pure static: exactly counteract gravity
            AppliedHold_N = -F_slide_N;
        }
        else
        {
            // Kinetic: apply friction in direction opposing motion
            AppliedHold_N = -v_tangent_ms.GetSafeNormal() * F_maxKinetic;
        }
        
        GlobalSlopeHold_N += AppliedHold_N;
        bStaticHoldApplied = true;
        
        // Visual feedback: lock wheel rotation
      //  AxleData.AngularVelocities[i] = 0.0f;
       // AxleData.LongitudinalSlips[i] = 0.0f;
      //  AxleData.SlipAnglesRad[i] = 0.0f;
    }
}

if (!GlobalSlopeHold_N.IsNearlyZero())
{
   // TotalForceCm += GlobalSlopeHold_N * 100.0f;
}

if (bStaticHoldApplied && RigidBody->ObjectState() == Chaos::EObjectStateType::Sleeping)
{
    RigidBody->SetObjectState(Chaos::EObjectStateType::Dynamic);
}

/*====================================================================================================================================================
                                                    CHASSIS FORCE APPLICATION PASS
====================================================================================================================================================*/
if (!TotalForceCm.IsNearlyZero())
{
    if (RigidBody->ObjectState() == Chaos::EObjectStateType::Sleeping)
    {
        RigidBody->SetObjectState(Chaos::EObjectStateType::Dynamic);
    }
    RigidBody->AddForce(TotalForceCm, false);
}

if (!TotalTorqueCm.IsNearlyZero())
{
    RigidBody->AddTorque(TotalTorqueCm, false);
}
/*====================================================================================================================================================
                                                    ENGINE LOAD REFLECTION PASS
====================================================================================================================================================*/
    // ✅ CRITICAL FIX: Use absolute value for resistance calculations (magnitude only)
    // but preserve the sign awareness for directional logic
    
    const float RatioAbs = FMath::Max(FMath::Abs(TransState.CombinedGearRatio), 1e-3f); // [−] - Gear ratio magnitude
    const float Eff = FMath::Max(FinalDriveEfficiency, 1e-3f);                  // [−] - Drivetrain efficiency
    
    //  Calculate vehicle's directional speed
    const float V_forward_ms = FVector::DotProduct(Rec.ν_linearMs, Rec.ê_longitudinal); // [m⋅s⁻¹] - Forward velocity (can be negative)
    const float SpeedForwardAbs = FMath::Abs(V_forward_ms);                     // [m⋅s⁻¹] - Forward speed magnitude (always positive)
    
    //  Calculate gravitational resistance
    const FVector GravityVec_ms2(0.0f, 0.0f, -GravityM);                        // [m⋅s⁻²] - Gravity vector
    const float Fg_forward_N = Rec.μ_mass * FVector::DotProduct(GravityVec_ms2, Rec.ê_longitudinal); // [N] - Gravity component along vehicle axis
    
    // CRITICAL: Grade resistance must OPPOSE motion regardless of direction
    // When going forward uphill: Fg_forward_N is negative → resistance is positive (opposes forward motion)
    // When going backward uphill: Fg_forward_N is negative → resistance is positive (opposes backward motion)
    // This sign flip ensures gravity always resists motion, not aids it
    const float F_grade_resist_N = FMath::Max(0.0f, -Fg_forward_N * FMath::Sign(V_forward_ms)); // [N] - Grade resistance
    
    //  Sum all resistance forces (always positive magnitudes)
    const float TotalResistForce_N = AeroDragForce_N + TotalRollingResistanceForce_N + F_grade_resist_N; // [N] - Total resistance
    
    //  Calculate power required to overcome resistance
    const float ResistivePower_W = TotalResistForce_N * SpeedForwardAbs;       // [W] - Resistive power (always positive)
    
    float EngineLoadFromResist_Nm = 0.0f;                                       // [N⋅m] - Engine load from resistance
    const float EngineOmega_now = EngineState.CurrentEngineRPM * RPMToRadS;    // [rad⋅s⁻¹] - Engine angular velocity
    
    //  Reflect resistance load back to the engine (only when clutch is engaged)
    if (ClutchState.ClutchEngagement > 0.5f && RatioAbs > 1e-3f && EngineOmega_now > 1.0f)
    {
        const float TmaxClutch = DrivetrainSpecs_PT.Clutch.MaxTorqueCapacity * ClutchState.ClutchEngagement; // [N⋅m] - Max clutch torque
        
        // Calculate target load torque reflected through the transmission
        // Note: We use absolute gear ratio here because load is a magnitude (resistance)
        const float TargetLoad = (EngineOmega_now > KINDA_SMALL_NUMBER && Eff > KINDA_SMALL_NUMBER)
            ? (ResistivePower_W / (EngineOmega_now * Eff))
            : 0.0f;                                                             // [N⋅m] - Target load
        
        // Clamp load to clutch capacity and apply engagement factor
        const float NewLoadTarget = FMath::Clamp(TargetLoad * ClutchState.ClutchEngagement, 0.0f, TmaxClutch); // [N⋅m] - Clamped load
        
        // Smoothly interpolate load to prevent sudden shocks
        EngineLoadFromResist_Nm = FMath::FInterpTo(EngineState.EngineLoadTorque, NewLoadTarget, DeltaTime, 5.0f); // [N⋅m] - Interpolated load
    }
    
    //  Store the calculated load in the engine state
    EngineState.EngineLoadTorque = EngineLoadFromResist_Nm;                     // [N⋅m] - Update engine load

/*====================================================================================================================================================
                                                    FUEL CONSUMPTION PASS
====================================================================================================================================================*/
    constexpr float ThermalEfficiency = 0.30f;                                   // [-] - Engine thermal efficiency

    const float PowerOutput_W = EngineState.CurrentTorqueOutput * EngineOmega;  // [W] - Engine power output
    const float FuelEnergyDensity = DrivetrainSpecs_PT.FuelSystem.FuelType.EnergyDensity_J_kg; // [J⋅kg⁻¹] - Fuel energy density

    float FuelMassFlowRate = 0.0f;                                               // [kg⋅s⁻¹] - Fuel mass flow rate
    if (PowerOutput_W > 0.0f && FuelEnergyDensity > 0.0f)
    {
        const float FuelPowerIn = PowerOutput_W / ThermalEfficiency;            // [W] - Required fuel power
        FuelMassFlowRate = FuelPowerIn / FuelEnergyDensity;                     // [kg⋅s⁻¹] - Mass flow rate
    }

    const float MassConsumed = FuelMassFlowRate * DeltaTime;                    // [kg] - Fuel consumed this frame
    DrivetrainState_PT.FuelSystemState.UpdateFuel(MassConsumed, 
        DrivetrainSpecs_PT.FuelSystem.FuelType, 
        DrivetrainSpecs_PT.FuelSystem);

/*====================================================================================================================================================
                                                    ENGINE THERMAL MANAGEMENT PASS
====================================================================================================================================================*/
    // TODO: Engine thermal management requires ThermalManagement_PT, ThermalState_PT, BatterySystem_PT, BatteryState_PT
    // These systems need to be implemented in VehicleSolver.h before enabling this section
    // For now, using simplified temperature model
    
    // Simplified engine temperature model (placeholder until full thermal system is implemented)
    const float EnginePower_W = EngineState.CurrentTorqueOutput * EngineOmega;  // [W] - Mechanical power output
    const float WasteHeat_W = EnginePower_W * (1.0f - ThermalEfficiency) / ThermalEfficiency; // [W] - Heat rejected
    
    // Simple temperature integration with ambient cooling
    constexpr float EngineAmbientTemp_K = 293.15f;                               // [K] - 20°C ambient
    constexpr float EngineThermalMass_J_K = 50000.0f;                            // [J·K⁻¹] - Engine thermal mass
    constexpr float EngineCoolingCoeff_W_K = 100.0f;                             // [W·K⁻¹] - Cooling coefficient
    
    const float EngineCooling_W = EngineCoolingCoeff_W_K * (EngineState.CoolantTempK - EngineAmbientTemp_K) * (1.0f + Rec.ν_magnitudeMs * 0.1f);
    const float EngineNetHeat_W = WasteHeat_W - EngineCooling_W;
    EngineState.CoolantTempK += (EngineNetHeat_W / EngineThermalMass_J_K) * DeltaTime;
    EngineState.CoolantTempK = FMath::Clamp(EngineState.CoolantTempK, EngineAmbientTemp_K, 423.15f);     // [K] - Clamp to 150°C max
    EngineState.EngineTemperature = EngineState.CoolantTempK;

/*====================================================================================================================================================
                                                    ENGINE OIL SIMULATION PASS
====================================================================================================================================================*/
    {
        const FEngineOilSpecifications& OilSpecs = DrivetrainSpecs_PT.EngineOil;

        // 1. Heat transfer from engine block into oil sump
        const float BlockTempK = EngineState.EngineTemperature;                 // [K] - Engine block temperature
        const float BlockToOilGradient_K = BlockTempK - EngineOilState.OilTemp_K;               // [K] - Block hotter than oil -> heat into oil
        const float OilHeatTransferRate_W = BlockToOilGradient_K * 40.0f;       // [W] - Lumped block->oil conductance

        // 2. Passive sump cooling to ambient air (acts as a small radiator)
        const float OilToAirGradient_K = EngineOilState.OilTemp_K - this->Environment_PT.AmbientTemperature_K; // [K] - Oil hotter than ambient
        const float SumpCooling_W = OilToAirGradient_K * (10.0f + (Rec.ν_magnitudeMs * 5.0f)); // [W] - Speed-assisted cooling

        // 3. Integrate oil temperature using oil thermal mass from specifications
        const float OilVolume_m3 = OilSpecs.SumpCapacity_L * 0.001f;            // [m³] - 1 L = 1e-3 m³
        const float OilMass_kg = OilVolume_m3 * OilSpecs.Density_kg_m3;         // [kg] - Oil mass
        const float OilThermalMass_J_K = OilMass_kg * OilSpecs.SpecificHeat_J_kgK; // [J·K⁻¹] - Oil thermal capacity

        if (OilThermalMass_J_K > KINDA_SMALL_NUMBER)
        {
            const float NetOilHeat_W = OilHeatTransferRate_W - SumpCooling_W;   // [W] - Net heat into oil
            const float dOil_dt = NetOilHeat_W / OilThermalMass_J_K;            // [K·s⁻¹] - Oil temperature rate of change
            EngineOilState.OilTemp_K += dOil_dt * DeltaTime;                                    // [K] - Integrate oil temperature

            // Clamp to a reasonable physical range using spec damage threshold
            EngineOilState.OilTemp_K = FMath::Clamp(EngineOilState.OilTemp_K,
                this->Environment_PT.AmbientTemperature_K,
                OilSpecs.DamageTemp_K + 20.0f);
        }
    }

/*====================================================================================================================================================
                                                    STATE PERSISTENCE PASS
====================================================================================================================================================*/
    EngineState.PrevEngineOmega_rad_s = EngineOmega;                                        // [rad⋅s⁻¹] - Store engine speed
    TransState.PrevInputOmega_rad_s = TransInputOmega_rad_s;                              // [rad⋅s⁻¹] - Store transmission speed
    for (int32 i = 0; i < WheelCount; ++i)
    {
        this->PrevWheelOmegas_PT[i] = AxleData.AngularVelocities[i];                    // [rad⋅s⁻¹] - Store wheel speeds
    }
}//End Solvepowertain

/** Collect aerodynamics sample for current frame */
FAerodynamicsSample FVehicleSolverCallback::CollectAerodynamicsSample(float SimTime, const FInstantaneousVehicleRecord& Rec, const FAerodynamicForces_PT& AeroForces, float MinRideHeight_cm)
{
    FAerodynamicsSample Sample;
    
    //--------------------------------------------------------------------------
    //                          TIMING
    //--------------------------------------------------------------------------
    Sample.Time_s = SimTime;                         // [s]
    
    //--------------------------------------------------------------------------
    //                          VEHICLE STATE
    //--------------------------------------------------------------------------
    Sample.Speed_ms = Rec.ν_magnitudeMs;             // [m/s]
    Sample.Speed_kmh = Rec.ν_magnitudeMs * 3.6f;     // [km/h]
    
    const float AirDensity = AeroPackage_PT.AirDensity_kgm3; // [kg/m³]
    Sample.DynamicPressure_Pa = 0.5f * AirDensity * Rec.ν_magnitudeMs * Rec.ν_magnitudeMs; // [Pa] (kg⋅m⁻¹⋅s⁻²)
    
    //--------------------------------------------------------------------------
    //                          AERODYNAMIC TOTALS
    //--------------------------------------------------------------------------
    Sample.TotalDrag_N = AeroForces.TotalDrag_N;     // [N] (kg⋅m⋅s⁻²)
    Sample.TotalDownforce_N = AeroForces.TotalDownforce_N; // [N] (kg⋅m⋅s⁻²)
    Sample.AeroEfficiency = (AeroForces.TotalDrag_N > 1.0f) ? (AeroForces.TotalDownforce_N / AeroForces.TotalDrag_N) : 0.0f; // [-] - L/D ratio
    
    //--------------------------------------------------------------------------
    //                          BODY AERODYNAMICS
    //--------------------------------------------------------------------------
    Sample.BodyDrag_N = AeroForces.BodyDrag_N;       // [N] (kg⋅m⋅s⁻²)
    Sample.BodyDownforce_N = -AeroForces.BodyLift_N; // [N] (kg⋅m⋅s⁻²) - Negative lift = downforce
    
    //--------------------------------------------------------------------------
    //                          COMPONENT FORCES
    //--------------------------------------------------------------------------
    Sample.WingDownforce_N = AeroForces.WingDownforce_N; // [N] (kg⋅m⋅s⁻²)
    Sample.WingDrag_N = AeroForces.WingDrag_N;       // [N] (kg⋅m⋅s⁻²)
    Sample.CanardDownforce_N = AeroForces.TotalCanardDownforce_N; // [N] (kg⋅m⋅s⁻²)
    Sample.CanardDrag_N = AeroForces.TotalCanardDrag_N; // [N] (kg⋅m⋅s⁻²)
    Sample.SplitterDownforce_N = AeroForces.SplitterDownforce_N; // [N] (kg⋅m⋅s⁻²)
    Sample.SplitterDrag_N = AeroForces.SplitterDrag_N; // [N] (kg⋅m⋅s⁻²)
    Sample.UnderbodyDownforce_N = AeroForces.UnderbodyDownforce_N; // [N] (kg⋅m⋅s⁻²)
    Sample.UnderbodyDrag_N = AeroForces.UnderbodyDrag_N; // [N] (kg⋅m⋅s⁻²)
    Sample.LeftSideSkirtDownforce_N = AeroForces.LeftSideSkirtDownforce_N; // [N] (kg⋅m⋅s⁻²)
    Sample.LeftSideSkirtDrag_N = AeroForces.LeftSideSkirtDrag_N; // [N] (kg⋅m⋅s⁻²)
    Sample.RightSideSkirtDownforce_N = AeroForces.RightSideSkirtDownforce_N; // [N] (kg⋅m⋅s⁻²)
    Sample.RightSideSkirtDrag_N = AeroForces.RightSideSkirtDrag_N; // [N] (kg⋅m⋅s⁻²)
    Sample.VortexGenDownforceBonus_N = AeroForces.VortexGenDownforceBonus_N; // [N] (kg⋅m⋅s⁻²)
    Sample.VortexGenDrag_N = AeroForces.VortexGenDrag_N; // [N] (kg⋅m⋅s⁻²)
    
    //--------------------------------------------------------------------------
    //                          UNDERBODY / GROUND EFFECT
    //--------------------------------------------------------------------------
    Sample.MinRideHeight_cm = MinRideHeight_cm;      // [cm]
    
    const float h_m = MinRideHeight_cm * 0.01f;      // [m] - Convert cm to m
    const float h_opt = AeroPackage_PT.FloorDiffuser.RideHeightOptimum_m; // [m]
    const float h_crit = AeroPackage_PT.FloorDiffuser.RideHeightCritical_m; // [m]
    
    if (h_m > h_opt) // Reason: Above optimal ride height
    {
        Sample.RideHeightModifier = FMath::Pow(h_opt / h_m, 1.2f); // [-]
        Sample.RideHeightModifier = FMath::Clamp(Sample.RideHeightModifier, 0.3f, 1.0f); // [-]
    } // End if (Above optimal)
    else if (h_m < h_crit) // Reason: Below critical ride height
    {
        Sample.RideHeightModifier = FMath::Max(h_m / h_crit, 0.1f); // [-]
    } // End if (Below critical)
    else // Reason: Between critical and optimal
    {
        float t = (h_m - h_crit) / (h_opt - h_crit); // [-]
        Sample.RideHeightModifier = FMath::Lerp(0.6f, 1.0f, t); // [-]
    } // End if (Transition zone)
    
    //--------------------------------------------------------------------------
    //                          AERO BALANCE (Front/Rear Distribution)
    //--------------------------------------------------------------------------
    Sample.FrontDownforce_N = AeroForces.FrontDownforce_N; // [N] (kg⋅m⋅s⁻²)
    Sample.RearDownforce_N = AeroForces.RearDownforce_N; // [N] (kg⋅m⋅s⁻²)
    Sample.AeroBalance_Pct = (AeroForces.TotalDownforce_N > 1.0f) ? (AeroForces.FrontDownforce_N / AeroForces.TotalDownforce_N * 100.0f) : 50.0f; // [%]
    
    //--------------------------------------------------------------------------
    //                          PER-WHEEL DOWNFORCE (50/50 left/right split)
    //--------------------------------------------------------------------------
    Sample.Wheel_AeroDownforce_N[0] = AeroForces.FrontDownforce_N * 0.5f; // [N] - FL
    Sample.Wheel_AeroDownforce_N[1] = AeroForces.FrontDownforce_N * 0.5f; // [N] - FR
    Sample.Wheel_AeroDownforce_N[2] = AeroForces.RearDownforce_N * 0.5f; // [N] - RL
    Sample.Wheel_AeroDownforce_N[3] = AeroForces.RearDownforce_N * 0.5f; // [N] - RR
    
    //--------------------------------------------------------------------------
    //                          PER-WHEEL LOADS
    //--------------------------------------------------------------------------
    const int32 WheelCount = FMath::Min(AxleData_PT.AxleIDs.Num(), 4);
    for (int32 i = 0; i < WheelCount; ++i)
    {
        Sample.Wheel_StaticLoad_N[i] = AxleData_PT.StaticLoads_LT[i]; // [N] (kg⋅m⋅s⁻²)
        Sample.Wheel_TotalLoad_N[i] = AxleData_PT.ComputedLoads_LT[i]; // [N] (kg⋅m⋅s⁻²)
        Sample.Wheel_LoadTransfer_N[i] = AxleData_PT.ComputedLoads_LT[i] - AxleData_PT.StaticLoads_LT[i]; // [N] (kg⋅m⋅s⁻²)
        Sample.Wheel_AeroContribution_N[i] = Sample.Wheel_AeroDownforce_N[i]; // [N] (kg⋅m⋅s⁻²)
    } // End for (Wheel loads)
    
    //--------------------------------------------------------------------------
    //                          LOAD TRANSFER COMPONENTS
    //--------------------------------------------------------------------------
    const float FrontLoadTransfer = Sample.Wheel_LoadTransfer_N[0] + Sample.Wheel_LoadTransfer_N[1]; // [N] (kg⋅m⋅s⁻²)
    const float RearLoadTransfer = Sample.Wheel_LoadTransfer_N[2] + Sample.Wheel_LoadTransfer_N[3]; // [N] (kg⋅m⋅s⁻²)
    Sample.LongLoadTransfer_N = (RearLoadTransfer - FrontLoadTransfer) * 0.5f; // [N] - Positive = rear bias
    
    const float LeftLoadTransfer = Sample.Wheel_LoadTransfer_N[0] + Sample.Wheel_LoadTransfer_N[2]; // [N] (kg⋅m⋅s⁻²)
    const float RightLoadTransfer = Sample.Wheel_LoadTransfer_N[1] + Sample.Wheel_LoadTransfer_N[3]; // [N] (kg⋅m⋅s⁻²)
    Sample.LatLoadTransfer_N = (RightLoadTransfer - LeftLoadTransfer) * 0.5f; // [N] - Positive = right bias
    
    Sample.LongAccel_G = Rec.α_g_longitudinal;       // [g]
    Sample.LatAccel_G = Rec.α_g_lateral;             // [g]
    
    return Sample;
} // End CollectAerodynamicsSample()

    if (AerodynamicsSampleBuffer.Num() == 0) return;

            S.Time_s, S.Speed_kmh, S.Speed_ms, S.DynamicPressure_Pa,
            S.TotalDrag_N, S.TotalDownforce_N, S.AeroEfficiency,
            S.BodyDrag_N, S.BodyDownforce_N,
            S.UnderbodyDrag_N, S.UnderbodyDownforce_N, S.MinRideHeight_cm, S.RideHeightModifier,
            S.FrontDownforce_N, S.RearDownforce_N, S.AeroBalance_Pct,
            S.Wheel_AeroDownforce_N[0], S.Wheel_AeroDownforce_N[1], S.Wheel_AeroDownforce_N[2], S.Wheel_AeroDownforce_N[3],
            S.Wheel_StaticLoad_N[0], S.Wheel_StaticLoad_N[1], S.Wheel_StaticLoad_N[2], S.Wheel_StaticLoad_N[3],
            S.Wheel_TotalLoad_N[0], S.Wheel_TotalLoad_N[1], S.Wheel_TotalLoad_N[2], S.Wheel_TotalLoad_N[3],
            S.Wheel_LoadTransfer_N[0], S.Wheel_LoadTransfer_N[1], S.Wheel_LoadTransfer_N[2], S.Wheel_LoadTransfer_N[3],
            S.Wheel_AeroContribution_N[0], S.Wheel_AeroContribution_N[1], S.Wheel_AeroContribution_N[2], S.Wheel_AeroContribution_N[3],
            S.LongLoadTransfer_N, S.LatLoadTransfer_N, S.LongAccel_G, S.LatAccel_G
        ));
    }

    // Write to file
    const FString Dir = FPaths::ProjectDir() / TEXT("Source/GRIT/VehicleFramework/Telemetry");
    IPlatformFile& PlatformFile = FPlatformFileManager::Get().GetPlatformFile();
    if (!PlatformFile.DirectoryExists(*Dir))
    {
        PlatformFile.CreateDirectoryTree(*Dir);
    }

    const FString FilePath = Dir / TEXT("AerodynamicsLog.csv");
    FFileHelper::SaveStringArrayToFile(Lines, *FilePath);
    UE_LOG(LogTemp, Log, TEXT("🌪️ Aerodynamics CSV written: %s (%d samples)"), *FilePath, AerodynamicsSampleBuffer.Num());
}

                                                      AERODYNAMICS DEBUG VISUALIZATION

/**
 * Draw debug visualization for aerodynamics system
 * Shows ride height plane, downforce vectors, and drag vectors
 */
void AVehicleSolver::DrawAerodynamicsDebug()
{
    if (!VehicleHull || !PhysicsCallback) return;
    
    const UWorld* World = GetWorld();
    if (!World) return;
    
    const FTransform HullXfm = VehicleHull->GetComponentTransform();
    const FVector HullLocation = HullXfm.GetLocation();
    const FVector ForwardDir = HullXfm.GetUnitAxis(EAxis::X);
    const FVector RightDir = HullXfm.GetUnitAxis(EAxis::Y);
    const FVector UpDir = HullXfm.GetUnitAxis(EAxis::Z);
    
    // Get aero forces from physics callback
    const FAerodynamicForces_PT& AeroForces = PhysicsCallback->AeroForces_PT;
    const float RideHeight_m = CachedRideHeight_m;
    const float RideHeight_cm = RideHeight_m * 100.0f;
    
    //--------------------------------------------------------------------------
    // RIDE HEIGHT VISUALIZATION
    //--------------------------------------------------------------------------
    // Draw a horizontal plane at ride height below the vehicle
    // This shows the effective ground clearance for aerodynamics
    {
        const FVector PlaneCenter = HullLocation - UpDir * RideHeight_cm;
        const float PlaneSize = 150.0f; // [cm] Half-size of debug plane
        
        // Draw ride height plane as a grid
        const FColor RideHeightColor = FColor::Cyan;
        const int32 GridLines = 5;
        
        for (int32 i = -GridLines; i <= GridLines; ++i)
        {
            const float Offset = (PlaneSize / GridLines) * i;
            
            // Longitudinal lines
            const FVector LineStart = PlaneCenter + RightDir * Offset - ForwardDir * PlaneSize;
            const FVector LineEnd = PlaneCenter + RightDir * Offset + ForwardDir * PlaneSize;
            DrawDebugLine(World, LineStart, LineEnd, RideHeightColor, false, -1.0f, 0, 1.0f);
            
            // Lateral lines
            const FVector LineStart2 = PlaneCenter + ForwardDir * Offset - RightDir * PlaneSize;
            const FVector LineEnd2 = PlaneCenter + ForwardDir * Offset + RightDir * PlaneSize;
            DrawDebugLine(World, LineStart2, LineEnd2, RideHeightColor, false, -1.0f, 0, 1.0f);
        }
        
        // Draw vertical lines from hull to ride height plane at corners
        const TArray<FVector> Corners = {
            HullLocation + ForwardDir * PlaneSize + RightDir * PlaneSize,
            HullLocation + ForwardDir * PlaneSize - RightDir * PlaneSize,
            HullLocation - ForwardDir * PlaneSize + RightDir * PlaneSize,
            HullLocation - ForwardDir * PlaneSize - RightDir * PlaneSize
        };
        
        for (const FVector& Corner : Corners)
        {
            const FVector CornerGround = Corner - UpDir * RideHeight_cm;
            DrawDebugLine(World, Corner, CornerGround, FColor::Yellow, false, -1.0f, 0, 2.0f);
        }
        
        // Draw ride height text
        const FString RideHeightText = FString::Printf(TEXT("Ride Height: %.1f cm (%.3f m)"), RideHeight_cm, RideHeight_m);
        DrawDebugString(World, PlaneCenter + ForwardDir * PlaneSize, RideHeightText, nullptr, FColor::Cyan, 0.0f, true, 1.2f);
    }
    
    //--------------------------------------------------------------------------
    // DOWNFORCE VISUALIZATION
    //--------------------------------------------------------------------------
    // Draw downforce vectors at front and rear
    {
        const float ForceScale = 0.02f; // [cm/N] Scale factor for visualization
        
        // Front downforce (at front of vehicle)
        const FVector FrontPos = HullLocation + ForwardDir * 120.0f; // 120cm forward
        const float FrontDownforce_cm = AeroForces.FrontDownforce_N * ForceScale;
        if (FMath::Abs(FrontDownforce_cm) > 1.0f)
        {
            const FVector FrontArrowEnd = FrontPos - UpDir * FrontDownforce_cm;
            DrawDebugDirectionalArrow(World, FrontPos, FrontArrowEnd, 15.0f, FColor::Green, false, -1.0f, 0, 3.0f);
            
            const FString FrontText = FString::Printf(TEXT("Front: %.0f N"), AeroForces.FrontDownforce_N);
            DrawDebugString(World, FrontArrowEnd, FrontText, nullptr, FColor::Green, 0.0f, true, 1.0f);
        }
        
        // Rear downforce (at rear of vehicle)
        const FVector RearPos = HullLocation - ForwardDir * 120.0f; // 120cm rearward
        const float RearDownforce_cm = AeroForces.RearDownforce_N * ForceScale;
        if (FMath::Abs(RearDownforce_cm) > 1.0f)
        {
            const FVector RearArrowEnd = RearPos - UpDir * RearDownforce_cm;
            DrawDebugDirectionalArrow(World, RearPos, RearArrowEnd, 15.0f, FColor::Blue, false, -1.0f, 0, 3.0f);
            
            const FString RearText = FString::Printf(TEXT("Rear: %.0f N"), AeroForces.RearDownforce_N);
            DrawDebugString(World, RearArrowEnd, RearText, nullptr, FColor::Blue, 0.0f, true, 1.0f);
        }
        
        // Total downforce at center
        const FString TotalDownforceText = FString::Printf(TEXT("Total Downforce: %.0f N"), AeroForces.TotalDownforce_N);
        DrawDebugString(World, HullLocation + UpDir * 50.0f, TotalDownforceText, nullptr, FColor::White, 0.0f, true, 1.2f);
    }
    
    //--------------------------------------------------------------------------
    // DRAG VISUALIZATION
    //--------------------------------------------------------------------------
    // Draw drag vector opposing velocity
    {
        const float DragScale = 0.05f; // [cm/N] Scale factor for drag visualization
        const float Drag_cm = AeroForces.TotalDrag_N * DragScale;
        
        if (Drag_cm > 5.0f)
        {
            const FVector DragStart = HullLocation;
            const FVector DragEnd = DragStart - ForwardDir * Drag_cm;
            DrawDebugDirectionalArrow(World, DragStart, DragEnd, 20.0f, FColor::Red, false, -1.0f, 0, 4.0f);
            
            const FString DragText = FString::Printf(TEXT("Drag: %.0f N"), AeroForces.TotalDrag_N);
            DrawDebugString(World, DragEnd, DragText, nullptr, FColor::Red, 0.0f, true, 1.0f);
        }
    }
    
    //--------------------------------------------------------------------------
    // COMPONENT BREAKDOWN (Optional detailed view)
    //--------------------------------------------------------------------------
    // Draw text showing individual aero component contributions
    {
        const FVector TextPos = HullLocation + RightDir * 200.0f + UpDir * 100.0f;
        
        TArray<FString> AeroInfo;
        AeroInfo.Add(FString::Printf(TEXT("=== AERO BREAKDOWN ===")));
        AeroInfo.Add(FString::Printf(TEXT("Wing: %.0f N down, %.0f N drag"), AeroForces.WingDownforce_N, AeroForces.WingDrag_N));
        AeroInfo.Add(FString::Printf(TEXT("Splitter: %.0f N down, %.0f N drag"), AeroForces.SplitterDownforce_N, AeroForces.SplitterDrag_N));
        AeroInfo.Add(FString::Printf(TEXT("Underbody: %.0f N down, %.0f N drag"), AeroForces.UnderbodyDownforce_N, AeroForces.UnderbodyDrag_N));
        AeroInfo.Add(FString::Printf(TEXT("Body: %.0f N lift, %.0f N drag"), AeroForces.BodyLift_N, AeroForces.BodyDrag_N));
        AeroInfo.Add(FString::Printf(TEXT("Balance: F/R = %.0f/%.0f N"), AeroForces.FrontDownforce_N, AeroForces.RearDownforce_N));
        
        float YOffset = 0.0f;
        for (const FString& Line : AeroInfo)
        {
            DrawDebugString(World, TextPos - UpDir * YOffset, Line, nullptr, FColor::Orange, 0.0f, true, 0.9f);
            YOffset += 15.0f;
        }
    }
}
