/*====================================================================================================================================================
                                                    AERODYNAMIC PHYSICS FOUNDATIONS
                                                    
    Core Equations (SI Units):
    ==========================
    1. Dynamic Pressure:        q = 0.5 * ρ * V²                    [Pa]
    2. Aerodynamic Force:       F = q * A * C                       [N]
    3. Ground Effect Modifier:  η(h) = f(h/c, Re, AR)              [-]
    4. Vortex Lift:            L_vortex = 0.5 * ρ * Γ * V * span   [N]
    
    Where:
    - ρ = air density [kg/m³]
    - V = airspeed [m/s]
    - A = reference area [m²]
    - C = force coefficient (Cd, Cl) [-]
    - h = ride height [m]
    - c = chord length [m]
    - Γ = circulation strength [m²/s]
    
    Critical Reynolds Numbers:
    - Re < 3×10⁵:  Laminar flow, low Cl
    - Re > 5×10⁵:  Turbulent flow, enhanced performance
    - Typical race car Re: 2×10⁶ - 8×10⁶
    
    Ground Effect Regions (h/c ratio):
    - h/c > 1.0:   Minimal ground effect (<5% enhancement)
    - 0.3 < h/c < 1.0:  Force enhancement region (10-40% gain)
    - 0.1 < h/c < 0.3:  Maximum downforce region (up to 60% gain)
    - h/c < 0.1:   Stall risk zone (vortex breakdown)
    
    Diffuser Performance (from Cooper et al. 1998, Senior & Zhang 2002):
    - Optimal diffuser angle: 10-17° (depends on length)
    - Maximum downforce at: h/θ ≈ 0.7 (h in meters, θ in radians)
    - Stall threshold: h/L < 0.05 (L = diffuser length)
    - Hysteresis region: exists between attachment/detachment
    
    References:
    - Ahmed et al. (1984): Bluff body aerodynamics
    - Katz (2006): "Aerodynamics of Race Cars" (Annual Review of Fluid Mechanics)
    - Zhang & Zerihan (2003): Ground effect wings
    - Cooper et al. (1998): Diffuser aerodynamics
    - Senior (2002): Diffuser vortex dynamics
====================================================================================================================================================*/

/*====================================================================================================================================================
                                                    FRONTAL AREA AERODYNAMICS (BODY + UNDERBODY)
====================================================================================================================================================*/

/** Physics thread frontal area aerodynamics - body drag/lift + underbody ground effect */
struct FFrontalAreaAero_PT
{
    //--------------------------------------------------------------------------
    // BODY COEFFICIENTS (Full-body including frontal projection)
    //--------------------------------------------------------------------------
    float Cd_Body = 0.28f;                                                       // [-] - Body drag coefficient (GT-R baseline: 0.26-0.28)
    float Cl_Body = 0.10f;                                                       // [-] - Body lift coefficient (positive = lift, negative = downforce)
    float FrontalArea_m2 = 2.17f;                                                // [m²] - Frontal reference area
    
    //--------------------------------------------------------------------------
    // UNDERBODY / FLOOR COEFFICIENTS (Ground effect aerodynamics)
    //--------------------------------------------------------------------------
    float Cd_Underbody = 0.02f;                                                  // [-] - Underbody drag contribution (low due to streamlined flow)
    float Cl_Underbody_Base = -0.15f;                                            // [-] - Base underbody downforce at optimal height
    float UnderbodyArea_m2 = 3.5f;                                               // [m²] - Underbody planform area (flat floor reference)
    
    // Ground effect parameters (from Zhang & Zerihan 2003, Senior 2002)
    float RideHeightSensitivity = 1.5f;                                          // [-] - Downforce gain per cm ride height reduction
    float OptimalRideHeight_cm = 6.0f;                                           // [cm] - Peak performance height (h/c ≈ 0.15-0.3)
    float StallRideHeight_cm = 2.5f;                                             // [cm] - Vortex breakdown threshold (h/c < 0.08)
    
    // Venturi / diffuser interaction parameters
    float VenturiEfficiency = 0.75f;                                             // [-] - Venturi channel efficiency (0.7-0.9 typical)
    float DiffuserExpansionRatio = 1.8f;                                         // [-] - Area ratio (outlet/inlet, 1.5-2.5 typical)
    
    //--------------------------------------------------------------------------
    // ENVIRONMENTAL
    //--------------------------------------------------------------------------
    float AirDensity_kg_m3 = 1.225f;                                             // [kg·m⁻³] - Air density at sea level, 15°C
    
    //--------------------------------------------------------------------------
    // FORCE APPLICATION POINTS (Relative to CoM)
    //--------------------------------------------------------------------------
    FVector BodyAeroCenter_Local = FVector(0.0f, 0.0f, 30.0f);                   // [cm] - Body center of pressure (CoP)
    FVector UnderbodyCenter_Local = FVector(0.0f, 0.0f, -15.0f);                 // [cm] - Underbody CoP (below CoM for stability)
    
    //--------------------------------------------------------------------------
    // COMPUTED TOTALS (Updated each frame)
    //--------------------------------------------------------------------------
    float Cd_Total = 0.30f;                                                      // [-] - Combined drag coefficient
    float Cl_Total = -0.05f;                                                     // [-] - Combined lift coefficient
    float Cl_Underbody_Effective = -0.15f;                                       // [-] - Ride-height-modified underbody Cl
    
    /** Compute total coefficients from body + underbody */
    void ComputeTotals()
    {
        Cd_Total = Cd_Body + Cd_Underbody;
        Cl_Total = Cl_Body + Cl_Underbody_Effective;
    }
    
    /** 
     * Compute ride height modifier for underbody downforce using validated curve
     * Based on Senior (2002), Zhang & Zerihan (2003) experimental data
     * 
     * Flow Regimes:
     * - Region A (h > h_opt):     Attached flow, monotonic increase
     * - Region B (h_stall < h < h_opt): Peak performance, edge vortices coherent
     * - Region C (h < h_stall):   Vortex breakdown, rapid downforce loss
     */
    float ComputeRideHeightModifier(float CurrentRideHeight_cm) const
    {
        // Normalize ride height to characteristic length (underbody chord)
        const float UnderbodyChord_cm = FMath::Sqrt(UnderbodyArea_m2 * 10000.0f); // Approximate chord from area
        const float h_norm = CurrentRideHeight_cm / UnderbodyChord_cm;   // [-] - Normalized ride height (h/c)
        
        if (CurrentRideHeight_cm < StallRideHeight_cm) // Region C: Vortex breakdown (stall)
        {
            // Rapid loss of downforce as vortices burst
            // Loss is non-linear: η = (h/h_stall)^2 for smooth transition
            const float StallRatio = CurrentRideHeight_cm / FMath::Max(StallRideHeight_cm, 0.1f);
            return FMath::Max(0.1f, StallRatio * StallRatio);  // Quadratic loss, floor at 10%
        }
        else if (CurrentRideHeight_cm < OptimalRideHeight_cm) // Region B: Force enhancement zone
        {
            // Peak performance zone - ground effect amplification
            // Linear gain model: η = 1 + k·Δh  (Katz 2006, eq. 3.14)
            const float DeltaHeight_cm = OptimalRideHeight_cm - CurrentRideHeight_cm;
            const float Enhancement = 1.0f + (DeltaHeight_cm * RideHeightSensitivity * 0.01f);
            return FMath::Clamp(Enhancement, 1.0f, 1.6f);  // Cap at 60% enhancement
        }
        else // Region A: Above optimal (diminishing returns)
        {
            // Exponential decay as ground effect weakens
            // η = exp(-k·Δh)  (Zhang & Zerihan 2003)
            const float DeltaHeight_cm = CurrentRideHeight_cm - OptimalRideHeight_cm;
            const float DecayFactor = FMath::Exp(-0.02f * DeltaHeight_cm);  // ~86% at +10cm, ~60% at +25cm
            return FMath::Max(0.5f, DecayFactor);  // Floor at 50% effectiveness
        }
    }
    
    /**
     * Compute diffuser stall risk factor (0 = stable, 1 = imminent stall)
     * Used for telemetry/warnings - alerts driver to unstable conditions
     */
    float ComputeStallRisk(float CurrentRideHeight_cm) const
    {
        if (CurrentRideHeight_cm >= OptimalRideHeight_cm) return 0.0f;  // No risk above optimal
        
        const float MarginAboveStall_cm = CurrentRideHeight_cm - StallRideHeight_cm;
        const float SafetyMargin_cm = OptimalRideHeight_cm - StallRideHeight_cm;
        
        if (MarginAboveStall_cm <= 0.0f) return 1.0f;  // Already stalled
        
        // Linear risk increase as we approach stall height
        return 1.0f - (MarginAboveStall_cm / SafetyMargin_cm);
    }
};

{
    None            UMETA(DisplayName = "None"),
    
    // Front components (generate 25-35% of total downforce in modern race cars)

    RearDiffuser    UMETA(DisplayName = "Rear Diffuser"),                // Rear expansion section (critical component)
    VortexGenerator UMETA(DisplayName = "Vortex Generators"),            // Energize floor boundary layer
    
    // Side components (5-10% of total downforce, primarily flow control)
    Sideskirts      UMETA(DisplayName = "Side Skirts/Fences"),           // Seal underbody from tire wake
    SideSkirtFins   UMETA(DisplayName = "Side Skirt Fins"),              // Induce edge vortices
    
    // Rear components (generate 30-40% of total downforce)
    RearWing        UMETA(DisplayName = "Rear Wing"),                    // Multi-element inverted airfoil

    ActiveAirBrake  UMETA(DisplayName = "Active Air Brake")              // Deployable high-drag surface

/*====================================================================================================================================================
                                                    FRONT SPLITTER / AIR DAM
                                                    
    Physics: Creates downforce via high-pressure stagnation above and low-pressure suction below.
    - Stagnation pressure: p₀ = p + 0.5ρV²
    - Suction zone: accelerated flow beneath splitter (Venturi effect)
    - Edge vortices: seal underbody from tire wake
    
    Typical Performance (from Bhattacharjee et al. 2019, NASCAR study):
    - Cl: -0.1 to -0.3 (depends on overhang length and ground clearance)
    - Cd: +0.02 to +0.05 (drag penalty for downforce generation)
    - L/D ratio: 2:1 to 4:1
    - Optimal ground clearance: 15-30mm
    - Overhang length: typically 30-80mm forward of bumper
====================================================================================================================================================*/

USTRUCT(BlueprintType)
struct FFrontSplitterSpec : public FAeroComponentBase
{
    GENERATED_BODY()
    
    UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Aerodynamics|Splitter", meta = (ClampMin = "0.0", ClampMax = "20.0"))
    float OverhangLength_cm = 5.0f;                                              // [cm] - Extension forward of bumper (3-8cm typical)
    
    UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Aerodynamics|Splitter", meta = (ClampMin = "0.5", ClampMax = "5.0"))
    float GroundClearance_cm = 2.0f;                                             // [cm] - Height above ground (1.5-3cm optimal)
    
    UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Aerodynamics|Splitter")
    bool bHasEndPlates = true;                                                   // [-] - Vertical plates at sides (critical for performance)
    
    UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Aerodynamics|Splitter")
    float EndPlateHeight_cm = 5.0f;                                              // [cm] - Vertical fence height
    
    /**
     * Compute ground effect modifier for splitter downforce
     * Very sensitive to ride height - performance degrades rapidly if too high or too low
     */
    float ComputeGroundEffectModifier(float CurrentRideHeight_cm) const
    {
        const float h_ratio = CurrentRideHeight_cm / GroundClearance_cm;
        
        if (h_ratio < 0.5f) // Too low - restricted inlet flow
        {
            return 0.6f + (h_ratio * 0.8f);  // 60% at ground contact, 100% at design height
        }
        else if (h_ratio < 1.5f) // Optimal zone
        {
            return 1.0f;
        }
        else // Too high - weakened ground effect
        {
            return FMath::Max(0.3f, 1.0f / FMath::Sqrt(h_ratio));  // Inverse square root decay
        }
    }
};

struct FFrontSplitterSpec_PT : public FAeroComponentBase_PT
{
    float OverhangLength_cm = 5.0f;
    float GroundClearance_cm = 2.0f;
    bool bHasEndPlates = true;
    float EndPlateHeight_cm = 5.0f;
};

    - L/D ratio: 2:1 to 4:1 (lower than wings due to pressure-based lift)

/*====================================================================================================================================================

    - Operates in ground effect (h/c = 0.5-1.5 for rear wings)
    - Multi-element design: main element + flap(s) for increased Cl_max
    - DRS (Drag Reduction System): reduces AoA to trade downforce for top speed

    - Cd: 0.15 to 0.40 (depends on element count, AoA)
    - L/D ratio: 4:1 to 8:1 (efficiency decreases with more downforce)
    - Ground effect gain: 15-40% (closer to ground = more downforce)

     * Compute ground effect modifier for rear wing

        const float MaxGain = 1.3f;
        const float GainFactor = (1.5f - HeightToChordRatio) / 1.3f;  // Normalized distance from free-stream
        return FMath::Lerp(1.0f, MaxGain, FMath::Clamp(GainFactor, 0.0f, 1.0f));
    }

/*====================================================================================================================================================
                                                    REAR DIFFUSER (Critical Component)
                                                    
    Physics: Expands underbody flow, recovering pressure and generating downforce.
    - Bernoulli equation: p + 0.5ρV² = constant
    - Expansion accelerates flow at throat, lowers pressure
    - Edge vortices stabilize flow attachment
    
    Key Parameters (from Cooper et al. 1998, Senior 2002):
    - Expansion ratio (AR): outlet_area / inlet_area = 1.5-2.5 optimal
    - Diffuser angle (θ): 10-17° typical (steeper = more power, but stall risk)
    - Ride height sensitivity: Critical parameter
    - Stall condition: h/L < 0.05 (L = diffuser length)
    
    Typical Performance:
    - Contributes 40-50% of total downforce in modern race cars
    - Cl_contribution: -0.3 to -0.8 (vehicle Cl, not diffuser-only)
    - Drag penalty: Low (Cd +0.02 to +0.05)
    - L/D ratio: 10:1 to 20:1 (most efficient aero device)
====================================================================================================================================================*/

    float ExpansionRatio = 2.0f;                                                 // [-] - Area ratio (outlet/inlet, 1.5-2.5 typical)

    UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Aerodynamics|RearDiffuser")
    bool bHasEndPlates= true;                                                   // [-] - Side fences (critical for performance)
UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Aerodynamics|RearDiffuser")
float RideHeightSensitivity = 2.5f;                                          // [-] - Downforce gain per cm ride height reduction

UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Aerodynamics|RearDiffuser")
float OptimalRideHeight_cm = 5.0f;                                           // [cm] - Peak performance height

UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Aerodynamics|RearDiffuser")
float StallRideHeight_cm = 2.0f;                                             // [cm] - Stall threshold

/**
 * Compute diffuser performance modifier based on ride height
 * Most critical aerodynamic component - performance degrades catastrophically at low ride heights
 */
float ComputePerformanceModifier(float CurrentRideHeight_cm) const
{
    // Stall zone (h < h_stall): Vortex breakdown, separation
    if (CurrentRideHeight_cm < StallRideHeight_cm)
    {
        // Severe non-linear loss (cubic decay from Senior 2002 experimental data)
        const float h_ratio = CurrentRideHeight_cm / StallRideHeight_cm;
        return FMath::Max(0.2f, h_ratio * h_ratio * h_ratio);  // Cubic loss, floor at 20%
    }
    
    // Optimal zone (h_stall < h < h_opt): Maximum downforce
    else if (CurrentRideHeight_cm < OptimalRideHeight_cm)
    {
        // Linear enhancement in this narrow band
        const float DeltaH = OptimalRideHeight_cm - CurrentRideHeight_cm;
        return 1.0f + (DeltaH * RideHeightSensitivity * 0.01f);  // Up to 60% gain
    }
    
    // High ride height (h > h_opt): Diminishing returns
    else
    {
        // Exponential decay as ground effect weakens
        const float DeltaH = CurrentRideHeight_cm - OptimalRideHeight_cm;
        return FMath::Max(0.4f, FMath::Exp(-0.03f * DeltaH));  // ~74% at +10cm, ~55% at +20cm
    }
}

/**
 * Compute diffuser stall risk (0 = stable, 1 = stalled)
 * Used for telemetry warnings
 */
float ComputeStallRisk(float CurrentRideHeight_cm) const
{
    if (CurrentRideHeight_cm >= OptimalRideHeight_cm) return 0.0f;
    
    const float Margin = CurrentRideHeight_cm - StallRideHeight_cm;
    const float SafetyBand = OptimalRideHeight_cm - StallRideHeight_cm;
    
    if (Margin <= 0.0f) return 1.0f;  // Already stalled
    
    // Non-linear risk (squared to emphasize danger zone)
    const float RiskLinear = 1.0f - (Margin / SafetyBand);
    return RiskLinear * RiskLinear;  // Quadratic increase
}
